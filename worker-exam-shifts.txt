/**
 * ============================================================================
 * ⚡ CLOUDFLARE WORKER: DEDUPLICATED EXAM DATES, TIMES & LANGUAGES API
 * ============================================================================
 * 
 * 🎯 PURPOSE:
 * Given an `exam_id` (numeric ID like 44, slug like 'rrb-ntpc', or code),
 * queries the `user_ranks` table in Cloudflare D1 and returns:
 *   1. Unique (exam_date, exam_time) shifts — ZERO duplicates (single single data).
 *   2. Grouped shift times by exam_date.
 *   3. Unique paper_language list — ZERO duplicates.
 *   4. Clean JSON API + Smart Web UI with Exam Selector Dropdown & DB Diagnostics.
 * 
 * 🗄️ DATABASE TABLES USED:
 *   - `user_ranks` (stores candidate exam_date, exam_time, paper_language)
 *   - `exams` (resolves exam_id <-> slug <-> external_exam_id)
 * 
 * ⚠️ IMPORTANT CLOUDFLARE REQUIREMENT:
 * You MUST bind your D1 Database in Cloudflare Worker Settings:
 * Worker Dashboard -> Settings -> Bindings -> Add D1 Database Binding:
 *   - Variable Name: DB
 *   - Database: (Select your D1 database)
 */

/**
 * 🛡️ Helper: Strict SQL Injection Sanitization
 */
function sanitizeSql(str) {
  if (str === null || str === undefined) return "";
  return String(str)
    .replace(/'/g, "''")
    .replace(/\\/g, "\\\\")
    .replace(/\0/g, "")
    .trim();
}

/**
 * 🗄️ Helper: Query Cloudflare D1 (Native env.DB First, REST API Fallback)
 */
async function queryD1(env, sql) {
  // 1. Native Cloudflare D1 Binding (Fastest & 0-overhead)
  if (env && env.DB) {
    try {
      const res = await env.DB.prepare(sql).all();
      if (res && res.results) return res.results;
      if (res) return res;
    } catch (e) {
      console.error("D1 Native Query Error:", e);
      throw new Error(`D1 Query Error: ${e.message}`);
    }
  }

  // 2. Cloudflare D1 REST API Fallback
  const accountId = (env && env.CF_ACCOUNT_ID) || DEFAULT_CF_ACCOUNT_ID;
  const databaseId = (env && env.CF_DATABASE_ID) || DEFAULT_CF_DATABASE_ID;
  const apiToken = (env && env.CF_API_TOKEN) || DEFAULT_CF_API_TOKEN;

  if (!env || !env.DB) {
    console.warn("⚠️ Warning: env.DB binding not detected. Attempting REST API fallback...");
  }

  try {
    const apiUrl = `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/query`;
    const response = await fetch(apiUrl, {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${apiToken}`,
        "Content-Type": "application/json"
      },
      body: JSON.stringify({ sql: sql })
    });
    if (response.ok) {
      const jsonRes = await response.json();
      if (jsonRes && jsonRes.result && jsonRes.result[0]) {
        if (jsonRes.result[0].results) return jsonRes.result[0].results;
        return jsonRes.result[0];
      }
    } else {
      const errText = await response.text();
      console.error("D1 REST Fallback Failed:", response.status, errText);
    }
  } catch (e) {
    console.error("D1 REST Fallback Error:", e);
  }

  return [];
}

/**
 * 🚀 Main Cloudflare Worker Fetch Handler
 */
export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const pathname = url.pathname.toLowerCase();
    const method = request.method.toUpperCase();

    // 1. CORS Preflight
    if (method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders, status: 204 });
    }

    // 2. Check D1 Database Connection Status
    const isD1Bound = !!(env && env.DB);

    // 3. Diagnostics Endpoint (/debug or /check_db)
    if (pathname === "/debug" || pathname === "/check_db" || url.searchParams.get("action") === "debug") {
      let d1Status = isD1Bound ? "CONNECTED (env.DB is active)" : "MISSING (env.DB not bound in Cloudflare dashboard)";
      let tableCounts = {};
      let errorMsg = null;

      try {
        const userRanksCount = await queryD1(env, "SELECT COUNT(*) AS count FROM user_ranks");
        const examsCount = await queryD1(env, "SELECT COUNT(*) AS count FROM exams");
        const sampleExams = await queryD1(env, "SELECT id, title, slug, external_exam_id FROM exams ORDER BY id DESC LIMIT 10");
        const activeExamsInRanks = await queryD1(env, "SELECT exam_slug, exam_id, COUNT(*) AS candidates FROM user_ranks GROUP BY exam_slug, exam_id ORDER BY candidates DESC LIMIT 15");

        tableCounts = {
          user_ranks_total: userRanksCount?.[0]?.count || 0,
          exams_total: examsCount?.[0]?.count || 0,
          active_exams_in_user_ranks: activeExamsInRanks || [],
          sample_exams: sampleExams || []
        };
      } catch (e) {
        errorMsg = e.message;
      }

      return new Response(JSON.stringify({
        success: !errorMsg,
        d1_binding_status: d1Status,
        is_d1_bound: isD1Bound,
        table_counts: tableCounts,
        error: errorMsg,
        help: !isD1Bound ? "Go to Cloudflare Dashboard -> Workers -> Your Worker -> Settings -> Bindings -> Add D1 Database Binding named 'DB'" : "D1 is properly bound."
      }, null, 2), {
        headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" }
      });
    }

    // 4. Exams List Endpoint (Used by Web UI dropdown)
    if (pathname === "/exams" || url.searchParams.get("action") === "exams") {
      try {
        const exams = await queryD1(env, "SELECT id, title, slug, external_exam_id FROM exams WHERE is_visible = 1 ORDER BY set_on_top DESC, id DESC");
        return new Response(JSON.stringify(exams || []), {
          headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" }
        });
      } catch (e) {
        return new Response(JSON.stringify([]), { headers: corsHeaders });
      }
    }

    // 5. Extract Input Parameters
    let examId = url.searchParams.get("exam_id") || url.searchParams.get("id") || url.searchParams.get("exam_slug") || url.searchParams.get("slug") || "";

    if (method === "POST") {
      try {
        const contentType = request.headers.get("content-type") || "";
        if (contentType.includes("application/json")) {
          const body = await request.json().catch(() => ({}));
          examId = body.exam_id || body.id || body.exam_slug || body.slug || examId;
        } else if (contentType.includes("application/x-www-form-urlencoded") || contentType.includes("multipart/form-data")) {
          const form = await request.formData().catch(() => new Map());
          examId = form.get("exam_id") || form.get("id") || form.get("exam_slug") || form.get("slug") || examId;
        }
      } catch (e) {}
    }

    examId = String(examId).trim();

    // 6. Serve Web UI if visited directly in browser without exam_id
    const wantsJson = url.searchParams.has("json") || 
                      url.searchParams.has("exam_id") || 
                      method === "POST" || 
                      (request.headers.get("accept") && request.headers.get("accept").includes("application/json"));

    if (!examId && !wantsJson && method === "GET") {
      return renderWebUi(isD1Bound);
    }

    // 7. Validate exam_id
    if (!examId) {
      return new Response(JSON.stringify({
        success: false,
        error: "Missing exam_id parameter. Please provide exam_id via ?exam_id=... or POST JSON body.",
        d1_bound: isD1Bound,
        example: `${url.origin}/?exam_id=44`
      }, null, 2), {
        status: 400,
        headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" }
      });
    }

    const cleanExamId = sanitizeSql(examId);

    try {
      // 8. SMART RESOLUTION: Check `exams` table to resolve numeric ID <-> Slug <-> External Exam ID
      let matchedExamSlug = "";
      let matchedExternalId = "";
      let matchedExamTitle = "";

      try {
        const examLookupSql = `
          SELECT id, slug, external_exam_id, title 
          FROM exams 
          WHERE id = '${cleanExamId}' 
             OR slug = '${cleanExamId}' 
             OR external_exam_id = '${cleanExamId}'
             OR slug LIKE '%${cleanExamId}%'
             OR title LIKE '%${cleanExamId}%'
          LIMIT 1;
        `;
        const examRows = await queryD1(env, examLookupSql);
        if (examRows && examRows.length > 0) {
          matchedExamSlug = examRows[0].slug || "";
          matchedExternalId = examRows[0].external_exam_id || "";
          matchedExamTitle = examRows[0].title || "";
        }
      } catch (lookupErr) {
        console.warn("Exam lookup fallback:", lookupErr);
      }

      // 9. Build Comprehensive WHERE Clause to Match All Candidate Records
      const conditions = [];
      conditions.push(`exam_id = '${cleanExamId}'`);
      conditions.push(`exam_slug = '${cleanExamId}'`);
      if (matchedExamSlug && matchedExamSlug !== cleanExamId) {
        conditions.push(`exam_slug = '${matchedExamSlug}'`);
      }
      if (matchedExternalId) {
        conditions.push(`exam_id = '${matchedExternalId}'`);
        conditions.push(`url LIKE '%${matchedExternalId}%'`);
      }
      // Also match partial slug if search looks like a slug
      if (cleanExamId.includes("-")) {
        conditions.push(`exam_slug LIKE '%${cleanExamId}%'`);
      }

      const whereMatch = `(${conditions.join(" OR ")})`;

      // 10. Query Deduplicated Exam Dates & Exam Times (SQL DISTINCT)
      const shiftsSql = `
        SELECT DISTINCT 
          TRIM(exam_date) AS exam_date, 
          TRIM(exam_time) AS exam_time 
        FROM user_ranks 
        WHERE ${whereMatch}
          AND exam_date IS NOT NULL 
          AND TRIM(exam_date) != '' 
          AND LOWER(exam_date) != 'null'
          AND exam_time IS NOT NULL 
          AND TRIM(exam_time) != ''
          AND LOWER(exam_time) != 'null'
        ORDER BY exam_date ASC, exam_time ASC;
      `;

      // 11. Query Deduplicated Paper Languages (SQL DISTINCT)
      const langsSql = `
        SELECT DISTINCT 
          TRIM(paper_language) AS paper_language 
        FROM user_ranks 
        WHERE ${whereMatch}
          AND paper_language IS NOT NULL 
          AND TRIM(paper_language) != '' 
          AND LOWER(paper_language) != 'null'
        ORDER BY paper_language ASC;
      `;

      // 12. Execute Both Queries in Parallel
      const [shiftRows, langRows] = await Promise.all([
        queryD1(env, shiftsSql),
        queryD1(env, langsSql)
      ]);

      // 13. Double In-Memory Deduplication & Logical Grouping
      const seenShifts = new Set();
      const shifts = [];
      const dateSet = new Set();
      const groupedByDate = {};

      (shiftRows || []).forEach(row => {
        const d = (row.exam_date || "").trim();
        const t = (row.exam_time || "").trim();
        if (!d || !t || d.toLowerCase() === "null" || t.toLowerCase() === "null") return;

        const uniqueKey = `${d}___${t}`;
        if (!seenShifts.has(uniqueKey)) {
          seenShifts.add(uniqueKey);
          shifts.push({ exam_date: d, exam_time: t });
          dateSet.add(d);

          if (!groupedByDate[d]) groupedByDate[d] = [];
          if (!groupedByDate[d].includes(t)) {
            groupedByDate[d].push(t);
          }
        }
      });

      // Deduplicate Paper Languages (Case-insensitive deduplication)
      const seenLangs = new Set();
      const languages = [];
      (langRows || []).forEach(row => {
        const lang = (row.paper_language || "").trim();
        if (!lang || lang.toLowerCase() === "null") return;

        const lower = lang.toLowerCase();
        if (!seenLangs.has(lower)) {
          seenLangs.add(lower);
          languages.push(lang);
        }
      });

      const sortedDates = Array.from(dateSet).sort();
      languages.sort((a, b) => a.localeCompare(b));

      // 14. Response Data
      const responseData = {
        success: true,
        exam_id: examId,
        resolved_exam: {
          title: matchedExamTitle || null,
          slug: matchedExamSlug || null,
          external_id: matchedExternalId || null
        },
        d1_database_bound: isD1Bound,
        table: "user_ranks",
        total_shifts: shifts.length,
        total_dates: sortedDates.length,
        total_languages: languages.length,
        dates: sortedDates,
        shifts: shifts,
        grouped_by_date: groupedByDate,
        paper_languages: languages,
        note: shifts.length === 0 ? (!isD1Bound ? "⚠️ env.DB binding is missing in Cloudflare Dashboard! Please add D1 Database binding named 'DB'." : "No candidate entries found in user_ranks for this exam.") : undefined
      };

      return new Response(JSON.stringify(responseData, null, 2), {
        status: 200,
        headers: {
          ...corsHeaders,
          "Content-Type": "application/json; charset=utf-8",
          "Cache-Control": "public, max-age=60, s-maxage=300"
        }
      });

    } catch (err) {
      console.error("Worker Execution Error:", err);
      return new Response(JSON.stringify({
        success: false,
        error: "Internal Server Error while querying user_ranks",
        d1_bound: isD1Bound,
        message: err.message || String(err)
      }, null, 2), {
        status: 500,
        headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" }
      });
    }
  }
};
