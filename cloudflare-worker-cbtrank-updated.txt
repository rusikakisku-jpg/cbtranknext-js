/**
 * Helper: Query Cloudflare D1 Database (Native Binding First, REST API Fallback)
 */
async function queryD1(env, sql) {
  // 1. Native Cloudflare D1 Binding (Fastest & Most Secure)
  if (env && env.DB) {
    try {
      const res = await env.DB.prepare(sql).all();
      if (res && res.results) return res.results;
      if (res) return res;
    } catch (e) {
      console.error("D1 Native Query Error:", e);
    }
  }

  // 2. Cloudflare D1 REST API Fallback
  const accountId = (env && env.CF_ACCOUNT_ID) || DEFAULT_CF_ACCOUNT_ID;
  const databaseId = (env && env.CF_DATABASE_ID) || DEFAULT_CF_DATABASE_ID;
  const apiToken = (env && env.CF_API_TOKEN) || DEFAULT_CF_API_TOKEN;

  try {
    const apiUrl = `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/query`;
    const response = await fetch(apiUrl, {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${apiToken}`,
        "Content-Type": "application/json"
      },
      body: JSON.stringify({ sql: sql })
    });
    if (response.ok) {
      const jsonRes = await response.json();
      if (jsonRes && jsonRes.result && jsonRes.result[0]) {
        if (jsonRes.result[0].results) return jsonRes.result[0].results;
        return jsonRes.result[0];
      }
    }
  } catch (e) {
    console.error("D1 REST Fallback Error:", e);
  }

  return [];
}

/**
 * Helper: Strict SQL String Sanitization
 */
function sanitizeSql(str) {
  if (str === null || str === undefined) return '';
  return String(str)
    .replace(/'/g, "''")
    .replace(/\\/g, "\\\\")
    .replace(/\0/g, "");
}

/**
 * Helper: Verify if incoming request is from authorized Admin or Server
 */
function isAuthorizedAdmin(request, env, url) {
  const adminSecret = (env && (env.ADMIN_API_KEY || env.API_SECRET)) || DEFAULT_ADMIN_SECRET;
  const authHeader = request.headers.get("Authorization") || "";
  const apiKeyHeader = request.headers.get("x-api-key") || request.headers.get("x-admin-key") || request.headers.get("x-api-secret") || "";
  const queryKey = url.searchParams.get("admin_key") || url.searchParams.get("api_key") || url.searchParams.get("key") || "";

  if (authHeader === `Bearer ${adminSecret}` || authHeader === adminSecret) return true;
  if (apiKeyHeader === adminSecret) return true;
  if (queryKey === adminSecret) return true;
  return false;
}

          const userId = sanitizeSql((body.user_id || '').toString());
          const userUrl = sanitizeSql((body.url || ''));
          const examSlug = sanitizeSql((body.exam_slug || 'general'));
          const paperLang = sanitizeSql((body.paper_language || ''));
          const urlHash = sanitizeSql((body.url_hash || ''));
          let totalMarks = parseFloat(body.total_marks);
          if (isNaN(totalMarks)) totalMarks = 0;

          const examDate = sanitizeSql((body.exam_date || ''));
          const examTime = sanitizeSql((body.exam_time || ''));
          const examId = sanitizeSql((body.exam_id || ''));
          const location = sanitizeSql((body.location || ''));
          const gender = sanitizeSql((body.gender || ''));
          const category = sanitizeSql((body.category || ''));
          const domain = sanitizeSql((body.domain || ''));
          const createdAt = getISTTimestamp();

          // 🔍 SMART CHECK: Isolate record by Roll Number (user_id) + Exam ID/Slug to prevent overwriting attempts of different exams
          let checkSql = "";
          let whereCondition = "";
          if (userId && (examId || (examSlug && examSlug !== 'general'))) {
            const examFilter = examId ? `(exam_id = '${examId}' OR exam_slug = '${examSlug}')` : `exam_slug = '${examSlug}'`;
            checkSql = `SELECT user_id FROM user_ranks WHERE user_id = '${userId}' AND ${examFilter} LIMIT 1`;
            whereCondition = `user_id = '${userId}' AND ${examFilter}`;
          } else if (userId) {
            checkSql = `SELECT user_id FROM user_ranks WHERE user_id = '${userId}' LIMIT 1`;
            whereCondition = `user_id = '${userId}'`;
          } else {
            checkSql = `SELECT user_id FROM user_ranks WHERE url = '${userUrl}' LIMIT 1`;
            whereCondition = `url = '${userUrl}'`;
          }

          const existing = await queryD1(env, checkSql);

          if (existing && existing.length > 0) {
            // 🔄 UPDATE existing candidate attempt record
            let updateClause = `url = '${userUrl}', total_marks = ${totalMarks}, updated_at = '${createdAt}'`;
            if (paperLang) updateClause += `, paper_language = '${paperLang}'`;
            if (examDate) updateClause += `, exam_date = '${examDate}'`;
            if (examTime) updateClause += `, exam_time = '${examTime}'`;
            if (examId) updateClause += `, exam_id = '${examId}'`;
            if (examSlug && examSlug !== 'general') updateClause += `, exam_slug = '${examSlug}'`;
            if (location) updateClause += `, location = '${location}'`;
            if (gender) updateClause += `, gender = '${gender}'`;
            if (category) updateClause += `, category = '${category}'`;
            if (domain) updateClause += `, domain = '${domain}'`;

            const updateSql = `UPDATE user_ranks SET ${updateClause} WHERE ${whereCondition}`;
            await queryD1(env, updateSql);

            return new Response(JSON.stringify({
              success: true,
              message: "User rank record updated successfully in user_ranks",
              action: "updated",
              user_id: userId,
              updated_at: createdAt
            }), { headers: corsHeaders, status: 200 });

          } else {
            // ➕ INSERT brand new candidate record
            const insertSql = `
              INSERT INTO user_ranks 
              (user_id, url, exam_slug, paper_language, url_hash, total_marks, exam_date, exam_time, exam_id, location, gender, category, domain, created_at, updated_at)
              VALUES 
              ('${userId}', '${userUrl}', '${examSlug}', '${paperLang}', '${urlHash}', ${totalMarks}, '${examDate}', '${examTime}', '${examId}', '${location}', '${gender}', '${category}', '${domain}', '${createdAt}', '${createdAt}')
            `;

            await queryD1(env, insertSql);

            return new Response(JSON.stringify({
              success: true,
              message: "User rank inserted successfully into user_ranks",
              action: "inserted",
              user_id: userId,
              created_at: createdAt
            }), { headers: corsHeaders, status: 201 });
          }

        } catch (err) {
          return new Response(JSON.stringify({ success: false, error: err.message }), { headers: corsHeaders, status: 500 });
        }
      }

      // 🚫 STRICT AUTH CHECK ON GET: Block all direct unauthorized public GET requests
      if (!isAuthorizedAdmin(request, env, url)) {
        return new Response(JSON.stringify({ 
          success: false, 
          error: "Unauthorized Access: Direct public access to user_ranks is strictly restricted. Provide valid Admin Authorization header or key."

      }

      // GET: Return data strictly for Authorized Admin (Sorted by total_marks DESC)

      } else if (qExamSlug && qExamSlug !== 'general') {

        } catch (e) {}
      }

      return new Response(JSON.stringify({

        table: "user_ranks",

        data: results || [] 
      }), { headers: corsHeaders, status: 200 });
    }

    // =========================================================================
    // 9️⃣ 🔒 STRICTLY PROTECTED VALID ANSWERKEY URLS ENDPOINT (/valid_answerkey_urls)
    // =========================================================================
    if (pathname === '/valid_answerkey_urls' || pathname === '/api/valid_answerkey_urls' || pathname === '/valid_urls' || pathname === '/api/valid_urls' || pathname === '/valid' || pathname === '/api/valid') {

          url: inputUrl,
          created_at: createdAt
        }), { headers: corsHeaders, status: 201 });
      }

      // 🚫 STRICT AUTH CHECK ON GET: Block direct unauthorized access to full URL list
      if (!isAuthorizedAdmin(request, env, url)) {
        return new Response(JSON.stringify({ 
          success: false, 
          error: "Unauthorized Access: Direct public access to valid_answerkey_urls is strictly restricted." 
        }), { headers: corsHeaders, status: 401 });
      }

      // If Admin queries specific URL
      if (inputUrl) {
        const cleanUrl = sanitizeSql(inputUrl);
        const sql = `SELECT id, url, created_at FROM valid_answerkey_urls WHERE url = '${cleanUrl}' ORDER BY id DESC LIMIT 1`;
        const results = await queryD1(env, sql);
        const exists = results && results.length > 0;
        return new Response(JSON.stringify({ 
          success: true, 
          table: "valid_answerkey_urls", 
          exists: exists, 
          data: exists ? results[0] : null 
        }), { headers: corsHeaders, status: 200 });
      }

      // If Admin queries full list
      const limit = Math.min(parseInt(url.searchParams.get('limit') || '100', 10), 1000);
      const sql = `SELECT id, url, created_at FROM valid_answerkey_urls ORDER BY id DESC LIMIT ${limit}`;
      const results = await queryD1(env, sql);
      return new Response(JSON.stringify({ 
        success: true, 
        table: "valid_answerkey_urls", 
        total: Array.isArray(results) ? results.length : 0, 
        data: results || [] 
      }), { headers: corsHeaders, status: 200 });
    }

    // =========================================================================
    // 🔟 🔒 STRICTLY PROTECTED INVALID ANSWERKEY URLS ENDPOINT (/invalid_answerkey_urls)
    // =========================================================================
    if (pathname === '/invalid_answerkey_urls' || pathname === '/api/invalid_answerkey_urls' || pathname === '/invalid_urls' || pathname === '/api/invalid_urls' || pathname === '/invalid' || pathname === '/api/invalid') {
      let inputUrl = url.searchParams.get('url') ? url.searchParams.get('url').trim() : '';
      if (request.method === 'POST') {
        try {
          const body = await request.clone().json();
          if (body && body.url) inputUrl = body.url.trim();
        } catch (e) {}
      }

      const isInsertRequest = request.method === 'POST' || url.searchParams.get('action') === 'insert' || url.searchParams.get('insert') === '1';

      if (isInsertRequest) {
        if (!inputUrl) {
          return new Response(JSON.stringify({ success: false, error: "Please provide a valid 'url' field in JSON body or query parameter." }), { headers: corsHeaders, status: 400 });
        }

        const cleanUrl = sanitizeSql(inputUrl);
        const createdAt = getISTTimestamp();

        // Prevent duplicate invalid URL inserts
        const existingCheck = await queryD1(env, `SELECT id FROM invalid_answerkey_urls WHERE url = '${cleanUrl}' LIMIT 1`);
        if (!existingCheck || existingCheck.length === 0) {
          const insertSql = `INSERT INTO invalid_answerkey_urls (url, created_at) VALUES ('${cleanUrl}', '${createdAt}')`;
          await queryD1(env, insertSql);
        }

        return new Response(JSON.stringify({
          success: true,
          message: "URL processed successfully in invalid_answerkey_urls",
          table: "invalid_answerkey_urls",
          url: inputUrl,
          created_at: createdAt
        }), { headers: corsHeaders, status: 201 });
      }

      // 🚫 STRICT AUTH CHECK ON GET: Block direct unauthorized access
      if (!isAuthorizedAdmin(request, env, url)) {
        return new Response(JSON.stringify({ 
          success: false, 
          error: "Unauthorized Access: Direct public access to invalid_answerkey_urls is strictly restricted." 
        }), { headers: corsHeaders, status: 401 });
      }

      if (inputUrl) {
        const cleanUrl = sanitizeSql(inputUrl);
        const sql = `SELECT id, url, created_at FROM invalid_answerkey_urls WHERE url = '${cleanUrl}' ORDER BY id DESC LIMIT 1`;
        const results = await queryD1(env, sql);
        const exists = results && results.length > 0;
        return new Response(JSON.stringify({ 
          success: true, 
          table: "invalid_answerkey_urls", 
          exists: exists, 
          data: exists ? results[0] : null 
        }), { headers: corsHeaders, status: 200 });
      }

      const limit = Math.min(parseInt(url.searchParams.get('limit') || '100', 10), 1000);
      const sql = `SELECT id, url, created_at FROM invalid_answerkey_urls ORDER BY id DESC LIMIT ${limit}`;
      const results = await queryD1(env, sql);
      return new Response(JSON.stringify({ 
        success: true, 
        table: "invalid_answerkey_urls", 
        total: Array.isArray(results) ? results.length : 0, 
        data: results || [] 
      }), { headers: corsHeaders, status: 200 });
    }

    // =========================================================================
    // 1️⃣1️⃣ ⚡ CALCULATOR API PROXY ENDPOINT (/calculator_api, /api/calculator_api)
    // Tries Dedicated Fast VPS First (Instant Response) with Fallback to Render
    // =========================================================================
    if (pathname === '/calculator_api' || pathname === '/api/calculator_api') {
      let targetAnswerKeyUrl = url.searchParams.get('url') ? url.searchParams.get('url').trim() : '';
      if (request.method === 'POST') {
        try {
          const body = await request.clone().json();
          if (body && body.url) targetAnswerKeyUrl = body.url.trim();
        } catch (e) {}
      }

      if (!targetAnswerKeyUrl) {
        return new Response(JSON.stringify({

        }, null, 2), { headers: corsHeaders, status: 400 });
      }

      try {
        if (targetAnswerKeyUrl.includes('%3A') || targetAnswerKeyUrl.includes('%2F')) {
          targetAnswerKeyUrl = decodeURIComponent(targetAnswerKeyUrl);
        }
      } catch (e) {}

      const primaryApi = (env && env.CALCULATOR_API_URL) || `${CALCULATOR_VPS_PRIMARY}?url=${encodeURIComponent(targetAnswerKeyUrl)}`;
      const backupApi = `${CALCULATOR_RENDER_BACKUP}?url=${encodeURIComponent(targetAnswerKeyUrl)}`;

      let vpsRes = null;
      try {
        // Try dedicated VPS API first (Fast, no cold start)
        vpsRes = await fetch(primaryApi, {
          method: "GET",
          headers: {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/130.0.0.0 Safari/537.36",

        vpsRes = null;
      }

      // If primary VPS timed out or failed, seamlessly fallback to backup Render endpoint
      if (!vpsRes || !vpsRes.ok) {
        try {
          vpsRes = await fetch(backupApi, {
            method: "GET",
            headers: {
              "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/130.0.0.0 Safari/537.36",
              "Accept": "application/json, text/plain, */*"
            }
          });

            success: false,
            error: `Failed to connect to Calculator APIs: ${err.message}`

      } else {
        return new Response(JSON.stringify({
          success: false,
          error: `Calculator API responded with status ${vpsRes ? vpsRes.status : 500}`
        }), { headers: corsHeaders, status: vpsRes ? vpsRes.status : 500 });
      }
    }

    // =========================================================================
    if (pathname === '/digialm' || pathname === '/api/digialm') {

      }

      try {

          legacyUrl = decodeURIComponent(legacyUrl);
        }
      } catch (e) {}

      const ep = `${CALCULATOR_VPS_PRIMARY}?url=${encodeURIComponent(legacyUrl)}`;
      try {
        const vpsRes = await fetch(ep, {
          method: "GET",
          headers: {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            "Accept": "application/json"
          }
        });
        if (vpsRes.ok) {
          const rawText = await vpsRes.text();
          return new Response(rawText, { headers: corsHeaders, status: 200 });
        }
      } catch (err) {}

      // Fallback
      try {
        const backupEp = `${CALCULATOR_RENDER_BACKUP}?url=${encodeURIComponent(legacyUrl)}`;
        const backupRes = await fetch(backupEp, {
          method: "GET",
          headers: {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            "Accept": "application/json"
          }
        });
        if (backupRes.ok) {
          const rawText = await backupRes.text();
          return new Response(rawText, { headers: corsHeaders, status: 200 });
        }
      } catch (e) {}

      return new Response(JSON.stringify({ success: false, error: "Server error or Invalid URL" }), { headers: corsHeaders, status: 500 });
    }

    // =========================================================================
    // 1️⃣3️⃣ 🌟 MASTER API DIRECTORY (Returned for root `/`, `/api`, `/endpoints`)
    // =========================================================================
    return new Response(JSON.stringify({

      timestamp: getISTTimestamp(),

      },
      endpoints: {

        },
        calculator_api: {

        },

        },

        },
        blog_by_slug: {

          method: "GET",

        },
        locations: {
