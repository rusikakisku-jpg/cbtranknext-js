/**
 * 🛡️ Helper: Strict SQL Injection Sanitization
 */
function sanitizeSql(str) {
  if (str === null || str === undefined) return "";
  return String(str)
    .replace(/'/g, "''")
    .replace(/\\/g, "\\\\")
    .replace(/\0/g, "")
    .trim();
}

/**
 * 🗄️ Helper: Query Cloudflare D1 (Native env.DB First, REST API Fallback)
 */
async function queryD1(env, sql) {
  // 1. Native Cloudflare D1 Binding
  if (env && env.DB) {
    try {
      const res = await env.DB.prepare(sql).all();
      if (res && res.results) return res.results;
      if (res) return res;
    } catch (e) {
      console.error("D1 Native Query Error:", e);
      throw new Error(`D1 Query Error: ${e.message}`);
    }
  }

  // 2. Cloudflare D1 REST API Fallback
  const accountId = (env && env.CF_ACCOUNT_ID) || DEFAULT_CF_ACCOUNT_ID;
  const databaseId = (env && env.CF_DATABASE_ID) || DEFAULT_CF_DATABASE_ID;
  const apiToken = (env && env.CF_API_TOKEN) || DEFAULT_CF_API_TOKEN;

  try {
    const apiUrl = `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/query`;
    const response = await fetch(apiUrl, {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${apiToken}`,
        "Content-Type": "application/json"
      },
      body: JSON.stringify({ sql: sql })
    });
    if (response.ok) {
      const jsonRes = await response.json();
      if (jsonRes && jsonRes.result && jsonRes.result[0]) {
        if (jsonRes.result[0].results) return jsonRes.result[0].results;
        return jsonRes.result[0];
      }
    }
  } catch (e) {
    console.error("D1 REST Fallback Error:", e);
  }

  return [];
}

/**
 * 🚀 Main Cloudflare Worker Fetch Handler
 */
export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const pathname = url.pathname.toLowerCase();
    const method = request.method.toUpperCase();

    // 1. CORS Preflight
    if (method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders, status: 204 });
    }

    // 2. Check D1 Database Connection Status
    const isD1Bound = !!(env && env.DB);

    // 3. Extract Input Parameter `exam_id`
    let examId = url.searchParams.get("exam_id") || url.searchParams.get("id") || "";

    if (method === "POST") {
      try {
        const contentType = request.headers.get("content-type") || "";
        if (contentType.includes("application/json")) {
          const body = await request.json().catch(() => ({}));
          examId = body.exam_id || body.id || examId;
        } else if (contentType.includes("application/x-www-form-urlencoded") || contentType.includes("multipart/form-data")) {
          const form = await request.formData().catch(() => new Map());
          examId = form.get("exam_id") || form.get("id") || examId;
        }
      } catch (e) {}
    }

    examId = String(examId).trim();

    // 4. Render Full Website UI if visited directly without exam_id
    const wantsJson = url.searchParams.has("json") || 
                      (request.headers.get("accept") && request.headers.get("accept").includes("application/json"));

    if (!examId && !wantsJson && method === "GET") {
      return renderWebPortal(isD1Bound);
    }

    // 5. Validate exam_id
    if (!examId) {
      return new Response(JSON.stringify({
        success: false,
        error: "Missing exam_id parameter. Please provide exam_id via ?exam_id=... or POST JSON body."
      }, null, 2), {
        status: 400,
        headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" }
      });
    }

    const cleanExamId = sanitizeSql(examId);

    try {
      // 6. 🎯 STRICT EXACT MATCH QUERY ON `user_ranks`
      // Selects shift dates, times, paper_language, and answer key url column
      const querySql = `
        SELECT 
          TRIM(exam_date) AS exam_date, 
          TRIM(exam_time) AS exam_time, 
          TRIM(paper_language) AS paper_language,
          TRIM(url) AS url,
          TRIM(user_id) AS user_id
        FROM user_ranks 
        WHERE exam_id = '${cleanExamId}' 
          AND exam_date IS NOT NULL 
          AND TRIM(exam_date) != '' 
          AND LOWER(exam_date) != 'null'
          AND exam_time IS NOT NULL 
          AND TRIM(exam_time) != ''
          AND LOWER(exam_time) != 'null'
        ORDER BY exam_date ASC, exam_time ASC, paper_language ASC;
      `;

      const rows = await queryD1(env, querySql);

      // 7. Group Shifts + Collect Candidate URLs & Shift-wise Languages
      const shiftMap = new Map();
      const allLanguagesSet = new Set();
      const dateSet = new Set();
      const candidateList = [];
      const seenCandidateUrls = new Set();

      (rows || []).forEach(row => {
        const d = (row.exam_date || "").trim();
        const t = (row.exam_time || "").trim();
        const lang = (row.paper_language || "").trim() || "Unknown";
        const candidateUrl = (row.url || "").trim();
        const userId = (row.user_id || "").trim();

        if (!d || !t || d.toLowerCase() === "null" || t.toLowerCase() === "null") return;

        const shiftKey = `${d}___${t}`;
        if (!shiftMap.has(shiftKey)) {
          shiftMap.set(shiftKey, {
            exam_date: d,
            exam_time: t,
            languages: [],
            first_url_by_lang: {}
          });
          dateSet.add(d);
        }

        const shiftObj = shiftMap.get(shiftKey);

        // Language tracking
        if (lang && lang.toLowerCase() !== "null") {
          const lowerLang = lang.toLowerCase();
          if (!shiftObj.languages.some(l => l.toLowerCase() === lowerLang)) {
            shiftObj.languages.push(lang);
          }
          if (!shiftObj.first_url_by_lang[lang] && candidateUrl) {
            shiftObj.first_url_by_lang[lang] = candidateUrl;
          }
          if (!Array.from(allLanguagesSet).some(l => l.toLowerCase() === lowerLang)) {
            allLanguagesSet.add(lang);
          }
        }

        // Candidate Answer Key URL tracking
        if (candidateUrl && candidateUrl.toLowerCase() !== "null" && !seenCandidateUrls.has(candidateUrl)) {
          seenCandidateUrls.add(candidateUrl);
          candidateList.push({
            serial_no: candidateList.length + 1,
            exam_date: d,
            exam_time: t,
            paper_language: lang,
            user_id: userId,
            url: candidateUrl
          });
        }
      });

      // Sort shifts & languages
      const shifts = Array.from(shiftMap.values()).map(s => {
        s.languages.sort((a, b) => a.localeCompare(b));
        return s;
      });

      const sortedDates = Array.from(dateSet).sort();
      const allLanguages = Array.from(allLanguagesSet).sort((a, b) => a.localeCompare(b));

      // Grouped by date
      const groupedByDate = {};
      shifts.forEach(s => {
        if (!groupedByDate[s.exam_date]) groupedByDate[s.exam_date] = [];
        groupedByDate[s.exam_date].push({
          exam_time: s.exam_time,
          languages: s.languages,
          first_url_by_lang: s.first_url_by_lang
        });
      });

      // 8. Prepare Clean Response Data
      const responseData = {
        success: true,
        exam_id: examId,
        match_type: "STRICT_EXACT_MATCH",
        table: "user_ranks",
        total_shifts: shifts.length,
        total_dates: sortedDates.length,
        total_languages: allLanguages.length,
        total_candidate_urls: candidateList.length,
        all_languages: allLanguages,
        dates: sortedDates,
        shifts: shifts,
        candidate_urls: candidateList,
        d1_database_bound: isD1Bound,
        note: shifts.length === 0 ? (!isD1Bound ? "⚠️ env.DB binding is missing in Cloudflare Dashboard! Please add D1 Database binding named 'DB'." : `No candidate records found in user_ranks where exam_id exactly equals '${examId}'.`) : undefined
      };

      // If user opened directly in browser without asking for json, show Full Website UI!
      if (!wantsJson && method === "GET") {
        return renderWebPortal(isD1Bound, responseData);
      }

      return new Response(JSON.stringify(responseData, null, 2), {
        status: 200,
        headers: {
          ...corsHeaders,
          "Content-Type": "application/json; charset=utf-8",
          "Cache-Control": "public, max-age=60, s-maxage=300"
        }
      });

    } catch (err) {
      console.error("Worker Execution Error:", err);
      return new Response(JSON.stringify({
        success: false,
        error: "Internal Server Error querying user_ranks",
        message: err.message || String(err),
        d1_bound: isD1Bound
      }, null, 2), {
        status: 500,
        headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8" }
      });
    }
  }
};

  <script>
    let currentData = ${initialDataJson};
    let selectedLanguages = new Set(); // Multi-select state

    // Initialize with preloaded data if present
    if (currentData && currentData.success) {
      document.getElementById('examIdInput').value = currentData.exam_id || '';
      renderAll(currentData);
    }

    async function fetchData() {
      const input = document.getElementById('examIdInput');
      const examId = input ? input.value.trim() : '';
      if (!examId) {
        alert('Please enter an Exam ID!');
        return;
      }

      const searchBtn = document.getElementById('searchBtn');
      const loader = document.getElementById('loader');
      const resultsSection = document.getElementById('resultsSection');

      if (searchBtn) { searchBtn.disabled = true; searchBtn.innerText = 'Searching...'; }
      if (loader) loader.style.display = 'block';
      if (resultsSection) resultsSection.style.display = 'none';

      try {
        const fetchUrl = '?exam_id=' + encodeURIComponent(examId) + '&json=1';
        const res = await fetch(fetchUrl);
        const data = await res.json();

        if (loader) loader.style.display = 'none';
        if (searchBtn) { searchBtn.disabled = false; searchBtn.innerText = 'Search Exam'; }

        if (!data.success) {
          alert('Error: ' + (data.error || 'Failed to fetch data'));
          return;
        }

        currentData = data;
        selectedLanguages.clear(); // Reset language filter on new search
        renderAll(data);
      } catch (err) {
        if (loader) loader.style.display = 'none';
        if (searchBtn) { searchBtn.disabled = false; searchBtn.innerText = 'Search Exam'; }
        alert('Fetch Error: ' + err.message);
      }
    }

    function renderAll(data) {
      const resultsSection = document.getElementById('resultsSection');
      if (resultsSection) resultsSection.style.display = 'block';

      // Match Notice
      const noticeElem = document.getElementById('matchNotice');
      if (noticeElem) {
        noticeElem.innerHTML = 
          '<div><b>Searched Exam ID:</b> <code>' + escapeHtml(data.exam_id) + '</code> &nbsp;|&nbsp; <b>Match Type:</b> Strict Exact Match (=)</div>' +
          (data.note ? '<div style="color:#f87171; font-weight:700;">' + escapeHtml(data.note) + '</div>' : '');
      }

      // KPI stats
      document.getElementById('statShifts').innerText = data.total_shifts || 0;
      document.getElementById('statDates').innerText = data.total_dates || 0;
      document.getElementById('statLangs').innerText = data.total_languages || 0;
      document.getElementById('statUrls').innerText = (data.candidate_urls ? data.candidate_urls.length : 0);

      // Render Section 1: Shifts Table
      const shiftsTbody = document.getElementById('shiftsBody');
      if (shiftsTbody) {
        if (data.shifts && data.shifts.length > 0) {
          shiftsTbody.innerHTML = data.shifts.map((s, idx) => {
            const langBadges = (s.languages && s.languages.length > 0)
              ? s.languages.map(l => {
                  const targetUrl = (s.first_url_by_lang && s.first_url_by_lang[l]) || '';
                  return '<button type="button" class="shift-lang-badge" data-lang="' + escapeHtml(l) + '" data-url="' + escapeHtml(targetUrl) + '" title="Click to open link in other tab">' +
                    escapeHtml(l) + ' <span class="icon">&#8599;</span></button>';
                }).join('')
              : '<span style="color:#64748b; font-size:13px;">None</span>';

            return '<tr>' +
              '<td><span class="serial-tag">' + (idx + 1) + '</span></td>' +
              '<td><b style="color:#fff;">' + escapeHtml(s.exam_date) + '</b></td>' +
              '<td>' + escapeHtml(s.exam_time) + '</td>' +
              '<td>' + langBadges + '</td>' +
            '</tr>';
          }).join('');
        } else {
          shiftsTbody.innerHTML = '<tr><td colspan="4" class="empty-state">No shifts found matching exact exam_id: <b>' + escapeHtml(data.exam_id) + '</b></td></tr>';
        }
      }

      // Render Language Filter Pills
      renderLangPills(data.all_languages || []);

      // Render Section 2: Serial-wise Candidate URLs Table
      renderSerialUrlsTable();
    }

    function renderLangPills(languages) {
      const container = document.getElementById('langPillsContainer');
      if (!container) return;

      if (!languages || languages.length === 0) {
        container.innerHTML = '<span style="color:#64748b; font-size:12px;">No languages</span>';
        return;
      }

      let html = '<div class="lang-pill ' + (selectedLanguages.size === 0 ? 'active' : '') + '" data-lang-all="1">All Languages</div>';
      languages.forEach(l => {
        const isSelected = selectedLanguages.has(l.toLowerCase());
        html += '<div class="lang-pill ' + (isSelected ? 'active' : '') + '" data-lang-toggle="' + escapeHtml(l.toLowerCase()) + '">' +
          (isSelected ? '✓ ' : '') + escapeHtml(l) +
        '</div>';
      });
      container.innerHTML = html;
    }

    function updatePillsUI() {
      const pills = document.querySelectorAll('#langPillsContainer .lang-pill');
      if (pills.length === 0) return;

      // First pill is 'All'
      if (selectedLanguages.size === 0) {
        pills[0].classList.add('active');
      } else {
        pills[0].classList.remove('active');
      }

      const allLangs = currentData?.all_languages || [];
      allLangs.forEach((l, idx) => {
        const pillElem = pills[idx + 1];
        if (!pillElem) return;
        if (selectedLanguages.has(l.toLowerCase())) {
          pillElem.classList.add('active');
          pillElem.textContent = '✓ ' + l;
        } else {
          pillElem.classList.remove('active');
          pillElem.textContent = l;
        }
      });
    }

    // Render Candidate URLs Table Serial-wise (#1, #2, #3...)
    function renderSerialUrlsTable() {
      const tbody = document.getElementById('candidateUrlsBody');
      if (!tbody) return;

      const allUrls = currentData?.candidate_urls || [];

      // Filter by selected languages
      const filtered = (selectedLanguages.size === 0)
        ? allUrls
        : allUrls.filter(item => selectedLanguages.has((item.paper_language || '').toLowerCase()));

      const countBadge = document.getElementById('urlCountBadge');
      if (countBadge) countBadge.innerText = filtered.length + ' Links Showing';

      if (filtered.length === 0) {
        tbody.innerHTML = '<tr><td colspan="7" class="empty-state">No candidate answer key URLs found for the selected filter.</td></tr>';
        return;
      }

      tbody.innerHTML = filtered.map((item, idx) => {
        const serialNo = idx + 1; // Serial-wise #1, #2, #3...
        return '<tr>' +
          '<td><span class="serial-tag">#' + serialNo + '</span></td>' +
          '<td><b style="color:#fff;">' + escapeHtml(item.exam_date) + '</b></td>' +
          '<td>' + escapeHtml(item.exam_time) + '</td>' +
          '<td><span style="background:#1e293b; color:#38bdf8; padding:3px 8px; border-radius:6px; font-weight:700; font-size:12px;">' + escapeHtml(item.paper_language || 'N/A') + '</span></td>' +
          '<td><span style="color:#94a3b8; font-family:monospace;">' + escapeHtml(item.user_id || 'N/A') + '</span></td>' +
          '<td><div class="url-cell" title="' + escapeHtml(item.url) + '">' + escapeHtml(item.url) + '</div></td>' +
          '<td style="text-align:center;">' +
            '<a href="' + escapeHtml(item.url) + '" target="_blank" rel="noopener noreferrer" class="btn-open-url">Open Link &#8599;</a>' +
            '<button type="button" class="btn-copy" data-url="' + escapeHtml(item.url) + '">Copy</button>' +
          '</td>' +
        '</tr>';
      }).join('');
    }

    // Single Global Event Listener for Clicks
    document.addEventListener('click', function(e) {
      // 1. Click on Shift Language Badge -> Open in other tab + filter
      const badge = e.target.closest('.shift-lang-badge');
      if (badge) {
        const directUrl = badge.getAttribute('data-url');
        const lang = badge.getAttribute('data-lang');
        if (directUrl) {
          window.open(directUrl, '_blank');
        }
        if (lang) {
          selectedLanguages.clear();
          selectedLanguages.add(lang.toLowerCase());
          updatePillsUI();
          renderSerialUrlsTable();
        }
        return;
      }

      // 2. Click on 'All Languages' pill
      const allPill = e.target.closest('[data-lang-all]');
      if (allPill) {
        selectedLanguages.clear();
        updatePillsUI();
        renderSerialUrlsTable();
        return;
      }

      // 3. Click on individual Language pill (Multi-Select toggle)
      const togglePill = e.target.closest('[data-lang-toggle]');
      if (togglePill) {
        const langLower = togglePill.getAttribute('data-lang-toggle');
        if (selectedLanguages.has(langLower)) {
          selectedLanguages.delete(langLower);
        } else {
          selectedLanguages.add(langLower);
        }
        updatePillsUI();
        renderSerialUrlsTable();
        return;
      }

      // 4. Click on Copy Button
      const copyBtn = e.target.closest('.btn-copy');
      if (copyBtn) {
        const copyUrlVal = copyBtn.getAttribute('data-url');
        if (copyUrlVal && navigator.clipboard) {
          navigator.clipboard.writeText(copyUrlVal).then(() => {
            const oldText = copyBtn.innerText;
            copyBtn.innerText = 'Copied! ✅';
            copyBtn.style.color = '#38bdf8';
            setTimeout(() => {
              copyBtn.innerText = oldText;
              copyBtn.style.color = '';
            }, 1500);
          });
        }
        return;
      }
    });

    function escapeHtml(str) {
      if (!str) return '';
      return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
    }
  </script>
