// Helper: Safe Cloudflare D1 Database Instance Extractor
// =============================================================================
function getD1Database(env) {
  if (!env) return null;
  if (env.DB && typeof env.DB.prepare === 'function') return env.DB;
  if (env.DATABASE && typeof env.DATABASE.prepare === 'function') return env.DATABASE;
  if (env.d1 && typeof env.d1.prepare === 'function') return env.d1;
  for (const key of Object.keys(env)) {
    const val = env[key];
    if (val && typeof val === 'object' && typeof val.prepare === 'function') {
      return val;
    }
  }
  return null;
}

function getR2Bucket(env) {
  if (!env) return null;
  if (env.R2 && typeof env.R2.put === 'function') return env.R2;
  if (env.BUCKET && typeof env.BUCKET.put === 'function') return env.BUCKET;
  if (env.UPLOAD_BUCKET && typeof env.UPLOAD_BUCKET.put === 'function') return env.UPLOAD_BUCKET;
  if (env.IMAGES_BUCKET && typeof env.IMAGES_BUCKET.put === 'function') return env.IMAGES_BUCKET;
  if (env.IMAGES && typeof env.IMAGES.put === 'function') return env.IMAGES;
  for (const key of Object.keys(env)) {
    const val = env[key];
    if (val && typeof val === 'object' && typeof val.put === 'function' && typeof val.head === 'function') {
      return val;
    }
  }
  return null;
}

const DEFAULT_R2_PUBLIC_DOMAIN = "https://upload.cbtrank.com";

function getR2PublicDomain(env) {
  const domain = (env && env.R2_PUBLIC_DOMAIN) ? env.R2_PUBLIC_DOMAIN : DEFAULT_R2_PUBLIC_DOMAIN;
  return domain.replace(/\/+$/, '');
}

// Helper: D1 Query Executor (Handles Parameterized Bindings & Mutations)
// =============================================================================
async function executeD1(env, sql, params = []) {
  const db = getD1Database(env);
  if (!db) {
    throw new Error("Cloudflare D1 database binding 'DB' is missing. Please bind your D1 database under Worker Settings -> Bindings -> D1 Database Bindings with variable name DB.");
  }

  const cleanParams = params.map(p => {
    if (p === undefined || p === null) return null;
    if (typeof p === 'boolean') return p ? 1 : 0;
    return p;
  });

  try {
    const stmt = db.prepare(sql);
    const bound = cleanParams.length > 0 ? stmt.bind(...cleanParams) : stmt;
    const isSelectOrPragma = /^\s*(SELECT|PRAGMA)/i.test(sql);

    if (isSelectOrPragma) {
      const res = await bound.all();
      return (res && res.results) ? res.results : (res || []);
    } else {
      const res = await bound.run();
      return res;
    }
  } catch (err) {
    console.error("D1 Execution Error:", err.message, "\nSQL:", sql, "\nParams:", params);
    throw err;
  }
}

// Helper: Inspect Existing Columns of 'blogs' Table via PRAGMA
// =============================================================================
async function getTableColumns(env) {
  try {
    const info = await executeD1(env, "PRAGMA table_info(blogs)");
    if (Array.isArray(info) && info.length > 0) {
      const names = info.map(c => {
        if (!c) return '';
        if (typeof c === 'string') return c;
        return c.name || c.NAME || c.Name || (Object.values(c)[1]) || '';
      }).filter(Boolean);
      if (names.length > 0) return names;
    }
  } catch (e) {
    // Table may not exist yet
  }
  return [
    'id', 'title', 'slug', 'excerpt', 'category', 'author', 'author_name',
    'cover_image', 'image', 'content', 'description', 'status', 'views',
    'published_at', 'publish_date', 'created_at', 'updated_at'
  ];
}

// Case-insensitive column resolver with alias fallback
function resolveColumnName(availableCols, primaryName, aliases = []) {
  if (!Array.isArray(availableCols) || availableCols.length === 0) return primaryName;
  const targetNames = [primaryName, ...aliases].map(s => s.toLowerCase());
  for (const target of targetNames) {
    for (const col of availableCols) {
      if (String(col).toLowerCase() === target) {
        return col; // Returns exact casing from database schema
      }
    }
  }
  return primaryName;
}

function hasColumn(availableCols, columnName, aliases = []) {
  if (!Array.isArray(availableCols) || availableCols.length === 0) return false;
  const targetNames = [columnName, ...aliases].map(s => s.toLowerCase());
  return availableCols.some(c => targetNames.includes(String(c).toLowerCase()));
}

// Helper: Normalize Cover Image URL to full public URL
function formatBlogCoverUrl(rawImage, domain = DEFAULT_R2_PUBLIC_DOMAIN) {
  if (!rawImage || typeof rawImage !== 'string') return '';
  const trimmed = rawImage.trim();
  if (!trimmed) return '';
  if (trimmed.startsWith('http://') || trimmed.startsWith('https://') || trimmed.startsWith('data:')) {
    return trimmed;
  }
  const cleanPath = trimmed.replace(/^\/+/, '');
  const cleanDomain = (domain || DEFAULT_R2_PUBLIC_DOMAIN).replace(/\/+$/, '');
  return `${cleanDomain}/${cleanPath}`;
}

// Case-insensitive Blog Row Normalizer with R2 Cover Image Resolution
function normalizeBlogRow(row, r2Domain = DEFAULT_R2_PUBLIC_DOMAIN) {
  if (!row || typeof row !== 'object') return row;
  const item = { ...row };

  const getProp = (...keys) => {
    const rowKeys = Object.keys(row);
    for (const k of keys) {
      const target = k.toLowerCase();
      for (const rk of rowKeys) {
        if (rk.toLowerCase() === target && row[rk] !== undefined && row[rk] !== null) {
          return row[rk];
        }
      }
    }
    return undefined;
  };

  const idVal = getProp('id');
  item.id = (idVal !== undefined && idVal !== null) ? Number(idVal) : item.id;
  item.title = String(getProp('title') || item.title || '').trim();
  item.slug = String(getProp('slug') || item.slug || '').trim();
  item.excerpt = String(getProp('excerpt') || item.excerpt || '').trim();
  item.category = String(getProp('category') || item.category || 'Exam Analysis').trim();

  const authorVal = String(getProp('author', 'author_name') || item.author || item.author_name || 'Team CBTRANK').trim();
  item.author = authorVal;
  item.author_name = authorVal;

  const rawCover = String(getProp('cover_image', 'image', 'coverimage', 'thumbnail', 'cover', 'img') || item.cover_image || item.image || '').trim();
  const formattedCover = formatBlogCoverUrl(rawCover, r2Domain);
  item.cover_image = formattedCover;
  item.image = formattedCover;

  const contentVal = String(getProp('content', 'description', 'body') || item.content || item.description || '').trim();
  item.content = contentVal;
  item.description = contentVal;

  const statusVal = String(getProp('status') || item.status || 'publish').toLowerCase();
  item.status = (statusVal === 'draft') ? 'draft' : 'publish';

  const viewsVal = getProp('views');
  item.views = (viewsVal !== undefined && viewsVal !== null && !isNaN(Number(viewsVal))) ? Number(viewsVal) : 0;

  item.created_at = String(getProp('created_at') || item.created_at || '');
  item.updated_at = String(getProp('updated_at') || item.updated_at || '');

  const pubAtVal = String(getProp('published_at', 'publish_date', 'publishdate', 'pub_date') || item.published_at || item.publish_date || '').trim();
  if (pubAtVal) {
    item.published_at = pubAtVal;
    item.publish_date = pubAtVal.includes('T') ? pubAtVal.split('T')[0] : pubAtVal.split(' ')[0];
  } else if (item.status === 'publish') {
    const fallbackDate = item.created_at || new Date().toISOString();
    item.published_at = fallbackDate;
    item.publish_date = fallbackDate.includes('T') ? fallbackDate.split('T')[0] : fallbackDate.split(' ')[0];
  } else {
    item.published_at = '';
    item.publish_date = '';
  }

  return item;
}

// Helper: Auto-create & Self-Healing Migration for 'blogs' Table

async function ensureBlogsTable(env) {
  const db = getD1Database(env);
  if (!db) return [];

  const createSql = `
    CREATE TABLE IF NOT EXISTS blogs (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      slug TEXT NOT NULL UNIQUE,
      excerpt TEXT DEFAULT '',
      category TEXT DEFAULT 'Exam Analysis',
      author TEXT DEFAULT 'Team CBTRANK',
      author_name TEXT DEFAULT 'Team CBTRANK',
      cover_image TEXT DEFAULT '',
      image TEXT DEFAULT '',
      content TEXT DEFAULT '',
      description TEXT DEFAULT '',
      status TEXT DEFAULT 'publish',
      views INTEGER DEFAULT 0,
      published_at TEXT DEFAULT '',
      publish_date TEXT DEFAULT '',
      created_at TEXT DEFAULT CURRENT_TIMESTAMP,
      updated_at TEXT DEFAULT CURRENT_TIMESTAMP
    );
  `;
  try {
    await db.prepare(createSql).run();
  } catch (e) {
    console.warn("Table create notice:", e.message);
  }

  // Self-healing: verify all required columns exist, add if missing safely
  const requiredCols = [
    { name: 'excerpt', type: "TEXT DEFAULT ''" },
    { name: 'category', type: "TEXT DEFAULT 'Exam Analysis'" },
    { name: 'author', type: "TEXT DEFAULT 'Team CBTRANK'" },
    { name: 'author_name', type: "TEXT DEFAULT 'Team CBTRANK'" },
    { name: 'cover_image', type: "TEXT DEFAULT ''" },
    { name: 'image', type: "TEXT DEFAULT ''" },
    { name: 'content', type: "TEXT DEFAULT ''" },
    { name: 'description', type: "TEXT DEFAULT ''" },
    { name: 'status', type: "TEXT DEFAULT 'publish'" },
    { name: 'views', type: "INTEGER DEFAULT 0" },
    { name: 'published_at', type: "TEXT DEFAULT ''" },
    { name: 'publish_date', type: "TEXT DEFAULT ''" },
    { name: 'created_at', type: "TEXT DEFAULT ''" },
    { name: 'updated_at', type: "TEXT DEFAULT ''" }
  ];

  for (const col of requiredCols) {
    try {
      await db.prepare(`ALTER TABLE blogs ADD COLUMN ${col.name} ${col.type}`).run();
    } catch (e) {
      // Column already exists or table locked, safely ignored
    }
  }

  return await getTableColumns(env);
}

// =============================================================================
// Helper: Seed Default 5 CBT Rank Blog Articles into D1 Table
// =============================================================================
async function seedDefaultBlogs(env) {
  const db = getD1Database(env);
  if (!db) return 0;
  const availableCols = await ensureBlogsTable(env);

  let insertedCount = 0;
  let maxId = 0;

  for (const b of DEFAULT_BLOGS) {
    try {
      const nowIso = new Date().toISOString();
      const candidateFields = [
        { key: 'title', val: b.title, aliases: [] },
        { key: 'slug', val: b.slug, aliases: [] },
        { key: 'excerpt', val: b.excerpt || '', aliases: [] },
        { key: 'category', val: b.category || 'Exam Analysis', aliases: [] },
        { key: 'author', val: b.author || 'Team CBTRANK', aliases: ['author_name'] },
        { key: 'author_name', val: b.author_name || b.author || 'Team CBTRANK', aliases: ['author'] },
        { key: 'cover_image', val: b.cover_image || '', aliases: ['image', 'coverImage', 'thumbnail'] },
        { key: 'image', val: b.image || b.cover_image || '', aliases: ['cover_image', 'coverImage', 'thumbnail'] },
        { key: 'content', val: b.content || '', aliases: ['description'] },
        { key: 'description', val: b.description || b.content || '', aliases: ['content'] },
        { key: 'status', val: b.status || 'publish', aliases: [] },
        { key: 'views', val: b.views || 0, aliases: [] },
        { key: 'published_at', val: b.published_at || b.created_at || nowIso, aliases: ['publish_date'] },
        { key: 'publish_date', val: b.publish_date || (b.published_at ? b.published_at.split(' ')[0] : nowIso.split('T')[0]), aliases: ['published_at'] }
      ];

      let colsToInsert = ['id'];
      let placeholders = ['?'];
      let valuesToInsert = [];
      valuesToInsert.push(b.id);
      const handledCols = new Set();

      for (const field of candidateFields) {
        if (hasColumn(availableCols, field.key, field.aliases)) {
          const actualCol = resolveColumnName(availableCols, field.key, field.aliases);
          const lower = actualCol.toLowerCase();
          if (!handledCols.has(lower)) {
            handledCols.add(lower);
            colsToInsert.push(actualCol);
            placeholders.push('?');
            valuesToInsert.push(field.val);
          }
        }
      }

      if (hasColumn(availableCols, 'created_at')) {
        const actualCol = resolveColumnName(availableCols, 'created_at');
        if (!handledCols.has(actualCol.toLowerCase())) {
          handledCols.add(actualCol.toLowerCase());
          colsToInsert.push(actualCol);
          placeholders.push('?');
          valuesToInsert.push(b.created_at || nowIso);
        }
      }
      if (hasColumn(availableCols, 'updated_at')) {
        const actualCol = resolveColumnName(availableCols, 'updated_at');
        if (!handledCols.has(actualCol.toLowerCase())) {
          handledCols.add(actualCol.toLowerCase());
          colsToInsert.push(actualCol);
          placeholders.push('?');
          valuesToInsert.push(b.updated_at || nowIso);
        }
      }

      const insertSql = `INSERT OR IGNORE INTO blogs (${colsToInsert.join(', ')}) VALUES (${placeholders.join(', ')})`;
      await executeD1(env, insertSql, valuesToInsert);
      insertedCount++;
      if (b.id > maxId) maxId = b.id;
    } catch (err) {
      console.warn("Seed row notice:", b.slug, err.message);
    }
  }

  if (maxId > 0) {
    try {
      await db.prepare("INSERT OR REPLACE INTO sqlite_sequence (name, seq) VALUES ('blogs', MAX(COALESCE((SELECT seq FROM sqlite_sequence WHERE name = 'blogs'), 0), ?))").bind(maxId).run();
    } catch (e) {}
  }

  return insertedCount;
}

// =============================================================================
// Helper: CORS & JSON Responses
// =============================================================================
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type, Authorization, x-api-key, x-filename, x-overwrite',
  'Access-Control-Max-Age': '86400'
};

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...corsHeaders,
      'Content-Type': 'application/json; charset=utf-8'
    }
  });
}

function errorResponse(message, status = 400, extra = {}) {
  return jsonResponse({ success: false, error: message, ...extra }, status);
}

// =============================================================================
// Main Worker Router & Handler
// =============================================================================
export default {
  async fetch(request, env, ctx) {
    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders, status: 204 });
    }

    const url = new URL(request.url);
    let pathname = url.pathname.replace(/\/+/g, '/');
    if (pathname.length > 1 && pathname.endsWith('/')) {
      pathname = pathname.slice(0, -1);
    }

    const dbInstance = getD1Database(env);
    const isD1Bound = !!dbInstance;
    const r2Bucket = getR2Bucket(env);
    const isR2Bound = !!r2Bucket;

    // -------------------------------------------------------------------------
    // ROUTE 0-A: Cloudflare R2 Image Asset Serving (GET / HEAD)
    // -------------------------------------------------------------------------
    const isImageExt = /\.(png|jpe?g|webp|gif|svg|ico|bmp|avif)$/i.test(pathname);
    const isR2PathPrefix = /^\/(?:cdn|uploads?|images?|r2|api\/images?|api\/cdn|api\/uploads?)\//i.test(pathname);
    const isUploadHost = url.hostname.toLowerCase().startsWith('upload.') || url.hostname.toLowerCase().includes('upload.cbtrank.com');

    // Health / info check on upload CDN root
    if (isUploadHost && (pathname === '/' || pathname === '')) {
      return jsonResponse({
        service: "CBT RANK - Cloudflare R2 Image CDN",
        status: "online",
        isR2Bound: isR2Bound,
        domain: getR2PublicDomain(env)
      });
    }

    const shouldHandleR2 = (request.method === 'GET' || request.method === 'HEAD') && (
      isR2PathPrefix ||
      isImageExt ||
      (isUploadHost && !pathname.startsWith('/api/'))
    );

    if (shouldHandleR2) {
      if (!isR2Bound) {
        return new Response("Cloudflare R2 Bucket is not bound to this Worker.", {
          status: 503,
          headers: { ...corsHeaders, 'Content-Type': 'text/plain; charset=utf-8' }
        });
      }

      let r2Key = pathname;
      r2Key = r2Key.replace(/^\/(?:cdn|uploads?|images?|r2|api\/images?|api\/cdn|api\/uploads?)\//i, '');
      r2Key = r2Key.replace(/^\/+/, '');
      try {
        r2Key = decodeURIComponent(r2Key);
      } catch (e) {}

      if (!r2Key) {
        return new Response("Missing image filename", {
          status: 400,
          headers: { ...corsHeaders, 'Content-Type': 'text/plain; charset=utf-8' }
        });
      }

      try {
        const object = await r2Bucket.get(r2Key);
        if (!object) {
          return new Response(`Image not found: '${r2Key}'`, {
            status: 404,
            headers: { ...corsHeaders, 'Content-Type': 'text/plain; charset=utf-8' }
          });
        }

        let contentType = object.httpMetadata?.contentType;
        if (!contentType || contentType === 'application/octet-stream') {
          const dotIdx = r2Key.lastIndexOf('.');
          const ext = dotIdx !== -1 ? r2Key.slice(dotIdx + 1).toLowerCase() : '';
          const mimeMap = {
            png: 'image/png',
            jpg: 'image/jpeg',
            jpeg: 'image/jpeg',
            webp: 'image/webp',
            gif: 'image/gif',
            svg: 'image/svg+xml',
            ico: 'image/x-icon',
            avif: 'image/avif',
            bmp: 'image/bmp'
          };
          contentType = mimeMap[ext] || 'image/png';
        }

        const headers = new Headers();
        Object.keys(corsHeaders).forEach(k => headers.set(k, corsHeaders[k]));
        headers.set('Content-Type', contentType);
        headers.set('Cache-Control', 'public, max-age=31536000, immutable');
        if (object.httpEtag) headers.set('ETag', object.httpEtag);
        if (object.size !== undefined) headers.set('Content-Length', String(object.size));

        const ifNoneMatch = request.headers.get('if-none-match');
        if (ifNoneMatch && object.httpEtag && ifNoneMatch === object.httpEtag) {
          return new Response(null, { status: 304, headers });
        }

        if (request.method === 'HEAD') {
          return new Response(null, { status: 200, headers });
        }

        return new Response(object.body, { status: 200, headers });
      } catch (err) {
        return new Response("Failed to fetch image from R2: " + err.message, {
          status: 500,
          headers: { ...corsHeaders, 'Content-Type': 'text/plain; charset=utf-8' }
        });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 0: Image Upload to Cloudflare R2 (/api/upload or /upload)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/upload' || pathname === '/upload') && request.method === 'POST') {
      if (!isR2Bound) {
        return errorResponse(
          "Cloudflare R2 Bucket is not bound to this Worker. Please add an R2 Bucket binding in Cloudflare Dashboard: Settings > Bindings > Add R2 Bucket binding with variable name 'R2' or 'BUCKET'.",
          500,
          { isR2Bound: false }
        );
      }

      try {
        const contentType = request.headers.get('content-type') || '';
        let fileData = null;
        let originalName = '';
        let mimeType = 'image/png';

        if (contentType.includes('multipart/form-data')) {
          const formData = await request.formData();
          const file = formData.get('file') || formData.get('image') || formData.get('upload');
          if (!file || typeof file === 'string') {
            return errorResponse("No file uploaded in form field 'file' or 'image'.", 400);
          }
          fileData = await file.arrayBuffer();
          originalName = file.name || 'image.png';
          mimeType = file.type || 'image/png';
        } else if (contentType.includes('application/json')) {
          const body = await request.json();
          originalName = body.filename || body.name || 'image.png';
          mimeType = body.type || 'image/png';
          if (body.data) {
            const match = body.data.match(/^data:([^;]+);base64,(.+)$/);
            if (match) {
              mimeType = match[1];
              const binary = atob(match[2]);
              const bytes = new Uint8Array(binary.length);
              for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
              fileData = bytes.buffer;
            } else {
              return errorResponse("Invalid base64 image data payload.", 400);
            }
          } else {
            return errorResponse("Missing 'data' in JSON upload payload.", 400);
          }
        } else if (contentType.startsWith('image/')) {
          fileData = await request.arrayBuffer();
          originalName = request.headers.get('x-filename') || ('image-' + Date.now() + '.png');
          mimeType = contentType;
        } else {
          return errorResponse("Unsupported Content-Type. Please upload via multipart/form-data or application/json.", 400);
        }

        if (!fileData || fileData.byteLength === 0) {
          return errorResponse("Uploaded image is empty (0 bytes).", 400);
        }

        // Sanitize filename to clean, SEO-friendly slug
        let ext = '';
        const dotIdx = originalName.lastIndexOf('.');
        if (dotIdx !== -1) {
          ext = originalName.slice(dotIdx).toLowerCase();
        } else {
          if (mimeType === 'image/jpeg') ext = '.jpg';
          else if (mimeType === 'image/webp') ext = '.webp';
          else if (mimeType === 'image/gif') ext = '.gif';
          else if (mimeType === 'image/svg+xml') ext = '.svg';
          else ext = '.png';
        }

        const basePart = dotIdx !== -1 ? originalName.slice(0, dotIdx) : originalName;
        let cleanSlug = basePart
          .toLowerCase()
          .replace(/[^a-z0-9]+/g, '-')
          .replace(/^-+|-+$/g, '');

        if (!cleanSlug) cleanSlug = 'image-' + Date.now();

        let finalKey = cleanSlug + ext;

        // Check if key exists; if so, append random hash to avoid overwriting unless requested
        const allowOverwrite = request.headers.get('x-overwrite') === '1' || url.searchParams.get('overwrite') === '1';
        if (!allowOverwrite) {
          try {
            const existing = await r2Bucket.head(finalKey);
            if (existing) {
              finalKey = cleanSlug + '-' + Math.random().toString(36).substring(2, 7) + ext;
            }
          } catch (e) {}
        }

        await r2Bucket.put(finalKey, fileData, {
          httpMetadata: {
            contentType: mimeType,
            cacheControl: 'public, max-age=31536000, immutable'
          }
        });

        const publicBase = getR2PublicDomain(env);
        const fullUrl = `${publicBase}/${finalKey}`;

        return jsonResponse({
          success: true,
          url: fullUrl,
          location: fullUrl,
          filename: finalKey,
          size: fileData.byteLength,
          contentType: mimeType,
          message: "Image successfully uploaded to Cloudflare R2!"
        }, 201);
      } catch (err) {
        return errorResponse("Failed to upload image to R2: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 0-B: Seed / Verify Default CBT Rank Blog Articles (/api/seed-blogs)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/seed-blogs' || pathname === '/api/blogs/seed') && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          seeded: DEFAULT_BLOGS.length,
          message: "Preview Mode: 5 default CBT Rank articles loaded.",
          isD1Bound: false
        });
      }
      try {
        const count = await seedDefaultBlogs(env);
        return jsonResponse({
          success: true,
          seeded: count,
          message: `Successfully seeded/verified ${count} default CBT Rank articles in D1 database!`,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Seeding failed: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 1: Schema Diagnostics & Setup
    // -------------------------------------------------------------------------
    if (pathname === '/api/schema' || pathname === '/api/diagnostics') {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500, { isD1Bound: false });
      try {
        const columns = await getTableColumns(env);
        const countRes = columns.length > 0 ? await executeD1(env, "SELECT COUNT(*) as total FROM blogs") : [{ total: 0 }];
        return jsonResponse({
          success: true,
          isD1Bound: true,
          isR2Bound: isR2Bound,
          r2Domain: getR2PublicDomain(env),
          tableExists: columns.length > 0,
          totalBlogs: countRes[0]?.total ?? 0,
          columns: columns
        });
      } catch (err) {
        return errorResponse(err.message, 500, { isD1Bound: true });
      }
    }

    if (pathname === '/api/setup-table' && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500, { isD1Bound: false });
      try {
        await ensureBlogsTable(env);
        const columns = await getTableColumns(env);
        return jsonResponse({
          success: true,
          message: "Table 'blogs' verified / created successfully in D1 database.",
          columns: columns,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to setup table: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 2: Quick Status Toggle (status = 'publish' <-> 'draft')
    // -------------------------------------------------------------------------
    if (pathname === '/api/blogs/toggle' && request.method === 'POST') {
      const nowIso = new Date().toISOString();
      const todayDate = nowIso.split('T')[0];

      if (!isD1Bound) {
        try {
          const body = await request.json().catch(() => ({}));
          const id = parseInt(body.id, 10);
          const newStatus = body.status ? String(body.status).toLowerCase() : 'publish';
          return jsonResponse({
            success: true,
            message: "Preview Mode: Updated locally",
            id: id,
            newStatus: newStatus,
            published_at: newStatus === 'publish' ? nowIso : '',
            publish_date: newStatus === 'publish' ? todayDate : '',
            isD1Bound: false
          });
        } catch (e) {
          return jsonResponse({ success: true, message: "Preview Mode: Updated locally", isD1Bound: false });
        }
      }
      try {
        await ensureBlogsTable(env);
        const body = await request.json().catch(() => ({}));
        const id = parseInt(body.id, 10);
        if (isNaN(id)) return errorResponse("Invalid or missing Blog ID.", 400, { isD1Bound: true });

        const currentBlog = await executeD1(env, "SELECT * FROM blogs WHERE id = ?", [id]);
        if (!currentBlog || currentBlog.length === 0) {
          return errorResponse("Blog post not found with ID: #" + id, 404, { isD1Bound: true });
        }

        const currentStatus = String(currentBlog[0].status || 'draft').toLowerCase();
        const newStatus = body.status
          ? String(body.status).toLowerCase()
          : (currentStatus === 'publish' || currentStatus === 'published' ? 'draft' : 'publish');

        const availableCols = await getTableColumns(env);
        let updateSql = "UPDATE blogs SET status = ?";
        let updateParams = [];
        updateParams.push(newStatus);

        if (newStatus === 'publish') {
          if (hasColumn(availableCols, 'published_at')) {
            const actualCol = resolveColumnName(availableCols, 'published_at');
            updateSql += `, ${actualCol} = ?`;
            updateParams.push(nowIso);
          }
          if (hasColumn(availableCols, 'publish_date')) {
            const actualCol = resolveColumnName(availableCols, 'publish_date');
            updateSql += `, ${actualCol} = ?`;
            updateParams.push(todayDate);
          }
        }
        if (hasColumn(availableCols, 'updated_at')) {
          const actualCol = resolveColumnName(availableCols, 'updated_at');
          updateSql += `, ${actualCol} = ?`;
          updateParams.push(nowIso);
        }
        updateSql += " WHERE id = ?";
        updateParams.push(id);

        try {
          await executeD1(env, updateSql, updateParams);
        } catch (updateErr) {
          await executeD1(env, "UPDATE blogs SET status = ? WHERE id = ?", [newStatus, id]);
        }

        return jsonResponse({
          success: true,
          message: `Blog #${id} status updated to '${newStatus}' in D1 database`,
          id: id,
          newStatus: newStatus,
          published_at: newStatus === 'publish' ? nowIso : '',
          publish_date: newStatus === 'publish' ? todayDate : '',
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Toggle failed: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 3: Add New Blog Post (POST /api/blogs or POST /api/blogs/add)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/blogs' || pathname === '/api/blogs/add') && request.method === 'POST') {
      if (!isD1Bound) {
        try {
          const body = await request.json();
          return jsonResponse({
            success: true,
            message: "Blog post added in Preview Mode (bind D1 in Cloudflare to persist).",
            id: Date.now(),
            isD1Bound: false
          });
        } catch (e) {
          return errorResponse("Invalid JSON payload", 400, { isD1Bound: false });
        }
      }
      try {
        await ensureBlogsTable(env);
        const body = await request.json().catch(() => ({}));
        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();

        if (!title) return errorResponse("Field 'title' is required.", 400, { isD1Bound: true });
        if (!slug) {
          slug = title
            .toLowerCase()
            .replace(/[^a-z0-9]+/g, '-')
            .replace(/^-+|-+$/g, '');
        }

        const existing = await executeD1(env, "SELECT id FROM blogs WHERE slug = ? LIMIT 1", [slug]);
        if (existing && existing.length > 0) {
          return errorResponse(`A blog article with slug '${slug}' already exists (ID: #${existing[0].id}). Please choose a unique slug.`, 400, { isD1Bound: true });
        }

        const excerpt = String(body.excerpt || '').trim();
        const category = String(body.category || 'Exam Analysis').trim();
        const author = String(body.author || body.author_name || 'Team CBTRANK').trim();
        const rawCoverImage = String(body.cover_image || body.image || '').trim();
        const coverImage = formatBlogCoverUrl(rawCoverImage, getR2PublicDomain(env));
        const content = String(body.content || body.description || '').trim();
        const status = String(body.status || 'publish').toLowerCase() === 'draft' ? 'draft' : 'publish';
        const views = parseInt(body.views, 10) || 0;

        const availableCols = await getTableColumns(env);
        const nowIso = new Date().toISOString();
        const todayDate = nowIso.split('T')[0];

        let publishedAt = String(body.published_at || body.publish_date || '').trim();
        if (status === 'publish') {
          if (!publishedAt) {
            publishedAt = nowIso;
          } else if (publishedAt.length === 10) {
            publishedAt = publishedAt + ' 00:00:00';
          }
        } else {
          if (!publishedAt) publishedAt = '';
        }
        const publishDate = publishedAt ? (publishedAt.includes('T') ? publishedAt.split('T')[0] : publishedAt.split(' ')[0]) : (status === 'publish' ? todayDate : '');

        // Determine next unique ID safely to satisfy NOT NULL constraint
        let nextId = 1;
        try {
          const maxRes = await executeD1(env, "SELECT COALESCE(MAX(id), 0) AS max_id FROM blogs");
          if (maxRes && maxRes.length > 0 && maxRes[0].max_id !== undefined && maxRes[0].max_id !== null) {
            nextId = parseInt(maxRes[0].max_id, 10) + 1;
            if (isNaN(nextId) || nextId < 1) nextId = 1;
          }
        } catch (e) {
          nextId = Date.now();
        }

        const candidateFields = [
          { key: 'title', val: title, aliases: [] },
          { key: 'slug', val: slug, aliases: [] },
          { key: 'excerpt', val: excerpt, aliases: [] },
          { key: 'category', val: category, aliases: [] },
          { key: 'author', val: author, aliases: ['author_name'] },
          { key: 'author_name', val: author, aliases: ['author'] },
          { key: 'cover_image', val: coverImage, aliases: ['image', 'coverImage', 'thumbnail'] },
          { key: 'image', val: coverImage, aliases: ['cover_image', 'coverImage', 'thumbnail'] },
          { key: 'content', val: content, aliases: ['description'] },
          { key: 'description', val: content, aliases: ['content'] },
          { key: 'status', val: status, aliases: [] },
          { key: 'views', val: views, aliases: [] },
          { key: 'published_at', val: publishedAt, aliases: ['publish_date'] },
          { key: 'publish_date', val: publishDate, aliases: ['published_at'] }
        ];

        let colsToInsert = ['id'];
        let placeholders = ['?'];
        let valuesToInsert = [];
        valuesToInsert.push(nextId);
        const handledCols = new Set();

        for (const field of candidateFields) {
          if (hasColumn(availableCols, field.key, field.aliases)) {
            const actualCol = resolveColumnName(availableCols, field.key, field.aliases);
            const lower = actualCol.toLowerCase();
            if (!handledCols.has(lower)) {
              handledCols.add(lower);
              colsToInsert.push(actualCol);
              placeholders.push('?');
              valuesToInsert.push(field.val);
            }
          }
        }

        if (hasColumn(availableCols, 'created_at')) {
          const actualCol = resolveColumnName(availableCols, 'created_at');
          if (!handledCols.has(actualCol.toLowerCase())) {
            handledCols.add(actualCol.toLowerCase());
            colsToInsert.push(actualCol);
            placeholders.push('?');
            valuesToInsert.push(nowIso);
          }
        }
        if (hasColumn(availableCols, 'updated_at')) {
          const actualCol = resolveColumnName(availableCols, 'updated_at');
          if (!handledCols.has(actualCol.toLowerCase())) {
            handledCols.add(actualCol.toLowerCase());
            colsToInsert.push(actualCol);
            placeholders.push('?');
            valuesToInsert.push(nowIso);
          }
        }

        const insertSql = `INSERT INTO blogs (${colsToInsert.join(', ')}) VALUES (${placeholders.join(', ')})`;
        await executeD1(env, insertSql, valuesToInsert);

        try {
          const db = getD1Database(env);
          if (db) {
            await db.prepare("INSERT OR REPLACE INTO sqlite_sequence (name, seq) VALUES ('blogs', MAX(COALESCE((SELECT seq FROM sqlite_sequence WHERE name = 'blogs'), 0), ?))").bind(nextId).run();
          }
        } catch (e) {}

        let newId = nextId;
        try {
          const insertedRow = await executeD1(env, "SELECT id FROM blogs WHERE slug = ? LIMIT 1", [slug]);
          if (insertedRow && insertedRow.length > 0) {
            newId = insertedRow[0].id;
          }
        } catch (e) {}

        return jsonResponse({
          success: true,
          message: "Blog article added successfully to D1 database!",
          id: newId,
          slug: slug,
          isD1Bound: true
        }, 201);
      } catch (err) {
        return errorResponse("Failed to add blog article: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 4: Update Blog Post (PUT /api/blogs/:id or POST /api/blogs/update)
    // -------------------------------------------------------------------------
    const updateMatch = pathname.match(/^\/api\/blogs\/(\d+)$/);
    if ((updateMatch && (request.method === 'PUT' || request.method === 'POST')) || (pathname === '/api/blogs/update' && request.method === 'POST')) {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          message: "Blog article updated in Preview Mode.",
          isD1Bound: false
        });
      }
      try {
        await ensureBlogsTable(env);
        const body = await request.json().catch(() => ({}));
        const blogId = updateMatch ? parseInt(updateMatch[1], 10) : parseInt(body.id, 10);
        if (isNaN(blogId)) return errorResponse("Invalid or missing Blog ID for update.", 400, { isD1Bound: true });

        const existing = await executeD1(env, "SELECT * FROM blogs WHERE id = ?", [blogId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No blog article found with ID: #${blogId}`, 404, { isD1Bound: true });
        }

        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();
        if (!title) return errorResponse("Title cannot be empty.", 400, { isD1Bound: true });
        if (!slug) {
          slug = title.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
        }

        const slugCheck = await executeD1(env, "SELECT id FROM blogs WHERE slug = ? AND id != ? LIMIT 1", [slug, blogId]);
        if (slugCheck && slugCheck.length > 0) {
          return errorResponse(`Slug '${slug}' is already in use by another article (ID: #${slugCheck[0].id}).`, 400, { isD1Bound: true });
        }

        const excerpt = String(body.excerpt || '').trim();
        const category = String(body.category || 'Exam Analysis').trim();
        const author = String(body.author || body.author_name || 'Team CBTRANK').trim();
        const rawCoverImage = String(body.cover_image || body.image || '').trim();
        const coverImage = formatBlogCoverUrl(rawCoverImage, getR2PublicDomain(env));
        const content = String(body.content || body.description || '').trim();
        const status = String(body.status || 'publish').toLowerCase() === 'draft' ? 'draft' : 'publish';
        const views = (body.views !== undefined && body.views !== '') ? parseInt(body.views, 10) : 0;

        const nowIso = new Date().toISOString();
        const todayDate = nowIso.split('T')[0];

        let publishedAt = String(body.published_at || body.publish_date || '').trim();
        if (status === 'publish') {
          if (!publishedAt) {
            const prevPub = existing[0].published_at || existing[0].publish_date;
            publishedAt = prevPub ? String(prevPub) : nowIso;
          } else if (publishedAt.length === 10) {
            publishedAt = publishedAt + ' 00:00:00';
          }
        }
        const publishDate = publishedAt ? (publishedAt.includes('T') ? publishedAt.split('T')[0] : publishedAt.split(' ')[0]) : (status === 'publish' ? todayDate : '');

        const availableCols = await getTableColumns(env);
        const candidateFields = [
          { key: 'title', val: title, aliases: [] },
          { key: 'slug', val: slug, aliases: [] },
          { key: 'excerpt', val: excerpt, aliases: [] },
          { key: 'category', val: category, aliases: [] },
          { key: 'author', val: author, aliases: ['author_name'] },
          { key: 'author_name', val: author, aliases: ['author'] },
          { key: 'cover_image', val: coverImage, aliases: ['image', 'coverImage', 'thumbnail'] },
          { key: 'image', val: coverImage, aliases: ['cover_image', 'coverImage', 'thumbnail'] },
          { key: 'content', val: content, aliases: ['description'] },
          { key: 'description', val: content, aliases: ['content'] },
          { key: 'status', val: status, aliases: [] },
          { key: 'views', val: views, aliases: [] },
          { key: 'published_at', val: publishedAt, aliases: ['publish_date'] },
          { key: 'publish_date', val: publishDate, aliases: ['published_at'] }
        ];

        let setClauses = [];
        let setValues = [];
        const handledCols = new Set();

        for (const field of candidateFields) {
          if (hasColumn(availableCols, field.key, field.aliases)) {
            const actualCol = resolveColumnName(availableCols, field.key, field.aliases);
            const lower = actualCol.toLowerCase();
            if (!handledCols.has(lower)) {
              handledCols.add(lower);
              setClauses.push(`${actualCol} = ?`);
              setValues.push(field.val);
            }
          }
        }

        if (hasColumn(availableCols, 'updated_at')) {
          const actualCol = resolveColumnName(availableCols, 'updated_at');
          setClauses.push(`${actualCol} = ?`);
          setValues.push(new Date().toISOString());
        }

        setValues.push(blogId);
        const updateSql = `UPDATE blogs SET ${setClauses.join(', ')} WHERE id = ?`;
        await executeD1(env, updateSql, setValues);

        return jsonResponse({
          success: true,
          message: "Blog article updated successfully in D1 database!",
          id: blogId,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to update blog article: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 5: Delete Blog Post (DELETE /api/blogs/:id or POST /api/blogs/delete)
    // -------------------------------------------------------------------------
    const deleteMatch = pathname.match(/^\/api\/blogs\/(\d+)$/);
    if ((deleteMatch && request.method === 'DELETE') || (pathname === '/api/blogs/delete' && (request.method === 'POST' || request.method === 'DELETE'))) {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          message: "Blog article deleted in Preview Mode.",
          isD1Bound: false
        });
      }
      try {
        await ensureBlogsTable(env);
        let blogId;
        if (deleteMatch) {
          blogId = parseInt(deleteMatch[1], 10);
        } else {
          const body = await request.json().catch(() => ({}));
          blogId = parseInt(body.id || url.searchParams.get('id'), 10);
        }

        if (isNaN(blogId)) return errorResponse("Invalid or missing Blog ID for deletion.", 400, { isD1Bound: true });

        const existing = await executeD1(env, "SELECT * FROM blogs WHERE id = ?", [blogId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No blog article found with ID: #${blogId}`, 404, { isD1Bound: true });
        }

        await executeD1(env, "DELETE FROM blogs WHERE id = ?", [blogId]);

        return jsonResponse({
          success: true,
          message: `Blog #${blogId} ('${existing[0].title}') deleted successfully from D1 database.`,
          id: blogId,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to delete blog article: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 6: Increment Views (POST /api/blogs/increment-view)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/blogs/increment-view' || pathname.endsWith('/view')) && request.method === 'POST') {
      if (!isD1Bound) {
        return jsonResponse({ success: true, message: "View recorded (preview mode)" });
      }
      try {
        const body = await request.json().catch(() => ({}));
        const slug = body.slug || url.searchParams.get('slug');
        const id = parseInt(body.id || url.searchParams.get('id'), 10);

        if (slug) {
          await executeD1(env, "UPDATE blogs SET views = views + 1 WHERE slug = ?", [slug]);
        } else if (!isNaN(id)) {
          await executeD1(env, "UPDATE blogs SET views = views + 1 WHERE id = ?", [id]);
        }
        return jsonResponse({ success: true, isD1Bound: true });
      } catch (e) {
        return errorResponse("Failed to increment views: " + e.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 7: Get Single Blog Details (GET /api/blogs/:identifier or /blogs/:slug)
    // -------------------------------------------------------------------------
    const getSingleMatch = pathname.match(/^\/(?:api\/)?blogs\/([a-zA-Z0-9_-]+)$/);
    if (getSingleMatch && request.method === 'GET' && pathname !== '/api/blogs' && pathname !== '/blogs') {
      const identifier = getSingleMatch[1];
      const r2Domain = getR2PublicDomain(env);
      if (!isD1Bound) {
        const found = DEFAULT_BLOGS.find(b => b.slug === identifier || String(b.id) === identifier);
        if (found) {
          return jsonResponse({ success: true, data: normalizeBlogRow(found, r2Domain), isD1Bound: false });
        }
        return errorResponse("Blog article not found (D1 database not bound)", 404, { isD1Bound: false });
      }
      try {
        await ensureBlogsTable(env);
        let sql = "SELECT * FROM blogs WHERE slug = ? LIMIT 1";
        let params = [];
        if (/^\d+$/.test(identifier)) {
          sql = "SELECT * FROM blogs WHERE id = ? OR slug = ? LIMIT 1";
          params = [parseInt(identifier, 10), identifier];
        } else {
          params = [identifier];
        }
        const rows = await executeD1(env, sql, params);
        if (rows && rows.length > 0) {
          const normalized = normalizeBlogRow(rows[0], r2Domain);
          return jsonResponse({ success: true, data: normalized, isD1Bound: true });
        }
        // Fallback check in DEFAULT_BLOGS
        const fallback = DEFAULT_BLOGS.find(b => b.slug === identifier || String(b.id) === identifier);
        if (fallback) {
          return jsonResponse({ success: true, data: normalizeBlogRow(fallback, r2Domain), isD1Bound: true });
        }
        return errorResponse(`Blog article '${identifier}' not found in D1 database`, 404, { isD1Bound: true });
      } catch (err) {
        return errorResponse(err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 8: List All Blogs (GET /api/blogs or /blogs or /get_blogs)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/blogs' || pathname === '/blogs' || pathname === '/get_blogs') && request.method === 'GET') {
      const search = url.searchParams.get('search') ? url.searchParams.get('search').trim() : '';
      const category = url.searchParams.get('category') ? url.searchParams.get('category').trim() : '';
      const statusParam = url.searchParams.get('status') ? url.searchParams.get('status').trim().toLowerCase() : '';
      const rawArray = url.searchParams.get('raw') === '1' || url.searchParams.get('format') === 'array';

      const filterPublishOnly = statusParam === 'publish' || statusParam === 'published';
      const filterDraftOnly = statusParam === 'draft';

      const r2Domain = getR2PublicDomain(env);

      if (!isD1Bound) {
        let rows = JSON.parse(JSON.stringify(DEFAULT_BLOGS));
        if (filterPublishOnly) {
          rows = rows.filter(r => r.status === 'publish' || r.status === 'published');
        } else if (filterDraftOnly) {
          rows = rows.filter(r => r.status === 'draft');
        }
        if (category && category !== 'All') {
          rows = rows.filter(r => r.category === category);
        }
        if (search) {
          const s = search.toLowerCase();
          rows = rows.filter(r => (r.title && r.title.toLowerCase().includes(s)) || (r.slug && r.slug.toLowerCase().includes(s)));
        }
        const normalizedRows = rows.map(r => normalizeBlogRow(r, r2Domain));

        if (rawArray) {
          return new Response(JSON.stringify(normalizedRows), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' },
            status: 200
          });
        }
        return jsonResponse({
          success: true,
          count: normalizedRows.length,
          data: normalizedRows,
          blogs: normalizedRows,
          isD1Bound: false
        });
      }

      try {
        await ensureBlogsTable(env);
        let sql = "SELECT * FROM blogs";
        let wheres = [];
        let params = [];

        if (filterPublishOnly) {
          wheres.push("(status = 'publish' OR status = 'published')");
        } else if (filterDraftOnly) {
          wheres.push("status = 'draft'");
        }
        if (category && category !== 'All') {
          wheres.push("category = ?");
          params.push(category);
        }
        if (search) {
          wheres.push("(title LIKE ? OR slug LIKE ? OR excerpt LIKE ? OR author LIKE ?)");
          const term = `%${search}%`;
          params.push(term, term, term, term);
        }

        if (wheres.length > 0) {
          sql += " WHERE " + wheres.join(" AND ");
        }
        sql += " ORDER BY id DESC";

        let rows = [];
        try {
          rows = await executeD1(env, sql, params);
        } catch (err) {
          console.warn("API blogs query notice:", err.message);
          rows = [];
        }

        if (!rows) rows = [];

        // If table is completely empty, auto-seed 5 CBT Rank articles into D1
        if (rows.length === 0 && !search && !category && !filterPublishOnly && !filterDraftOnly) {
          try {
            await seedDefaultBlogs(env);
            rows = await executeD1(env, sql, params);
          } catch (e) {}
        }

        let normalizedRows = (rows || []).map(r => normalizeBlogRow(r, r2Domain));
        if (normalizedRows.length === 0 && !search && !category) {
          normalizedRows = DEFAULT_BLOGS.map(r => normalizeBlogRow(r, r2Domain));
        }

        if (rawArray) {
          return new Response(JSON.stringify(normalizedRows), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' },
            status: 200
          });
        }

        return jsonResponse({
          success: true,
          count: normalizedRows.length,
          data: normalizedRows,
          blogs: normalizedRows,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse(err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 9: Web Dashboard Portal (GET / or browser requests)
    // -------------------------------------------------------------------------
    if (request.method === 'GET') {
      let initialBlogs = [];
      let tableCols = [];
      let dbError = null;

      if (isD1Bound) {
        try {
          tableCols = await ensureBlogsTable(env);
        } catch (e) {
          console.warn("Table ensure warning:", e);
        }
        try {
          let rawBlogs = await executeD1(env, "SELECT * FROM blogs ORDER BY id DESC");
          // If D1 database is connected but has no articles yet, auto-seed the 5 CBT Rank articles!
          if (!rawBlogs || rawBlogs.length === 0) {
            await seedDefaultBlogs(env);
            rawBlogs = await executeD1(env, "SELECT * FROM blogs ORDER BY id DESC");
          }
          const r2Domain = getR2PublicDomain(env);
          initialBlogs = (rawBlogs || []).map(r => normalizeBlogRow(r, r2Domain));
        } catch (e) {
          dbError = e.message;
          console.error("D1 Blogs Load Error:", e);
        }
      }

      // Fallback: If D1 is not bound or returned 0 rows, use 5 built-in CBT Rank articles
      if (!initialBlogs || initialBlogs.length === 0) {
        const r2Domain = getR2PublicDomain(env);
        initialBlogs = DEFAULT_BLOGS.map(r => normalizeBlogRow(r, r2Domain));
      }

      return renderBlogsManagerPortal(isD1Bound, isR2Bound, tableCols.length > 0, initialBlogs, dbError, getR2PublicDomain(env));
    }

    return errorResponse("Method or endpoint not found", 404);
  }
};

// Helper to escape HTML in server rendered strings
function escapeHtml(str) {
  if (str === null || str === undefined) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

    function formatDisplayDate(dateStr) {
      if (!dateStr || typeof dateStr !== 'string') return '';
      var s = dateStr.trim();
      if (!s) return '';
      var dPart = s.split('T')[0].split(' ')[0];
      var parts = dPart.split('-');
      if (parts.length === 3 && parts[0].length === 4) {
        var months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
        var y = parts[0];
        var mIdx = parseInt(parts[1], 10) - 1;
        var d = parseInt(parts[2], 10);
        if (mIdx >= 0 && mIdx < 12 && !isNaN(d)) {
          return (d < 10 ? '0' + d : d) + ' ' + months[mIdx] + ' ' + y;
        }
        return dPart;
      }
      return dPart;
    }

    function slugify(text) {
      return (text || '')
        .toString()
        .toLowerCase()
        .trim()
        .replace(/\s+/g, '-')
        .replace(/[^\w\-]+/g, '')
        .replace(/\-\-+/g, '-')
        .replace(/^-+/, '')
        .replace(/-+$/, '');
    }

    async function handleFormSubmit(e) {
      if (e && e.preventDefault) e.preventDefault();
      var saveBtn = document.getElementById('saveBlogBtn');
      var isEdit = document.getElementById('modalMode').value === 'edit';
      var blogId = document.getElementById('formBlogId').value;
      var title = document.getElementById('formTitle').value.trim();
      var slug = document.getElementById('formSlug').value.trim();

      if (!title) {
        alert('Please enter the Article Title.');
        document.getElementById('formTitle').focus();
        return;
      }
      if (!slug) {
        slug = slugify(title);
        document.getElementById('formSlug').value = slug;
      }

      var category = document.getElementById('formCategory').value;
      if (category === 'Other') {
        var customCat = document.getElementById('formCustomCategory').value.trim();
        if (!customCat) {
          alert('Please enter a custom category name.');
          document.getElementById('formCustomCategory').focus();
          return;
        }
        category = customCat;
      }

      saveBtn.disabled = true;
      saveBtn.innerText = '⏳ Saving...';

      var statusVal = document.getElementById('formStatus').value;
      var pubDateVal = document.getElementById('formPublishDate') ? document.getElementById('formPublishDate').value.trim() : '';
      var finalPublishedAt = '';
      var finalPublishDate = '';
      if (statusVal === 'publish') {
        finalPublishDate = pubDateVal || new Date().toISOString().split('T')[0];
        finalPublishedAt = finalPublishDate + ' 00:00:00';
      } else {
        finalPublishDate = pubDateVal || '';
        finalPublishedAt = pubDateVal ? pubDateVal + ' 00:00:00' : '';
      }

      var payload = {
        title: title,
        slug: slug,
        category: category,
        author: document.getElementById('formAuthor').value.trim() || 'Team CBTRANK',
        cover_image: document.getElementById('formCoverImage').value.trim(),
        excerpt: document.getElementById('formExcerpt').value.trim(),
        content: getContentValue(),
        status: statusVal,
        published_at: finalPublishedAt,
        publish_date: finalPublishDate,
        views: parseInt(document.getElementById('formViews').value, 10) || 0
      };

      if (isD1Connected) {
        try {
          var res;
          if (isEdit) {
            payload.id = blogId;
            res = await fetch('/api/blogs/' + blogId, {
              method: 'PUT',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(payload)
            });
          } else {
            res = await fetch('/api/blogs', {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(payload)
            });
          }

          var data = await res.json();
          saveBtn.disabled = false;
          saveBtn.innerText = isEdit ? '💾 Update Article' : '💾 Create Article';

          if (!data.success) {
            alert('Database Error: ' + (data.error || 'Failed to save blog article to D1'));
            return;
          }

          closeModal('blogModal');
          showToast(isEdit ? 'Article updated in D1 Database!' : 'Article added to D1 Database!');
          await refreshData();
        } catch (err) {
          saveBtn.disabled = false;
          saveBtn.innerText = isEdit ? '💾 Update Article' : '💾 Create Article';
          alert('Network / API Error: ' + err.message);
        }
      } else {
        var newId = isEdit ? parseInt(blogId, 10) : (allBlogs.length > 0 ? Math.max.apply(null, allBlogs.map(function(x) { return x.id || 0; })) + 1 : 1);
        payload.id = newId;
        var existingItem = isEdit ? allBlogs.find(function(x) { return x.id == newId; }) : null;
        payload.created_at = (existingItem && existingItem.created_at) ? existingItem.created_at : new Date().toISOString();
        if (isEdit) {
          for (var k = 0; k < allBlogs.length; k++) {
            if (allBlogs[k].id == newId) { allBlogs[k] = payload; break; }
          }
        } else {
          allBlogs.unshift(payload);
        }
        saveLocalStorage();
        saveBtn.disabled = false;
        saveBtn.innerText = isEdit ? '💾 Update Article' : '💾 Create Article';
        closeModal('blogModal');
        showToast(isEdit ? 'Article updated & saved locally!' : 'Article added & saved locally!');
        renderAllViews();
        updateStats();
      }
    }

    async function toggleStatus(id) {
      var blog = allBlogs.find(function(x) { return x.id == id; });
      if (!blog) return;

      var currentStatus = (String(blog.status).toLowerCase() === 'publish' || String(blog.status).toLowerCase() === 'published') ? 'publish' : 'draft';
      var newStatus = (currentStatus === 'publish') ? 'draft' : 'publish';
      var prevPublishedAt = blog.published_at;
      var prevPublishDate = blog.publish_date;

      var nowIso = new Date().toISOString();
      var todayStr = nowIso.split('T')[0];

      if (!isD1Connected) {
        blog.status = newStatus;
        if (newStatus === 'publish') {
          blog.published_at = nowIso;
          blog.publish_date = todayStr;
        }
        saveLocalStorage();
        renderAllViews();
        updateStats();
        showToast('Article #' + id + ' status: ' + newStatus + (newStatus === 'publish' ? ' (' + formatDisplayDate(todayStr) + ')' : '') + ' (Saved locally)');
        return;
      }

      blog.status = newStatus;
      if (newStatus === 'publish') {
        blog.published_at = nowIso;
        blog.publish_date = todayStr;
      }
      renderAllViews();
      updateStats();

      try {
        var res = await fetch('/api/blogs/toggle', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ id: id, status: newStatus })
        });
        var data = await res.json();

        if (!data.success) {
          blog.status = currentStatus;
          blog.published_at = prevPublishedAt;
          blog.publish_date = prevPublishDate;
          renderAllViews();
          updateStats();
          alert('Toggle failed in D1: ' + (data.error || 'Unknown error'));
        } else {
          if (data.published_at) blog.published_at = data.published_at;
          if (data.publish_date) blog.publish_date = data.publish_date;
          renderAllViews();
          updateStats();
          showToast('Article #' + id + ' status updated to ' + newStatus + (newStatus === 'publish' ? ' (Publish Date: ' + formatDisplayDate(blog.publish_date || todayStr) + ')' : '') + ' in D1');
        }
      } catch (err) {
        blog.status = currentStatus;
        blog.published_at = prevPublishedAt;
        blog.publish_date = prevPublishDate;
        renderAllViews();
        updateStats();
        alert('Network error on toggle: ' + err.message);
      }
    }

    async function refreshData() {
      if (!isD1Connected) {
        try {
          var saved = localStorage.getItem('cbtrank_blogs_store');
          if (saved) {
            var parsed = JSON.parse(saved);
            if (Array.isArray(parsed)) allBlogs = parsed;
          }
        } catch (e) {}
        renderAllViews();
        updateStats();
        return;
      }
      try {
        var res = await fetch('/api/blogs');
        var json = await res.json();
        if (json.success && Array.isArray(json.data)) {
          allBlogs = json.data;
          renderAllViews();
          updateStats();
          showToast('Articles refreshed from D1 (' + allBlogs.length + ' live)');
        }
      } catch (e) {
        console.error('Refresh error:', e);
      }
    }

        var data = await res.json();
        if (data.success) {
          showToast(data.message || 'Articles seeded successfully into D1!');
          await refreshData();
        } else {
          alert('Seed error: ' + (data.error || 'Failed to seed articles'));
        }
      } catch (e) {
        alert('Request failed: ' + e.message);

    function updateActiveFilterPill() {
      var pills = document.querySelectorAll('.filter-pill');
      pills.forEach(function(p) {
        var text = p.innerText.trim();
        if (currentStatusFilter === 'publish' && text === 'Published') {
          p.classList.add('active');
        } else if (currentStatusFilter === 'draft' && text === 'Drafts') {
          p.classList.add('active');
        } else if (currentStatusFilter === 'all' && currentCategoryFilter === 'All' && text === 'All Categories') {
          p.classList.add('active');
        } else if (currentStatusFilter === 'all' && text === currentCategoryFilter) {

    function onSearchChange(val) {

    function getFilteredBlogs() {
      return allBlogs.filter(function(b) {
        if (currentStatusFilter === 'publish') {
          var st = String(b.status || '').toLowerCase();
          if (st !== 'publish' && st !== 'published') return false;
        } else if (currentStatusFilter === 'draft') {
          if (String(b.status || '').toLowerCase() !== 'draft') return false;
        }

        if (currentCategoryFilter !== 'All') {
          if (b.category !== currentCategoryFilter) return false;
        }

        if (searchQuery) {
          var titleMatch = b.title && b.title.toLowerCase().includes(searchQuery);
          var slugMatch = b.slug && b.slug.toLowerCase().includes(searchQuery);
          var catMatch = b.category && b.category.toLowerCase().includes(searchQuery);
          var authorMatch = (b.author || b.author_name) && (b.author || b.author_name).toLowerCase().includes(searchQuery);
          var excerptMatch = b.excerpt && b.excerpt.toLowerCase().includes(searchQuery);
          if (!titleMatch && !slugMatch && !catMatch && !authorMatch && !excerptMatch) {
            return false;
          }
        }
        return true;
      });
    }

    function updateStats() {
      var total = allBlogs.length;
      var pub = allBlogs.filter(function(b) {
        var st = String(b.status || '').toLowerCase();
        return st === 'publish' || st === 'published';
      }).length;
      var drafts = total - pub;
      var totalViews = allBlogs.reduce(function(acc, b) { return acc + (parseInt(b.views, 10) || 0); }, 0);

      var sTotal = document.getElementById('statTotal');
      var sPub = document.getElementById('statPublished');
      var sDrafts = document.getElementById('statDrafts');
      var sViews = document.getElementById('statViews');

      if (sTotal) sTotal.innerText = total;
      if (sPub) sPub.innerText = pub;
      if (sDrafts) sDrafts.innerText = drafts;
      if (sViews) sViews.innerText = totalViews;
    }

        .replace(/</g, '&lt;')

        .replace(/"/g, '&quot;')
