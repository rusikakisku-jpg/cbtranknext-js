// =============================================================================
// Helper: Safe Cloudflare D1 Database Instance Extractor
// =============================================================================
function getD1Database(env) {
  if (!env) return null;
  if (env.DB && typeof env.DB.prepare === 'function') return env.DB;
  if (env.DATABASE && typeof env.DATABASE.prepare === 'function') return env.DATABASE;
  if (env.d1 && typeof env.d1.prepare === 'function') return env.d1;
  // Scan all env properties for custom D1 binding names
  for (const key of Object.keys(env)) {
    const val = env[key];
    if (val && typeof val === 'object' && typeof val.prepare === 'function') {
      return val;
    }
  }
  return null;
}

// =============================================================================
// Helper: D1 Query Executor (Handles Parameterized Bindings & Mutations)
// =============================================================================
async function executeD1(env, sql, params = []) {
  const db = getD1Database(env);
  if (!db) {
    throw new Error("Cloudflare D1 database binding 'DB' is missing. Please bind your D1 database under Worker Settings -> Bindings -> D1 Database Bindings with variable name DB.");
  }

  const cleanParams = params.map(p => {
    if (p === undefined || p === null) return null;
    if (typeof p === 'boolean') return p ? 1 : 0;
    return p;
  });

  try {
    const stmt = db.prepare(sql);
    const bound = cleanParams.length > 0 ? stmt.bind(...cleanParams) : stmt;
    const isSelectOrPragma = /^\s*(SELECT|PRAGMA)/i.test(sql);

    if (isSelectOrPragma) {
      const res = await bound.all();
      return (res && res.results) ? res.results : (res || []);
    } else {
      const res = await bound.run();
      return res;
    }
  } catch (err) {
    console.error("D1 Execution Error:", err.message, "\nSQL:", sql, "\nParams:", params);
    throw err;
  }
}

// =============================================================================
// Helper: Inspect Existing Columns of 'blogs' Table via PRAGMA
// =============================================================================
async function getTableColumns(env) {
  try {
    const info = await executeD1(env, "PRAGMA table_info(blogs)");
    if (Array.isArray(info) && info.length > 0) {
      return info.map(c => c.name);
    }
  } catch (e) {
    // Table may not exist yet
  }
  return [];
}

// =============================================================================
// Helper: Auto-create & Self-Healing Migration for 'blogs' Table
// =============================================================================
async function ensureBlogsTable(env) {
  const db = getD1Database(env);
  if (!db) return [];

  const createSql = `
    CREATE TABLE IF NOT EXISTS blogs (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      slug TEXT NOT NULL UNIQUE,
      excerpt TEXT DEFAULT '',
      category TEXT DEFAULT 'Exam Analysis',
      author TEXT DEFAULT 'Team CBTRANK',
      author_name TEXT DEFAULT 'Team CBTRANK',
      cover_image TEXT DEFAULT '',
      image TEXT DEFAULT '',
      content TEXT DEFAULT '',
      description TEXT DEFAULT '',
      status TEXT DEFAULT 'publish',
      views INTEGER DEFAULT 0,
      created_at TEXT DEFAULT CURRENT_TIMESTAMP,
      updated_at TEXT DEFAULT CURRENT_TIMESTAMP
    );
  `;
  await db.prepare(createSql).run();

  // Self-healing: verify all required columns exist, add if missing safely
  const cols = await getTableColumns(env);
  const requiredCols = [
    { name: 'excerpt', type: "TEXT DEFAULT ''" },
    { name: 'category', type: "TEXT DEFAULT 'Exam Analysis'" },
    { name: 'author', type: "TEXT DEFAULT 'Team CBTRANK'" },
    { name: 'author_name', type: "TEXT DEFAULT 'Team CBTRANK'" },
    { name: 'cover_image', type: "TEXT DEFAULT ''" },
    { name: 'image', type: "TEXT DEFAULT ''" },
    { name: 'content', type: "TEXT DEFAULT ''" },
    { name: 'description', type: "TEXT DEFAULT ''" },
    { name: 'status', type: "TEXT DEFAULT 'publish'" },
    { name: 'views', type: "INTEGER DEFAULT 0" },
    { name: 'created_at', type: "TEXT DEFAULT ''" },
    { name: 'updated_at', type: "TEXT DEFAULT ''" }
  ];

  for (const col of requiredCols) {
    if (!cols.includes(col.name)) {
      try {
        await db.prepare(`ALTER TABLE blogs ADD COLUMN ${col.name} ${col.type}`).run();
      } catch (e) {
        // Handled safely
      }
    }
  }

  return await getTableColumns(env);
}

// Helper: Seed Default 5 CBT Rank Blog Articles into D1 Table
// =============================================================================
async function seedDefaultBlogs(env) {
  const db = getD1Database(env);
  if (!db) return 0;
  await ensureBlogsTable(env);

  let insertedCount = 0;
  let maxId = 0;

  for (const b of DEFAULT_BLOGS) {
    try {
      await db.prepare(`
        INSERT OR IGNORE INTO blogs (id, title, slug, excerpt, category, author, author_name, cover_image, image, content, description, status, views, created_at, updated_at)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).bind(
        b.id,
        b.title,
        b.slug,
        b.excerpt || '',
        b.category || 'Exam Analysis',
        b.author || 'Team CBTRANK',
        b.author_name || b.author || 'Team CBTRANK',
        b.cover_image || '',
        b.image || b.cover_image || '',
        b.content || '',
        b.description || b.content || '',
        b.status || 'publish',
        b.views || 0,
        b.created_at || new Date().toISOString(),
        b.updated_at || new Date().toISOString()
      ).run();
      insertedCount++;
      if (b.id > maxId) maxId = b.id;
    } catch (err) {
      console.warn("Seed row notice:", b.slug, err.message);
    }
  }

  // Ensure sqlite_sequence is at least maxId to prevent AUTOINCREMENT collision
  if (maxId > 0) {
    try {
      await db.prepare("INSERT OR REPLACE INTO sqlite_sequence (name, seq) VALUES ('blogs', MAX(COALESCE((SELECT seq FROM sqlite_sequence WHERE name = 'blogs'), 0), ?))").bind(maxId).run();
    } catch (e) {}
  }

  return insertedCount;
}

// Helper: CORS & JSON Responses
// =============================================================================
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type, Authorization, x-api-key',
  'Access-Control-Max-Age': '86400'
};

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...corsHeaders,
      'Content-Type': 'application/json; charset=utf-8'
    }
  });
}

function errorResponse(message, status = 400, extra = {}) {
  return jsonResponse({ success: false, error: message, ...extra }, status);
}

// =============================================================================
// Main Worker Router & Handler
// =============================================================================
export default {
  async fetch(request, env, ctx) {
    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders, status: 204 });
    }

    const url = new URL(request.url);
    let pathname = url.pathname.replace(/\/+/g, '/');
    if (pathname.length > 1 && pathname.endsWith('/')) {
      pathname = pathname.slice(0, -1);
    }

    const dbInstance = getD1Database(env);
    const isD1Bound = !!dbInstance;

    // -------------------------------------------------------------------------
    // ROUTE 0: Seed / Reset Default Blog Posts (/api/seed-blogs)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/seed-blogs' || pathname === '/api/blogs/seed') && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          seeded: DEFAULT_BLOGS.length,
          message: "Running in preview mode (D1 not bound). Displaying built-in default blog articles.",
          isD1Bound: false
        });
      }
      try {
        const count = await seedDefaultBlogs(env);
        return jsonResponse({
          success: true,
          seeded: count,
          message: `Successfully seeded/verified ${count} default CBT Rank blog articles in D1 database.`,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Seeding failed: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 1: Schema Diagnostics & Setup
    // -------------------------------------------------------------------------
    if (pathname === '/api/schema' || pathname === '/api/diagnostics') {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500, { isD1Bound: false });
      try {
        const columns = await getTableColumns(env);
        const countRes = columns.length > 0 ? await executeD1(env, "SELECT COUNT(*) as total FROM blogs") : [{ total: 0 }];
        return jsonResponse({
          success: true,
          isD1Bound: true,
          tableExists: columns.length > 0,
          totalBlogs: countRes[0]?.total ?? 0,
          columns: columns
        });
      } catch (err) {
        return errorResponse(err.message, 500, { isD1Bound: true });
      }
    }

    if (pathname === '/api/setup-table' && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500, { isD1Bound: false });
      try {
        await ensureBlogsTable(env);
        const columns = await getTableColumns(env);
        return jsonResponse({
          success: true,
          message: "Table 'blogs' verified / created successfully in D1 database.",
          columns: columns,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to setup table: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 2: Quick Status Toggle (status = 'publish' <-> 'draft')
    // -------------------------------------------------------------------------
    if (pathname === '/api/blogs/toggle' && request.method === 'POST') {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          message: "Preview Mode: Updated in memory",
          isD1Bound: false
        });
      }
      try {
        await ensureBlogsTable(env);
        const body = await request.json();
        const id = parseInt(body.id, 10);
        if (isNaN(id)) return errorResponse("Invalid or missing Blog ID.", 400, { isD1Bound: true });

        const currentBlog = await executeD1(env, "SELECT id, status FROM blogs WHERE id = ?", [id]);
        if (!currentBlog || currentBlog.length === 0) {
          return errorResponse("Blog post not found with ID: #" + id, 404, { isD1Bound: true });
        }

        const currentStatus = String(currentBlog[0].status || 'draft').toLowerCase();
        const newStatus = body.status
          ? String(body.status).toLowerCase()
          : (currentStatus === 'publish' || currentStatus === 'published' ? 'draft' : 'publish');

        const availableCols = await getTableColumns(env);
        let updateSql = "UPDATE blogs SET status = ?";
        let updateParams = [newStatus];
        if (availableCols.includes('updated_at')) {
          updateSql += ", updated_at = ?";
          updateParams.push(new Date().toISOString());
        }
        updateSql += " WHERE id = ?";
        updateParams.push(id);

        await executeD1(env, updateSql, updateParams);

        return jsonResponse({
          success: true,
          message: `Blog #${id} status updated to '${newStatus}' in D1 database`,
          id: id,
          newStatus: newStatus,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Toggle failed: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 3: Add New Blog Post (POST /api/blogs or POST /api/blogs/add)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/blogs' || pathname === '/api/blogs/add') && request.method === 'POST') {
      if (!isD1Bound) {
        try {
          const body = await request.json();
          return jsonResponse({
            success: true,
            message: "Blog post added in Preview Mode (bind D1 in Cloudflare to persist).",
            id: Date.now(),
            isD1Bound: false
          });
        } catch (e) {
          return errorResponse("Invalid JSON payload", 400, { isD1Bound: false });
        }
      }
      try {
        await ensureBlogsTable(env);
        const body = await request.json();
        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();

        if (!title) return errorResponse("Field 'title' is required.", 400, { isD1Bound: true });
        if (!slug) {
          slug = title
            .toLowerCase()
            .replace(/[^a-z0-9]+/g, '-')
            .replace(/^-+|-+$/g, '');
        }

        const existing = await executeD1(env, "SELECT id FROM blogs WHERE slug = ? LIMIT 1", [slug]);
        if (existing && existing.length > 0) {
          return errorResponse(`A blog article with slug '${slug}' already exists (ID: #${existing[0].id}). Please choose a unique slug.`, 400, { isD1Bound: true });
        }

        const excerpt = String(body.excerpt || '').trim();
        const category = String(body.category || 'Exam Analysis').trim();
        const author = String(body.author || body.author_name || 'Team CBTRANK').trim();
        const coverImage = String(body.cover_image || body.image || '').trim();
        const content = String(body.content || body.description || '').trim();
        const status = String(body.status || 'publish').toLowerCase() === 'draft' ? 'draft' : 'publish';
        const views = parseInt(body.views, 10) || 0;

        const availableCols = await getTableColumns(env);
        const nowIso = new Date().toISOString();

        // 1. Determine next unique ID safely to prevent 'NOT NULL constraint failed: blogs.id'
        let nextId = 1;
        try {
          const maxRes = await executeD1(env, "SELECT COALESCE(MAX(id), 0) AS max_id FROM blogs");
          if (maxRes && maxRes.length > 0 && maxRes[0].max_id !== undefined && maxRes[0].max_id !== null) {
            nextId = parseInt(maxRes[0].max_id, 10) + 1;
            if (isNaN(nextId) || nextId < 1) nextId = 1;
          }
        } catch (e) {
          nextId = Date.now();
        }

        const dataMap = {
          title,
          slug,
          excerpt,
          category,
          author,
          author_name: author,
          cover_image: coverImage,
          image: coverImage,
          content,
          description: content,
          status,
          views
        };

        // Explicitly include id as first column to satisfy NOT NULL constraints
        let colsToInsert = ['id'];
        let placeholders = ['?'];
        let valuesToInsert = [nextId];

        Object.keys(dataMap).forEach(col => {
          if (availableCols.length === 0 || availableCols.includes(col)) {
            colsToInsert.push(col);
            placeholders.push('?');
            valuesToInsert.push(dataMap[col]);
          }
        });

        if (availableCols.includes('created_at')) {
          colsToInsert.push('created_at');
          placeholders.push('?');
          valuesToInsert.push(nowIso);
        }
        if (availableCols.includes('updated_at')) {
          colsToInsert.push('updated_at');
          placeholders.push('?');
          valuesToInsert.push(nowIso);
        }

        const insertSql = `INSERT INTO blogs (${colsToInsert.join(', ')}) VALUES (${placeholders.join(', ')})`;
        const result = await executeD1(env, insertSql, valuesToInsert);

        // Sync autoincrement sequence if sqlite_sequence exists
        try {
          const dbInstance = getD1Database(env);
          if (dbInstance) {
            await dbInstance.prepare("INSERT OR REPLACE INTO sqlite_sequence (name, seq) VALUES ('blogs', MAX(COALESCE((SELECT seq FROM sqlite_sequence WHERE name = 'blogs'), 0), ?))").bind(nextId).run();
          }
        } catch (e) {}

        // Retrieve real assigned ID directly from D1 to guarantee persistence
        let newId = nextId;
        try {
          const insertedRow = await executeD1(env, "SELECT id FROM blogs WHERE slug = ? LIMIT 1", [slug]);
          if (insertedRow && insertedRow.length > 0) {
            newId = insertedRow[0].id;
          }
        } catch (e) {}

        return jsonResponse({
          success: true,
          message: "Blog article added successfully to D1 database!",
          id: newId,
          slug: slug,
          isD1Bound: true
        }, 201);
      } catch (err) {
        return errorResponse("Failed to add blog article: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 4: Update Blog Post (PUT /api/blogs/:id or POST /api/blogs/update)
    // -------------------------------------------------------------------------
    const updateMatch = pathname.match(/^\/api\/blogs\/(\d+)$/);
    if ((updateMatch && (request.method === 'PUT' || request.method === 'POST')) || (pathname === '/api/blogs/update' && request.method === 'POST')) {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          message: "Blog article updated in Preview Mode.",
          isD1Bound: false
        });
      }
      try {
        await ensureBlogsTable(env);
        const body = await request.json();
        const blogId = updateMatch ? parseInt(updateMatch[1], 10) : parseInt(body.id, 10);
        if (isNaN(blogId)) return errorResponse("Invalid or missing Blog ID for update.", 400, { isD1Bound: true });

        const existing = await executeD1(env, "SELECT id, slug FROM blogs WHERE id = ?", [blogId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No blog article found with ID: #${blogId}`, 404, { isD1Bound: true });
        }

        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();
        if (!title) return errorResponse("Title cannot be empty.", 400, { isD1Bound: true });
        if (!slug) {
          slug = title.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
        }

        const slugCheck = await executeD1(env, "SELECT id FROM blogs WHERE slug = ? AND id != ? LIMIT 1", [slug, blogId]);
        if (slugCheck && slugCheck.length > 0) {
          return errorResponse(`Slug '${slug}' is already in use by another article (ID: #${slugCheck[0].id}).`, 400, { isD1Bound: true });
        }

        const excerpt = String(body.excerpt || '').trim();
        const category = String(body.category || 'Exam Analysis').trim();
        const author = String(body.author || body.author_name || 'Team CBTRANK').trim();
        const coverImage = String(body.cover_image || body.image || '').trim();
        const content = String(body.content || body.description || '').trim();
        const status = String(body.status || 'publish').toLowerCase() === 'draft' ? 'draft' : 'publish';
        const views = (body.views !== undefined && body.views !== '') ? parseInt(body.views, 10) : 0;

        const availableCols = await getTableColumns(env);
        const dataMap = {
          title,
          slug,
          excerpt,
          category,
          author,
          author_name: author,
          cover_image: coverImage,
          image: coverImage,
          content,
          description: content,
          status,
          views
        };

        let setClauses = [];
        let setValues = [];

        Object.keys(dataMap).forEach(col => {
          if (availableCols.length === 0 || availableCols.includes(col)) {
            setClauses.push(`${col} = ?`);
            setValues.push(dataMap[col]);
          }
        });

        if (availableCols.includes('updated_at')) {
          setClauses.push("updated_at = ?");
          setValues.push(new Date().toISOString());
        }

        setValues.push(blogId);
        const updateSql = `UPDATE blogs SET ${setClauses.join(', ')} WHERE id = ?`;
        await executeD1(env, updateSql, setValues);

        return jsonResponse({
          success: true,
          message: "Blog article updated successfully in D1 database!",
          id: blogId,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to update blog article: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 5: Delete Blog Post (DELETE /api/blogs/:id or POST /api/blogs/delete)
    // -------------------------------------------------------------------------
    const deleteMatch = pathname.match(/^\/api\/blogs\/(\d+)$/);
    if ((deleteMatch && request.method === 'DELETE') || (pathname === '/api/blogs/delete' && (request.method === 'POST' || request.method === 'DELETE'))) {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          message: "Blog article deleted in Preview Mode.",
          isD1Bound: false
        });
      }
      try {
        await ensureBlogsTable(env);
        let blogId;
        if (deleteMatch) {
          blogId = parseInt(deleteMatch[1], 10);
        } else {
          const body = await request.json().catch(() => ({}));
          blogId = parseInt(body.id || url.searchParams.get('id'), 10);
        }

        if (isNaN(blogId)) return errorResponse("Invalid or missing Blog ID for deletion.", 400, { isD1Bound: true });

        const existing = await executeD1(env, "SELECT id, title FROM blogs WHERE id = ?", [blogId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No blog article found with ID: #${blogId}`, 404, { isD1Bound: true });
        }

        await executeD1(env, "DELETE FROM blogs WHERE id = ?", [blogId]);

        return jsonResponse({
          success: true,
          message: `Blog #${blogId} ('${existing[0].title}') deleted successfully from D1 database.`,
          id: blogId,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to delete blog article: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 6: Increment Views (POST /api/blogs/increment-view)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/blogs/increment-view' || pathname.endsWith('/view')) && request.method === 'POST') {
      if (!isD1Bound) {
        return jsonResponse({ success: true, message: "View recorded (preview mode)" });
      }
      try {
        const body = await request.json().catch(() => ({}));
        const slug = body.slug || url.searchParams.get('slug');
        const id = parseInt(body.id || url.searchParams.get('id'), 10);

        if (slug) {
          await executeD1(env, "UPDATE blogs SET views = views + 1 WHERE slug = ?", [slug]);
        } else if (!isNaN(id)) {
          await executeD1(env, "UPDATE blogs SET views = views + 1 WHERE id = ?", [id]);
        }
        return jsonResponse({ success: true, isD1Bound: true });
      } catch (e) {
        return errorResponse("Failed to increment views: " + e.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 7: Get Single Blog Details (GET /api/blogs/:identifier or /blogs/:slug)
    // -------------------------------------------------------------------------
    const getSingleMatch = pathname.match(/^\/(?:api\/)?blogs\/([a-zA-Z0-9_-]+)$/);
    if (getSingleMatch && request.method === 'GET' && pathname !== '/api/blogs' && pathname !== '/blogs') {
      const identifier = getSingleMatch[1];
      if (!isD1Bound) {
        const found = DEFAULT_BLOGS.find(b => b.slug === identifier || String(b.id) === identifier);
        if (found) return jsonResponse({ success: true, data: found, isD1Bound: false });
        return errorResponse("Blog article not found", 404, { isD1Bound: false });
      }
      try {
        let sql = "SELECT * FROM blogs WHERE slug = ? LIMIT 1";
        let params = [identifier];
        if (/^\d+$/.test(identifier)) {
          sql = "SELECT * FROM blogs WHERE id = ? OR slug = ? LIMIT 1";
          params = [parseInt(identifier, 10), identifier];
        }
        const rows = await executeD1(env, sql, params);
        if (rows && rows.length > 0) {
          return jsonResponse({ success: true, data: rows[0], isD1Bound: true });
        }
        return errorResponse(`Blog article '${identifier}' not found in D1 database`, 404, { isD1Bound: true });
      } catch (err) {
        return errorResponse(err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 8: List All Blogs (GET /api/blogs or /blogs or /get_blogs)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/blogs' || pathname === '/blogs' || pathname === '/get_blogs') && request.method === 'GET') {
      const search = url.searchParams.get('search') ? url.searchParams.get('search').trim() : '';
      const category = url.searchParams.get('category') ? url.searchParams.get('category').trim() : '';
      const statusParam = url.searchParams.get('status') ? url.searchParams.get('status').trim().toLowerCase() : '';
      const rawArray = url.searchParams.get('raw') === '1' || url.searchParams.get('format') === 'array';

      // If requested by external frontend Next.js app/data/blogs.ts without specific status,
      // default behavior serves published posts or all based on parameter
      const filterPublishOnly = statusParam === 'publish' || statusParam === 'published';
      const filterDraftOnly = statusParam === 'draft';

      if (!isD1Bound) {
        let rows = JSON.parse(JSON.stringify(DEFAULT_BLOGS));
        if (filterPublishOnly) rows = rows.filter(r => r.status === 'publish' || r.status === 'published');
        if (filterDraftOnly) rows = rows.filter(r => r.status === 'draft');
        if (category && category !== 'All') rows = rows.filter(r => r.category === category);
        if (search) {
          const sLower = search.toLowerCase();
          rows = rows.filter(r =>
            (r.title && r.title.toLowerCase().includes(sLower)) ||
            (r.slug && r.slug.toLowerCase().includes(sLower)) ||
            (r.excerpt && r.excerpt.toLowerCase().includes(sLower)) ||
            (r.author && r.author.toLowerCase().includes(sLower))
          );
        }
        if (rawArray) {
          return new Response(JSON.stringify(rows), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' },
            status: 200
          });
        }
        return jsonResponse({
          success: true,
          count: rows.length,
          data: rows,
          blogs: rows,
          isD1Bound: false
        });
      }
      try {
        let sql = "SELECT * FROM blogs";
        let wheres = [];
        let params = [];

        if (filterPublishOnly) {
          wheres.push("(status = 'publish' OR status = 'published')");
        } else if (filterDraftOnly) {
          wheres.push("status = 'draft'");
        }
        if (category && category !== 'All') {
          wheres.push("category = ?");
          params.push(category);
        }
        if (search) {
          wheres.push("(title LIKE ? OR slug LIKE ? OR excerpt LIKE ? OR author LIKE ?)");
          const term = `%${search}%`;
          params.push(term, term, term, term);
        }

        if (wheres.length > 0) {
          sql += " WHERE " + wheres.join(" AND ");
        }
        sql += " ORDER BY id DESC";

        let rows = [];
        try {
          rows = await executeD1(env, sql, params);
        } catch (err) {
          console.warn("API blogs query fallback:", err.message);
        }

        if (!rows || rows.length === 0) {
          if (!search && !category && !statusParam) {
            rows = JSON.parse(JSON.stringify(DEFAULT_BLOGS));
          } else {
            rows = [];
          }
        }

        if (rawArray) {
          return new Response(JSON.stringify(rows), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' },
            status: 200
          });
        }

        return jsonResponse({
          success: true,
          count: rows.length,
          data: rows,
          blogs: rows,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse(err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 9: Web Dashboard Portal (GET / or browser requests)
    // -------------------------------------------------------------------------
    if (request.method === 'GET') {
      let initialBlogs = [];
      let tableCols = [];
      let dbError = null;

      if (isD1Bound) {
        try {
          // 1. Auto-create table & migrate missing columns
          tableCols = await ensureBlogsTable(env);

          // 2. Fetch existing blogs
          initialBlogs = await executeD1(env, "SELECT * FROM blogs ORDER BY id DESC");

          // 3. Auto-seed 5 default CBT Rank blogs if database table is empty
          if (!initialBlogs || initialBlogs.length === 0) {
            await seedDefaultBlogs(env);
            initialBlogs = await executeD1(env, "SELECT * FROM blogs ORDER BY id DESC");
          }
        } catch (e) {
          dbError = e.message;
          console.error("D1 Blogs Load Error:", e);
        }
      }

      // 4. Fallback: If D1 is not bound or query returned 0 rows, use 5 built-in default blogs
      if (!initialBlogs || initialBlogs.length === 0) {
        initialBlogs = JSON.parse(JSON.stringify(DEFAULT_BLOGS));
      }

      return renderBlogsManagerPortal(isD1Bound, tableCols.length > 0, initialBlogs, dbError);
    }

    return errorResponse("Method or endpoint not found", 404);
  }
};

    async function handleFormSubmit(e) {
      if (e && e.preventDefault) e.preventDefault();
      var saveBtn = document.getElementById('saveBlogBtn');
      var isEdit = document.getElementById('modalMode').value === 'edit';
      var blogId = document.getElementById('formBlogId').value;
      var title = document.getElementById('formTitle').value.trim();
      var slug = document.getElementById('formSlug').value.trim();

      if (!title) {
        alert('Please enter the Article Title.');
        document.getElementById('formTitle').focus();
        return;
      }
      if (!slug) {
        slug = slugify(title);
        document.getElementById('formSlug').value = slug;
      }

      var category = document.getElementById('formCategory').value;
      if (category === 'Other') {
        var customCat = document.getElementById('formCustomCategory').value.trim();
        if (!customCat) {
          alert('Please enter a custom category name.');
          document.getElementById('formCustomCategory').focus();
          return;
        }
        category = customCat;
      }

      saveBtn.disabled = true;
      saveBtn.innerText = '⏳ Saving...';

      var payload = {
        title: title,
        slug: slug,
        category: category,
        author: document.getElementById('formAuthor').value.trim() || 'Team CBTRANK',
        author_name: document.getElementById('formAuthor').value.trim() || 'Team CBTRANK',
        cover_image: document.getElementById('formCoverImage').value.trim(),
        image: document.getElementById('formCoverImage').value.trim(),
        excerpt: document.getElementById('formExcerpt').value.trim(),
        content: getContentValue().trim(),
        description: getContentValue().trim(),
        status: document.getElementById('formStatus').value,
        views: parseInt(document.getElementById('formViews').value, 10) || 0
      };

      if (isD1Connected) {
        try {
          var res;
          if (isEdit) {
            payload.id = parseInt(blogId, 10);
            res = await fetch('/api/blogs/' + blogId, {
              method: 'PUT',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(payload)
            });
          } else {
            res = await fetch('/api/blogs', {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(payload)
            });
          }

          var data = await res.json();
          saveBtn.disabled = false;
          saveBtn.innerText = isEdit ? '💾 Update Article' : '💾 Create Article';

          if (!data.success) {
            alert('Database Error: ' + (data.error || 'Failed to save blog article to D1'));
            return;
          }

          closeModal('blogModal');
          showToast(isEdit ? 'Article updated in D1 Database!' : 'Article added to D1 Database!');
          await refreshData();
        } catch (err) {
          saveBtn.disabled = false;
          saveBtn.innerText = isEdit ? '💾 Update Article' : '💾 Create Article';
          alert('Network / API Error: ' + err.message);
        }
      } else {
        // Preview Mode Fallback (when D1 is not bound)
        var newId = isEdit ? parseInt(blogId, 10) : (allBlogs.length > 0 ? Math.max.apply(null, allBlogs.map(function(x) { return x.id || 0; })) + 1 : 1);
        payload.id = newId;
        payload.created_at = new Date().toISOString();
        if (isEdit) {
          for (var k = 0; k < allBlogs.length; k++) {
            if (allBlogs[k].id == newId) { allBlogs[k] = payload; break; }
          }
        } else {
          allBlogs.unshift(payload);
        }
        saveLocalStorage();
        saveBtn.disabled = false;
        saveBtn.innerHTML = isEdit ? '💾 Update Article' : '💾 Create Article';
        closeModal('blogModal');
        showToast(isEdit ? 'Article updated & saved locally!' : 'Article added & saved locally!');
        renderAllViews();
        updateStats();
      }
    }

    async function executeDelete(id) {
      var confirmBtn = document.getElementById('confirmDeleteBtn');
      confirmBtn.disabled = true;
      confirmBtn.innerText = 'Deleting...';

        try {

          showToast('Article #' + id + ' deleted from D1 Database!');
          await refreshData();
        } catch (err) {

          confirmBtn.innerText = 'Yes, Delete Article';
          alert('Network Error: ' + err.message);
        }
      } else {
        allBlogs = allBlogs.filter(function(x) { return x.id != id; });
        saveLocalStorage();
        confirmBtn.disabled = false;
        confirmBtn.innerText = 'Yes, Delete Article';
        closeModal('deleteModal');

      }
    }

    async function toggleStatus(id) {
      var blog = allBlogs.find(function(x) { return x.id == id; });
      if (!blog) return;

      var currentStatus = (String(blog.status).toLowerCase() === 'publish' || String(blog.status).toLowerCase() === 'published') ? 'publish' : 'draft';
      var newStatus = (currentStatus === 'publish') ? 'draft' : 'publish';

      if (!isD1Connected) {
        blog.status = newStatus;
        saveLocalStorage();
        renderAllViews();
        updateStats();
        showToast('Article #' + id + ' status: ' + newStatus + ' (Saved locally)');
        return;
      }

      // Optimistic update
      blog.status = newStatus;
      renderAllViews();
      updateStats();

      try {
        var res = await fetch('/api/blogs/toggle', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ id: id, status: newStatus })
        });
        var data = await res.json();
        if (!data.success) {
          blog.status = currentStatus;
          renderAllViews();
          updateStats();
          alert('Toggle failed in D1: ' + (data.error || 'Unknown error'));
        } else {
          showToast('Article #' + id + ' status updated to ' + newStatus + ' in D1');
        }
      } catch (err) {
        blog.status = currentStatus;
        renderAllViews();
        updateStats();
        alert('Network error on toggle: ' + err.message);
      }
    }

    async function refreshData() {
      if (!isD1Connected) {
        try {
          var saved = localStorage.getItem('cbtrank_blogs_store');
          if (saved) {
            var parsed = JSON.parse(saved);
            if (Array.isArray(parsed)) allBlogs = parsed;
          }
        } catch (e) {}
        renderAllViews();
        updateStats();
        return;
      }
      try {
        var res = await fetch('/api/blogs');
        var json = await res.json();
        if (json.success && Array.isArray(json.data)) {
          allBlogs = json.data;
          renderAllViews();
          updateStats();
          showToast('Articles refreshed from D1 (' + allBlogs.length + ' articles)');
        }
      } catch (e) {
        console.error('Refresh error:', e);
      }

    async function seedDatabase() {
      if (!confirm('Seed the 5 default CBT Rank blog articles into Cloudflare D1?')) return;
      if (!isD1Connected) {
        try { localStorage.removeItem('cbtrank_blogs_store'); } catch (e) {}
        showToast('Reset to default 5 CBT articles!');
        location.reload();
        return;
      }
      try {
        showToast('Seeding 5 default articles into D1...', 'info');
        var res = await fetch('/api/seed-blogs', { method: 'POST' });
        var data = await res.json();
        if (data.success) {
          showToast(data.message || 'Successfully seeded articles!');
          await refreshData();
        } else {
          alert('Seed Notice: ' + (data.error || 'Could not seed database'));
        }
      } catch (err) {
        alert('Network Error: ' + err.message);
      }
    }

    function updateActiveFilterPill() {
      var pills = document.querySelectorAll('.filter-pill');
      pills.forEach(function(p) {
        var text = p.innerText.trim();
        if (currentStatusFilter === 'publish' && text === 'Published') {
          p.classList.add('active');
        } else if (currentStatusFilter === 'draft' && text === 'Drafts') {
          p.classList.add('active');
        } else if (currentStatusFilter === 'all' && currentCategoryFilter === 'All' && text === 'All Categories') {
          p.classList.add('active');
        } else if (currentStatusFilter === 'all' && text === currentCategoryFilter) {
          p.classList.add('active');
        } else {
          p.classList.remove('active');
        }
      });

    function getFilteredBlogs() {
      return allBlogs.filter(function(b) {
        // Status filter
        if (currentStatusFilter === 'publish') {
          var st = String(b.status || '').toLowerCase();
          if (st !== 'publish' && st !== 'published') return false;
        } else if (currentStatusFilter === 'draft') {
          if (String(b.status || '').toLowerCase() !== 'draft') return false;
        }

        // Category filter
        if (currentCategoryFilter !== 'All') {
          if (b.category !== currentCategoryFilter) return false;
        }

        // Search query
        if (searchQuery) {
          var titleMatch = b.title && b.title.toLowerCase().includes(searchQuery);
          var slugMatch = b.slug && b.slug.toLowerCase().includes(searchQuery);
          var catMatch = b.category && b.category.toLowerCase().includes(searchQuery);
          var authorMatch = (b.author || b.author_name) && (b.author || b.author_name).toLowerCase().includes(searchQuery);
          var excerptMatch = b.excerpt && b.excerpt.toLowerCase().includes(searchQuery);
          if (!titleMatch && !slugMatch && !catMatch && !authorMatch && !excerptMatch) {
            return false;
          }
        }
        return true;
      });
    }
