/**
 * Self-Healing Auto-Migration:
 * Ensures sub_type, custom_data exist in user_ranks and custom_fields exists in exams
 */
async function ensureDynamicSchema(env) {
  if (isSchemaMigrated) return;
  if (!env) return;

  // 1. Native D1 Binding
  if (env.DB && typeof env.DB.prepare === 'function') {
    try {
      try { await env.DB.prepare("ALTER TABLE user_ranks ADD COLUMN sub_type TEXT DEFAULT ''").run(); } catch (e) {}
      try { await env.DB.prepare("ALTER TABLE user_ranks ADD COLUMN custom_data TEXT DEFAULT '{}'").run(); } catch (e) {}
      try { await env.DB.prepare("ALTER TABLE exams ADD COLUMN custom_fields TEXT DEFAULT '[]'").run(); } catch (e) {}
      try { await env.DB.prepare("ALTER TABLE blogs ADD COLUMN views INTEGER DEFAULT 0").run(); } catch (e) {}
      isSchemaMigrated = true;
      return;
    } catch (err) {}
  }

  // 2. Cloudflare D1 REST API Fallback
  const accountId = (env && env.CF_ACCOUNT_ID) || DEFAULT_CF_ACCOUNT_ID;
  const databaseId = (env && env.CF_DATABASE_ID) || DEFAULT_CF_DATABASE_ID;
  const apiToken = (env && env.CF_API_TOKEN) || DEFAULT_CF_API_TOKEN;

  if (accountId && databaseId && apiToken) {
    try {
      const apiUrl = `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/query`;
      await fetch(apiUrl, {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${apiToken}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({ 
          sql: "ALTER TABLE user_ranks ADD COLUMN sub_type TEXT DEFAULT ''; ALTER TABLE user_ranks ADD COLUMN custom_data TEXT DEFAULT '{}'; ALTER TABLE exams ADD COLUMN custom_fields TEXT DEFAULT '[]'; ALTER TABLE blogs ADD COLUMN views INTEGER DEFAULT 0;" 
        })
      });
      isSchemaMigrated = true;
    } catch (e) {}
  }
}

/**
 * Helper: Query Cloudflare D1 Database (Native Binding First, REST API Fallback)
 */
async function queryD1(env, sql) {
  // 1. Native Cloudflare D1 Binding (Fastest & Most Secure)
  if (env && env.DB) {
    try {
      const res = await env.DB.prepare(sql).all();
      if (res && res.results) return res.results;
      if (res) return res;
    } catch (e) {
      console.error("D1 Native Query Error:", e);
    }
  }

  // 2. Cloudflare D1 REST API Fallback
  const accountId = (env && env.CF_ACCOUNT_ID) || DEFAULT_CF_ACCOUNT_ID;
  const databaseId = (env && env.CF_DATABASE_ID) || DEFAULT_CF_DATABASE_ID;
  const apiToken = (env && env.CF_API_TOKEN) || DEFAULT_CF_API_TOKEN;

  try {
    const apiUrl = `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/query`;
    const response = await fetch(apiUrl, {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${apiToken}`,
        "Content-Type": "application/json"
      },
      body: JSON.stringify({ sql: sql })
    });
    if (response.ok) {
      const jsonRes = await response.json();
      if (jsonRes && jsonRes.result && jsonRes.result[0]) {
        if (jsonRes.result[0].results) return jsonRes.result[0].results;
        return jsonRes.result[0];
      }
    }
  } catch (e) {
    console.error("D1 REST Fallback Error:", e);
  }

  return [];
}

/**
 * Helper: Strict SQL String Sanitization
 */
function sanitizeSql(str) {
  if (str === null || str === undefined) return '';
  return String(str)
    .replace(/'/g, "''")
    .replace(/\\/g, "\\\\")
    .replace(/\0/g, "");
}

/**
 * Helper: Verify if incoming request is from authorized Admin or Server
 */
function isAuthorizedAdmin(request, env, url) {
  const adminSecret = (env && (env.ADMIN_API_KEY || env.API_SECRET)) || DEFAULT_ADMIN_SECRET;
  const authHeader = request.headers.get("Authorization") || "";
  const apiKeyHeader = request.headers.get("x-api-key") || request.headers.get("x-admin-key") || request.headers.get("x-api-secret") || "";
  const queryKey = url.searchParams.get("admin_key") || url.searchParams.get("api_key") || url.searchParams.get("key") || "";

  if (authHeader === `Bearer ${adminSecret}` || authHeader === adminSecret) return true;
  if (apiKeyHeader === adminSecret) return true;
  if (queryKey === adminSecret) return true;
  return false;
}

          const tB = parseTs(b);
          if (tB !== tA) return tB - tA;

      try {

            TRIM(exam_date) AS exam_date, 
            TRIM(exam_time) AS exam_time, 
            TRIM(paper_language) AS paper_language,
            TRIM(sub_type) AS sub_type,
            TRIM(url) AS url,
            TRIM(user_id) AS user_id
          FROM user_ranks

            AND TRIM(exam_date) != '' 
            AND LOWER(exam_date) != 'null'
            AND exam_time IS NOT NULL

            AND LOWER(exam_time) != 'null'
          ORDER BY exam_date ASC, exam_time ASC, paper_language ASC;

        // Graceful fallback if sub_type column does not exist yet

              SELECT 
                TRIM(exam_date) AS exam_date, 
                TRIM(exam_time) AS exam_time, 
                TRIM(paper_language) AS paper_language,

                TRIM(user_id) AS user_id

                AND exam_time IS NOT NULL

                AND LOWER(exam_time) != 'null'

              if (domain) legacyUpdateClause += `, domain = '${domain}'`;
              const legacyUpdateSql = `UPDATE user_ranks SET ${legacyUpdateClause} WHERE ${whereCondition}`;
              await queryD1(env, legacyUpdateSql);
            }

            return new Response(JSON.stringify({
              success: true,
              message: "User rank record updated successfully in user_ranks",
              action: "updated",
              user_id: userId,
              updated_at: createdAt
            }), { headers: corsHeaders, status: 200 });

          } else {
            // ➕ INSERT brand new candidate record
            try {
              const insertSql = `
                INSERT INTO user_ranks 
                (user_id, url, exam_slug, paper_language, url_hash, total_marks, exam_date, exam_time, exam_id, location, gender, category, domain, sub_type, custom_data, created_at, updated_at)
                VALUES

              `;

            } catch (iErr) {
              // Legacy insert fallback if sub_type or custom_data column is missing
              const legacyInsertSql = `
                INSERT INTO user_ranks 
                (user_id, url, exam_slug, paper_language, url_hash, total_marks, exam_date, exam_time, exam_id, location, gender, category, domain, created_at, updated_at)

                ('${userId}', '${userUrl}', '${examSlug}', '${paperLang}', '${urlHash}', ${totalMarks}, '${examDate}', '${examTime}', '${examId}', '${location}', '${gender}', '${category}', '${domain}', '${createdAt}', '${createdAt}')

            return new Response(JSON.stringify({
              success: true,
              message: "User rank inserted successfully into user_ranks",
              action: "inserted",
              user_id: userId,
              created_at: createdAt
            }), { headers: corsHeaders, status: 201 });
          }

        } catch (err) {
          return new Response(JSON.stringify({ success: false, error: err.message }), { headers: corsHeaders, status: 500 });
        }
      }

      // 🚫 STRICT AUTH CHECK ON GET: Block all direct unauthorized public GET requests
      if (!isAuthorizedAdmin(request, env, url)) {
        return new Response(JSON.stringify({ 
          success: false, 
          error: "Unauthorized Access: Direct public access to user_ranks is strictly restricted. Provide valid Admin Authorization header or key."

      }

      const whereClause = whereClauses.length > 0 ? `WHERE ${whereClauses.join(' AND ')}` : '';

    // =========================================================================
    if (pathname === '/invalid_answerkey_urls' || pathname === '/api/invalid_answerkey_urls' || pathname === '/invalid_urls' || pathname === '/api/invalid_urls' || pathname === '/invalid' || pathname === '/api/invalid') {

        // Prevent duplicate invalid URL inserts
        const existingCheck = await queryD1(env, `SELECT id FROM invalid_answerkey_urls WHERE url = '${cleanUrl}' LIMIT 1`);
        if (!existingCheck || existingCheck.length === 0) {
          const insertSql = `INSERT INTO invalid_answerkey_urls (url, created_at) VALUES ('${cleanUrl}', '${createdAt}')`;

        return new Response(JSON.stringify({
          success: true,
          message: "URL processed successfully in invalid_answerkey_urls",
          table: "invalid_answerkey_urls",
          url: inputUrl,
          created_at: createdAt
        }), { headers: corsHeaders, status: 201 });
      }

      // 🚫 STRICT AUTH CHECK ON GET: Block direct unauthorized access
      if (!isAuthorizedAdmin(request, env, url)) {
        return new Response(JSON.stringify({ 
          success: false, 
          error: "Unauthorized Access: Direct public access to invalid_answerkey_urls is strictly restricted." 
        }), { headers: corsHeaders, status: 401 });
      }

      if (inputUrl) {
        const cleanUrl = sanitizeSql(inputUrl);
        const sql = `SELECT id, url, created_at FROM invalid_answerkey_urls WHERE url = '${cleanUrl}' ORDER BY id DESC LIMIT 1`;
        const results = await queryD1(env, sql);
        const exists = results && results.length > 0;
        return new Response(JSON.stringify({ 
          success: true, 
          table: "invalid_answerkey_urls", 
          exists: exists,

      const results = await queryD1(env, sql);
      return new Response(JSON.stringify({

    // 1️⃣1️⃣ ⚡ CALCULATOR API PROXY ENDPOINT (/calculator_api, /api/calculator_api)

    if (pathname === '/calculator_api' || pathname === '/api/calculator_api') {

      if (!vpsRes || !vpsRes.ok) {
