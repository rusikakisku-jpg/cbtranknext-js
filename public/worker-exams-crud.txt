// =============================================================================
// Helper: Discover Cloudflare D1 Database Binding in 'env'
// =============================================================================
function getD1Database(env) {
  if (!env) return null;
  if (env.DB && typeof env.DB.prepare === 'function') return env.DB;
  if (env.DATABASE && typeof env.DATABASE.prepare === 'function') return env.DATABASE;
  if (env.d1 && typeof env.d1.prepare === 'function') return env.d1;
  // Auto-scan all env properties to support custom variable names (e.g. cbt_db, prod_db)
  for (const key of Object.keys(env)) {
    const val = env[key];
    if (val && typeof val === 'object' && typeof val.prepare === 'function') {
      return val;
    }
  }
  return null;
}

// =============================================================================
// Helper: D1 Query Executor (Handles Parameterized Bindings & Mutations)
// =============================================================================
async function executeD1(env, sql, params = []) {
  const db = getD1Database(env);
  if (!db) {
    throw new Error("Cloudflare D1 database binding 'DB' is missing. Please bind your D1 database under Worker Settings -> Bindings -> D1 Database Bindings with variable name DB.");
  }

  const cleanParams = params.map(p => {
    if (p === undefined || p === null) return null;
    if (typeof p === 'boolean') return p ? 1 : 0;
    if (typeof p === 'number' && isNaN(p)) return 0;
    return p;
  });

  try {
    const stmt = db.prepare(sql);
    const bound = cleanParams.length > 0 ? stmt.bind(...cleanParams) : stmt;
    const isSelectOrPragma = /^\s*(SELECT|PRAGMA)/i.test(sql);

    if (isSelectOrPragma) {
      const res = await bound.all();
      return (res && res.results) ? res.results : (res || []);
    } else {
      const res = await bound.run();
      return res;
    }
  } catch (err) {
    console.error("D1 Execution Error:", err.message, "\nSQL:", sql, "\nParams:", params);
    throw err;
  }
}

// =============================================================================
// Helper: Inspect Existing Columns of 'exams' Table via PRAGMA
// =============================================================================
async function getTableColumns(env) {
  try {
    const info = await executeD1(env, "PRAGMA table_info(exams)");
    if (Array.isArray(info) && info.length > 0) {
      return info.map(c => c.name);
    }
  } catch (e) {
    // Table may not exist yet
  }
  return [];
}

// Case-insensitive column resolver with alias fallback
function resolveColumnName(availableCols, primaryName, aliases = []) {
  if (!Array.isArray(availableCols) || availableCols.length === 0) return primaryName;
  const targetNames = [primaryName, ...aliases].map(s => s.toLowerCase());
  for (const target of targetNames) {
    for (const col of availableCols) {
      if (String(col).toLowerCase() === target) {
        return col; // Returns exact case from database
      }
    }
  }
  return primaryName;
}

function hasColumn(availableCols, columnName, aliases = []) {
  if (!Array.isArray(availableCols) || availableCols.length === 0) return true;
  const targetNames = [columnName, ...aliases].map(s => s.toLowerCase());
  return availableCols.some(c => targetNames.includes(String(c).toLowerCase()));
}

// Normalize row properties from D1 (case-insensitive + aliases + types)
function normalizeExamRow(row) {
  if (!row || typeof row !== 'object') return row;
  const item = { ...row };

  const getProp = (...keys) => {
    const rowKeys = Object.keys(row);
    for (const k of keys) {
      const target = k.toLowerCase();
      for (const rk of rowKeys) {
        if (rk.toLowerCase() === target && row[rk] !== undefined && row[rk] !== null) {
          return row[rk];
        }
      }
    }
    return undefined;
  };

  const rVal = getProp('marks_right', 'mark_right', 'positive_marks', 'right_marks');
  if (rVal !== undefined && rVal !== null && rVal !== '') {
    const n = Number(rVal);
    item.marks_right = !isNaN(n) ? n : 1.0;
  } else {
    item.marks_right = 1.0;
  }

  const wVal = getProp('marks_wrong', 'mark_wrong', 'negative_marks', 'wrong_marks');
  if (wVal !== undefined && wVal !== null && wVal !== '') {
    const n = Number(wVal);
    item.marks_wrong = !isNaN(n) ? Math.abs(n) : 0.25;
  } else {
    item.marks_wrong = 0.25;
  }

  const locVal = getProp('location_id');
  if (typeof locVal === 'string') {
    try { item.location_id = JSON.parse(locVal); } catch (e) { item.location_id = [locVal]; }
  } else if (locVal !== undefined) {
    item.location_id = locVal;
  }

  const vis = getProp('is_visible');
  if (vis !== undefined) item.is_visible = (vis == 1 || vis === true || vis === 'true' || vis === '1') ? 1 : 0;
  const top = getProp('set_on_top');
  if (top !== undefined) item.set_on_top = (top == 1 || top === true || top === 'true' || top === '1') ? 1 : 0;
  const lat = getProp('is_latest');
  if (lat !== undefined) item.is_latest = (lat == 1 || lat === true || lat === 'true' || lat === '1') ? 1 : 0;

  return item;
}

// Helper: Auto-create & Self-Healing Migration for 'exams' Table
// =============================================================================
async function ensureExamsTable(env) {
  const db = getD1Database(env);
  if (!db) return [];

  const createSql = `
    CREATE TABLE IF NOT EXISTS exams (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      subtitle TEXT DEFAULT '',
      slug TEXT NOT NULL UNIQUE,
      is_latest INTEGER DEFAULT 0,
      is_visible INTEGER DEFAULT 0,
      set_on_top INTEGER DEFAULT 0,
      location_type_id TEXT DEFAULT 'States/UT',
      location_id TEXT DEFAULT '["All India"]',
      marks_right REAL DEFAULT 1.0,
      marks_wrong REAL DEFAULT 0.25,
      answerkey_login_link TEXT DEFAULT '',
      description TEXT DEFAULT '',
      external_exam_id TEXT DEFAULT '',
      youtube_link TEXT DEFAULT '',
      created_at TEXT DEFAULT CURRENT_TIMESTAMP,
      updated_at TEXT DEFAULT CURRENT_TIMESTAMP
    );
  `;
  await db.prepare(createSql).run();

  // Self-healing: verify all required columns exist, add if missing
  const cols = await getTableColumns(env);
  const requiredCols = [
    { name: 'subtitle', type: "TEXT DEFAULT ''" },
    { name: 'is_latest', type: "INTEGER DEFAULT 0" },
    { name: 'is_visible', type: "INTEGER DEFAULT 0" },
    { name: 'set_on_top', type: "INTEGER DEFAULT 0" },
    { name: 'location_type_id', type: "TEXT DEFAULT 'States/UT'" },
    { name: 'location_id', type: "TEXT DEFAULT '[\"All India\"]'" },
    { name: 'marks_right', type: "REAL DEFAULT 1.0", aliases: ['mark_right', 'positive_marks', 'right_marks'] },
    { name: 'marks_wrong', type: "REAL DEFAULT 0.25", aliases: ['mark_wrong', 'negative_marks', 'wrong_marks'] },
    { name: 'answerkey_login_link', type: "TEXT DEFAULT ''" },
    { name: 'description', type: "TEXT DEFAULT ''" },
    { name: 'external_exam_id', type: "TEXT DEFAULT ''" },
    { name: 'youtube_link', type: "TEXT DEFAULT ''" },
    { name: 'created_at', type: "TEXT DEFAULT ''" },
    { name: 'updated_at', type: "TEXT DEFAULT ''" }
  ];

  for (const col of requiredCols) {
    const colExists = cols.some(c => {
      const cLower = String(c).toLowerCase();
      if (cLower === col.name.toLowerCase()) return true;
      if (col.aliases && col.aliases.some(a => a.toLowerCase() === cLower)) return true;
      return false;
    });
    if (!colExists) {
      try {
        await db.prepare(`ALTER TABLE exams ADD COLUMN ${col.name} ${col.type}`).run();
      } catch (e) {
        // Handled
      }
    }
  }

  return await getTableColumns(env);
}

// =============================================================================
// Helper: Location Value Normalizer
// =============================================================================
function normalizeLocationId(loc) {
  if (!loc) return '["All India"]';
  if (Array.isArray(loc)) return JSON.stringify(loc);
  if (typeof loc === 'string') {
    const trimmed = loc.trim();
    if (trimmed.startsWith('[') && trimmed.endsWith(']')) {
      try {
        JSON.parse(trimmed);
        return trimmed;
      } catch (e) {}
    }
    const parts = trimmed.split(',').map(s => s.trim()).filter(Boolean);
    if (parts.length > 1) {
      return JSON.stringify(parts);
    }
    return JSON.stringify([trimmed]);
  }
  return JSON.stringify([String(loc)]);
}

// =============================================================================
// Helper: CORS & JSON Responses
// =============================================================================
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type, Authorization, x-api-key',
  'Access-Control-Max-Age': '86400'
};

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...corsHeaders,
      'Content-Type': 'application/json; charset=utf-8'
    }
  });
}

function errorResponse(message, status = 400, extra = {}) {
  return jsonResponse({ success: false, error: message, ...extra }, status);
}

// =============================================================================
// Main Worker Router & Handler
// =============================================================================
export default {
  async fetch(request, env, ctx) {
    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders, status: 204 });
    }

    const url = new URL(request.url);
    let pathname = url.pathname.replace(/\/+/g, '/');
    if (pathname.length > 1 && pathname.endsWith('/')) {
      pathname = pathname.slice(0, -1);
    }

    const dbInstance = getD1Database(env);
    const isD1Bound = !!dbInstance;

    // -------------------------------------------------------------------------
    // ROUTE 0: Seed / Table Setup Notice (/api/seed-exams)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/seed-exams' || pathname === '/api/exams/seed') && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound. Please bind D1 in Cloudflare Settings.", 500);
      try {
        await ensureExamsTable(env);
        const countRes = await executeD1(env, "SELECT COUNT(*) as total FROM exams");
        return jsonResponse({
          success: true,
          totalExams: countRes[0]?.total ?? 0,
          message: "Database 'exams' table is active and 100% dynamic. You can add or manage exams via '+ Add New Exam'.",
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Notice: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 1: Schema Diagnostics & Setup
    // -------------------------------------------------------------------------
    if (pathname === '/api/schema' || pathname === '/api/diagnostics') {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500, { isD1Bound: false });
      try {
        const columns = await getTableColumns(env);
        const countRes = columns.length > 0 ? await executeD1(env, "SELECT COUNT(*) as total FROM exams") : [{ total: 0 }];
        return jsonResponse({
          success: true,
          isD1Bound: true,
          tableExists: columns.length > 0,
          totalExams: countRes[0]?.total ?? 0,
          columns: columns
        });
      } catch (err) {
        return errorResponse(err.message, 500);
      }
    }

    if (pathname === '/api/setup-table' && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500);
      try {
        await ensureExamsTable(env);
        const columns = await getTableColumns(env);
        return jsonResponse({
          success: true,
          message: "Table 'exams' verified / created successfully.",
          columns: columns
        });
      } catch (err) {
        return errorResponse("Failed to setup table: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 2: Quick Status Toggle (is_visible, is_latest, set_on_top)
    // -------------------------------------------------------------------------
    if (pathname === '/api/exams/toggle' && request.method === 'POST') {
      if (!isD1Bound) {
        return errorResponse("Cloudflare D1 database binding 'DB' is missing.", 500, { isD1Bound: false });
      }
      try {
        await ensureExamsTable(env);
        const body = await request.json();
        const id = parseInt(body.id, 10);
        const field = String(body.field || '').trim();
        const allowedFields = ['is_visible', 'is_latest', 'set_on_top'];

        if (isNaN(id) || !allowedFields.includes(field)) {
          return errorResponse("Invalid id or toggle field. Allowed: " + allowedFields.join(', '));
        }

        const currentExam = await executeD1(env, `SELECT ${field} FROM exams WHERE id = ?`, [id]);
        if (!currentExam || currentExam.length === 0) {
          return errorResponse("Exam not found with id: " + id, 404);
        }

        const currentValue = currentExam[0][field] ? 1 : 0;
        const newValue = (body.value !== undefined) ? (body.value ? 1 : 0) : (currentValue ? 0 : 1);

        const availableCols = await getTableColumns(env);
        const nowIso = new Date().toISOString();

        // EXCLUSIVE TOP RULE:
        // Only 1 exam can be on top. Setting set_on_top = 1 automatically removes top
        // from all other exams AND forces this exam to be Visible = 1 and Latest = 1.
        if (field === 'set_on_top' && newValue === 1) {
          await executeD1(env, "UPDATE exams SET set_on_top = 0 WHERE id != ?", [id]);

          let updateSql = "UPDATE exams SET set_on_top = 1";
          let updateParams = [];

          if (hasColumn(availableCols, 'is_visible')) {
            updateSql += ", " + resolveColumnName(availableCols, 'is_visible') + " = 1";
          }
          if (hasColumn(availableCols, 'is_latest')) {
            updateSql += ", " + resolveColumnName(availableCols, 'is_latest') + " = 1";
          }
          if (hasColumn(availableCols, 'updated_at')) {
            updateSql += ", " + resolveColumnName(availableCols, 'updated_at') + " = ?";
            updateParams.push(nowIso);
          }
          updateSql += " WHERE id = ?";
          updateParams.push(id);

          await executeD1(env, updateSql, updateParams);

          return jsonResponse({
            success: true,
            message: `Exam #${id} set as sole Top exam (auto-marked Visible & Latest)`,
            id: id,
            field: field,
            newValue: 1,
            is_visible: 1,
            is_latest: 1,
            isD1Bound: true
          });
        }

        let updateSql = `UPDATE exams SET ${field} = ?`;
        let updateParams = [newValue];
        if (hasColumn(availableCols, 'updated_at')) {
          updateSql += ", " + resolveColumnName(availableCols, 'updated_at') + " = ?";
          updateParams.push(nowIso);
        }
        updateSql += " WHERE id = ?";
        updateParams.push(id);

        await executeD1(env, updateSql, updateParams);

        return jsonResponse({
          success: true,
          message: `Field '${field}' updated to ${newValue} in D1 database`,
          id: id,
          field: field,
          newValue: newValue,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Toggle failed: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 3: Add New Exam (POST /api/exams or POST /api/exams/add)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/exams' || pathname === '/api/exams/add') && request.method === 'POST') {
      if (!isD1Bound) {
        return errorResponse("Cloudflare D1 database binding 'DB' is missing. Please bind your D1 database under Worker Settings -> Bindings -> D1 Database Bindings with variable name DB.", 500, { isD1Bound: false });
      }
      try {
        await ensureExamsTable(env);
        const body = await request.json();
        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();

        if (!title) return errorResponse("Field 'title' is required.");
        if (!slug) {
          slug = title
            .toLowerCase()
            .replace(/[^a-z0-9]+/g, '-')
            .replace(/^-+|-+$/g, '');
        }

        const existing = await executeD1(env, "SELECT id FROM exams WHERE slug = ? LIMIT 1", [slug]);
        if (existing && existing.length > 0) {
          return errorResponse(`An exam with slug '${slug}' already exists (ID: #${existing[0].id}). Please choose a unique slug.`, 400, { isD1Bound: true });
        }

        const subtitle = String(body.subtitle || '').trim();
        let setOnTop = body.set_on_top ? 1 : 0;
        let isLatest = (body.is_latest !== undefined && body.is_latest !== null) ? (body.is_latest ? 1 : 0) : 0;
        let isVisible = (body.is_visible !== undefined && body.is_visible !== null) ? (body.is_visible ? 1 : 0) : 0;

        // EXCLUSIVE TOP RULE:
        // If this new exam is set on top, it MUST be visible and latest, and all other exams lose top status.
        if (setOnTop === 1) {
          isVisible = 1;
          isLatest = 1;
          await executeD1(env, "UPDATE exams SET set_on_top = 0");
        }
        const locTypeId = String(body.location_type_id || 'States/UT').trim();
        const locId = normalizeLocationId(body.location_id);
        let marksRight = 1.0;
        if (body.marks_right !== undefined && body.marks_right !== null && body.marks_right !== '') {
          const parsedR = parseFloat(body.marks_right);
          if (!isNaN(parsedR)) marksRight = parsedR;
        }

        let marksWrong = 0.25;
        if (body.marks_wrong !== undefined && body.marks_wrong !== null && body.marks_wrong !== '') {
          const parsedW = parseFloat(body.marks_wrong);
          if (!isNaN(parsedW)) marksWrong = Math.abs(parsedW);
        }

        const answerLink = String(body.answerkey_login_link || '').trim();
        const description = String(body.description || '').trim();
        const externalExamId = String(body.external_exam_id || '').trim();
        const youtubeLink = String(body.youtube_link || '').trim();

        const availableCols = await getTableColumns(env);
        const nowIso = new Date().toISOString();

        // 1. Determine next unique ID safely to prevent 'NOT NULL constraint failed: exams.id'
        let nextId = 1;
        try {
          const maxRes = await executeD1(env, "SELECT COALESCE(MAX(id), 0) AS max_id FROM exams");
          if (maxRes && maxRes.length > 0 && maxRes[0].max_id !== undefined && maxRes[0].max_id !== null) {
            nextId = parseInt(maxRes[0].max_id, 10) + 1;
            if (isNaN(nextId) || nextId < 1) nextId = 1;
          }
        } catch (e) {
          nextId = Date.now();
        }

        const rightColName = resolveColumnName(availableCols, 'marks_right', ['mark_right', 'positive_marks', 'right_marks']);
        const wrongColName = resolveColumnName(availableCols, 'marks_wrong', ['mark_wrong', 'negative_marks', 'wrong_marks']);

        const dataMap = {
          title, subtitle, slug, is_latest: isLatest, is_visible: isVisible, set_on_top: setOnTop,
          location_type_id: locTypeId, location_id: locId,
          [rightColName]: marksRight,
          [wrongColName]: marksWrong,
          answerkey_login_link: answerLink, description, external_exam_id: externalExamId, youtube_link: youtubeLink
        };

        // Explicitly include id as first column to satisfy NOT NULL constraints across all D1 schema variants
        let colsToInsert = ['id'];
        let placeholders = ['?'];
        let valuesToInsert = [nextId];

        Object.keys(dataMap).forEach(col => {
          if (availableCols.length === 0 || hasColumn(availableCols, col)) {
            const actualCol = resolveColumnName(availableCols, col);
            colsToInsert.push(actualCol);
            placeholders.push('?');
            valuesToInsert.push(dataMap[col]);
          }
        });

        if (availableCols.includes('created_at')) {
          colsToInsert.push('created_at');
          placeholders.push('?');
          valuesToInsert.push(nowIso);
        }
        if (availableCols.includes('updated_at')) {
          colsToInsert.push('updated_at');
          placeholders.push('?');
          valuesToInsert.push(nowIso);
        }

        const insertSql = `INSERT INTO exams (${colsToInsert.join(', ')}) VALUES (${placeholders.join(', ')})`;
        const result = await executeD1(env, insertSql, valuesToInsert);

        // Sync autoincrement sequence if sqlite_sequence exists
        try {
          const dbInstance = getD1Database(env);
          if (dbInstance) {
            await dbInstance.prepare("INSERT OR REPLACE INTO sqlite_sequence (name, seq) VALUES ('exams', MAX(COALESCE((SELECT seq FROM sqlite_sequence WHERE name = 'exams'), 0), ?))").bind(nextId).run();
          }
        } catch (e) {}

        // Retrieve real assigned ID directly from D1 to guarantee persistence
        let newId = nextId;
        try {
          const insertedRow = await executeD1(env, "SELECT id FROM exams WHERE slug = ? LIMIT 1", [slug]);
          if (insertedRow && insertedRow.length > 0) {
            newId = insertedRow[0].id;
          }
        } catch (e) {}

        return jsonResponse({
          success: true,
          message: "Exam added successfully to D1 database!",
          id: newId,
          slug: slug,
          isD1Bound: true
        }, 201);
      } catch (err) {
        return errorResponse("Failed to add exam: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 4: Update Exam (PUT /api/exams/:id or POST /api/exams/update)
    // -------------------------------------------------------------------------
    const updateMatch = pathname.match(/^\/api\/exams\/(\d+)$/);
    if ((updateMatch && (request.method === 'PUT' || request.method === 'POST')) || (pathname === '/api/exams/update' && request.method === 'POST')) {
      if (!isD1Bound) {
        return errorResponse("Cloudflare D1 database binding 'DB' is missing.", 500, { isD1Bound: false });
      }
      try {
        await ensureExamsTable(env);
        const body = await request.json();
        const examId = updateMatch ? parseInt(updateMatch[1], 10) : parseInt(body.id, 10);
        if (isNaN(examId)) return errorResponse("Invalid or missing Exam ID for update.");

        const existing = await executeD1(env, "SELECT id, slug FROM exams WHERE id = ?", [examId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No exam found with ID: #${examId}`, 404);
        }

        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();
        if (!title) return errorResponse("Title cannot be empty.");
        if (!slug) {
          slug = title.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
        }

        const slugCheck = await executeD1(env, "SELECT id FROM exams WHERE slug = ? AND id != ? LIMIT 1", [slug, examId]);
        if (slugCheck && slugCheck.length > 0) {
          return errorResponse(`Slug '${slug}' is already in use by another exam (ID: #${slugCheck[0].id}).`);
        }

        const subtitle = String(body.subtitle || '').trim();
        let setOnTop = body.set_on_top ? 1 : 0;
        let isLatest = (body.is_latest !== undefined && body.is_latest !== null) ? (body.is_latest ? 1 : 0) : 1;
        let isVisible = (body.is_visible !== undefined && body.is_visible !== null) ? (body.is_visible ? 1 : 0) : 1;

        // EXCLUSIVE TOP RULE:
        // If this exam is set on top, it MUST be visible and latest, and all other exams lose top status.
        if (setOnTop === 1) {
          isVisible = 1;
          isLatest = 1;
          await executeD1(env, "UPDATE exams SET set_on_top = 0 WHERE id != ?", [examId]);
        }
        const locTypeId = String(body.location_type_id || 'States/UT').trim();
        const locId = normalizeLocationId(body.location_id);
        let marksRight = 1.0;
        if (body.marks_right !== undefined && body.marks_right !== null && body.marks_right !== '') {
          const parsedR = parseFloat(body.marks_right);
          if (!isNaN(parsedR)) marksRight = parsedR;
        }

        let marksWrong = 0.25;
        if (body.marks_wrong !== undefined && body.marks_wrong !== null && body.marks_wrong !== '') {
          const parsedW = parseFloat(body.marks_wrong);
          if (!isNaN(parsedW)) marksWrong = Math.abs(parsedW);
        }

        const answerLink = String(body.answerkey_login_link || '').trim();
        const description = String(body.description || '').trim();
        const externalExamId = String(body.external_exam_id || '').trim();
        const youtubeLink = String(body.youtube_link || '').trim();

        const availableCols = await getTableColumns(env);
        const rightColName = resolveColumnName(availableCols, 'marks_right', ['mark_right', 'positive_marks', 'right_marks']);
        const wrongColName = resolveColumnName(availableCols, 'marks_wrong', ['mark_wrong', 'negative_marks', 'wrong_marks']);

        const dataMap = {
          title, subtitle, slug, is_latest: isLatest, is_visible: isVisible, set_on_top: setOnTop,
          location_type_id: locTypeId, location_id: locId,
          [rightColName]: marksRight,
          [wrongColName]: marksWrong,
          answerkey_login_link: answerLink, description, external_exam_id: externalExamId, youtube_link: youtubeLink
        };

        let setClauses = [];
        let setValues = [];

        Object.keys(dataMap).forEach(col => {
          if (availableCols.length === 0 || hasColumn(availableCols, col)) {
            const actualCol = resolveColumnName(availableCols, col);
            setClauses.push(`${actualCol} = ?`);
            setValues.push(dataMap[col]);
          }
        });

        if (availableCols.length === 0 || hasColumn(availableCols, 'updated_at')) {
          const actualCol = resolveColumnName(availableCols, 'updated_at');
          setClauses.push(`${actualCol} = ?`);
          setValues.push(new Date().toISOString());
        }

        setValues.push(examId);
        const updateSql = `UPDATE exams SET ${setClauses.join(', ')} WHERE id = ?`;
        await executeD1(env, updateSql, setValues);

        return jsonResponse({
          success: true,
          message: "Exam updated successfully in D1 database!",
          id: examId,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to update exam: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 5: Delete Exam (DELETE /api/exams/:id or POST /api/exams/delete)
    // -------------------------------------------------------------------------
    const deleteMatch = pathname.match(/^\/api\/exams\/(\d+)$/);
    if ((deleteMatch && request.method === 'DELETE') || (pathname === '/api/exams/delete' && (request.method === 'POST' || request.method === 'DELETE'))) {
      if (!isD1Bound) {
        return errorResponse("Cloudflare D1 database binding 'DB' is missing.", 500, { isD1Bound: false });
      }
      try {
        await ensureExamsTable(env);
        let examId;
        if (deleteMatch) {
          examId = parseInt(deleteMatch[1], 10);
        } else {
          const body = await request.json().catch(() => ({}));
          examId = parseInt(body.id || url.searchParams.get('id'), 10);
        }

        if (isNaN(examId)) return errorResponse("Invalid or missing Exam ID for deletion.");

        const existing = await executeD1(env, "SELECT id, title FROM exams WHERE id = ?", [examId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No exam found with ID: #${examId}`, 404);
        }

        await executeD1(env, "DELETE FROM exams WHERE id = ?", [examId]);

        return jsonResponse({
          success: true,
          message: `Exam #${examId} ('${existing[0].title}') deleted successfully from D1 database.`,
          id: examId,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Failed to delete exam: " + err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 6: Get Single Exam Details (GET /api/exams/:id or :slug)
    // -------------------------------------------------------------------------
    const getSingleMatch = pathname.match(/^\/api\/exams\/([a-zA-Z0-9_-]+)$/);
    if (getSingleMatch && request.method === 'GET') {
      const identifier = getSingleMatch[1];
      if (!isD1Bound) {
        return errorResponse("Cloudflare D1 database binding 'DB' is missing.", 500, { isD1Bound: false });
      }
      try {
        const identifier = getSingleMatch[1];
        let sql = "SELECT * FROM exams WHERE slug = ? LIMIT 1";
        let params = [identifier];
        if (/^\d+$/.test(identifier)) {
          sql = "SELECT * FROM exams WHERE id = ? OR slug = ? LIMIT 1";
          params = [parseInt(identifier, 10), identifier];
        }
        const rows = await executeD1(env, sql, params);
        if (rows && rows.length > 0) {
          const item = rows[0];
          return jsonResponse({ success: true, data: normalizeExamRow(item) });
        }
        return errorResponse("Exam not found", 404);
      } catch (err) {
        return errorResponse(err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 7: List All Exams (GET /api/exams or /exams)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/exams' || pathname === '/exams' || pathname === '/get_exams') && request.method === 'GET') {
      const search = url.searchParams.get('search') ? url.searchParams.get('search').trim() : '';
      const visibleOnly = url.searchParams.get('visible_only') === '1' || url.searchParams.get('active') === '1';
      const rawArray = url.searchParams.get('raw') === '1' || url.searchParams.get('format') === 'array';

      if (!isD1Bound) {
        return errorResponse("Cloudflare D1 database binding 'DB' is missing. Please bind your D1 database under Worker Settings -> Bindings -> D1 Database Bindings with variable name DB.", 500, { isD1Bound: false });
      }
      try {
        await ensureExamsTable(env);
        let sql = "SELECT * FROM exams";
        let wheres = [];
        let params = [];

        if (visibleOnly) {
          wheres.push("is_visible = 1");
        }
        if (search) {
          wheres.push("(title LIKE ? OR slug LIKE ? OR external_exam_id LIKE ?)");
          const term = `%${search}%`;
          params.push(term, term, term);
        }

        if (wheres.length > 0) {
          sql += " WHERE " + wheres.join(" AND ");
        }
        sql += " ORDER BY set_on_top DESC, id DESC";

        let rawRows = [];
        try {
          rawRows = await executeD1(env, sql, params);
        } catch (err) {
          console.warn("API query notice:", err.message);
        }

        const rows = Array.isArray(rawRows) ? rawRows.map(normalizeExamRow) : [];

        if (rawArray) {
          return new Response(JSON.stringify(rows), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' },
            status: 200
          });
        }

        return jsonResponse({
          success: true,
          count: rows.length,
          data: rows,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse(err.message, 500, { isD1Bound: true });
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 8: Web Dashboard Portal (GET / or browser requests)
    // -------------------------------------------------------------------------
    if (request.method === 'GET') {
      let initialExams = [];
      let tableCols = [];
      let dbError = null;

      if (isD1Bound) {
        try {
          // 1. Auto-create table & migrate missing columns
          tableCols = await ensureExamsTable(env);
          const hasSetOnTop = hasColumn(tableCols, 'set_on_top');
          const orderSql = hasSetOnTop ? "ORDER BY set_on_top DESC, id DESC" : "ORDER BY id DESC";

          // 2. Fetch real exams directly from D1
          const rawRows = await executeD1(env, "SELECT * FROM exams " + orderSql);
          if (Array.isArray(rawRows)) {
            initialExams = rawRows.map(normalizeExamRow);
          }
        } catch (e) {
          dbError = e.message;
          console.error("D1 Load Error:", e);
        }
      }

      return renderExamsManagerPortal(isD1Bound, tableCols.length > 0, initialExams, dbError);
    }

    return errorResponse("Method or endpoint not found", 404);
  }
};

    // Safe Truthy check for SQLite boolean/integer/string values
    function isTruthy(val, defaultVal) {
      if (val === undefined || val === null) return !!defaultVal;
      if (typeof val === 'boolean') return val;
      if (typeof val === 'number') return val === 1;
      if (typeof val === 'string') {
        var s = val.trim().toLowerCase();
        return s === '1' || s === 'true';
      }
      return Boolean(val);
    }

    // Format location array/string cleanly for text input
    function formatLocationInput(loc) {
      if (!loc) return 'All India';
      if (Array.isArray(loc)) return loc.join(', ');
      if (typeof loc === 'string') {
        var trimmed = loc.trim();
        if (trimmed.startsWith('[') && trimmed.endsWith(']')) {
          try {
            var parsed = JSON.parse(trimmed);
            if (Array.isArray(parsed)) return parsed.join(', ');
          } catch (e) {}
        }
        return trimmed;
      }
      return String(loc);
    }

    function initTinyMce(callback) {
      if (typeof tinymce === 'undefined') {
        if (callback) callback(null);
        return;
      }
      var existing = tinymce.get('formDescription');
      if (existing) {
        if (callback) callback(existing);
        return;
      }
      try {
        tinymce.init({
          selector: '#formDescription',
          base_url: 'https://cdnjs.cloudflare.com/ajax/libs/tinymce/6.8.2',
          suffix: '.min',
          skin: 'oxide-dark',
          content_css: 'dark',
          height: 280,
          menubar: false,

          toolbar: 'undo redo | blocks | bold italic underline forecolor backcolor | bullist numlist | table link | code fullscreen',
          content_style: 'body { font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif; font-size: 14px; color: #f1f5f9; background: #0f172a; line-height: 1.6; padding: 12px; } a { color: #38bdf8; } table { border-collapse: collapse; width: 100%; } table, th, td { border: 1px solid #334155; padding: 6px 10px; }',
          branding: false,
          promotion: false,
          setup: function(editor) {
            editor.on('init', function() {
              isTinyMceReady = true;
              if (callback) callback(editor);
            });
            editor.on('change keyup', function() {
              var val = editor.getContent();
              document.getElementById('formDescription').value = val;
            });
          }
        });
      } catch (e) {
        console.warn('TinyMCE init skipped:', e);
        if (callback) callback(null);
      }

      var btn = document.getElementById('toggleEditorModeBtn');
      var txt = document.getElementById('formDescription');

      if (!isRawHtmlMode) {
        // Switch to Raw HTML Code Mode
        if (ed) {
          var currentVal = ed.getContent();
          txt.value = currentVal;
          ed.hide();
        }
        txt.style.display = 'block';
        if (btn) btn.innerHTML = '📝 Visual Mode';
        isRawHtmlMode = true;
      } else {
        // Switch to Visual TinyMCE Mode
        if (ed) {
          var currentVal = txt.value;
          ed.show();
          ed.setContent(currentVal);
        }
        if (btn) btn.innerHTML = '💻 HTML Code';
        isRawHtmlMode = false;
      }

    function slugify(text) {
      return (text || '')
        .toString()
        .toLowerCase()
        .trim()
        .replace(/\s+/g, '-')
        .replace(/[^a-z0-9\-]+/g, '')
        .replace(/\-\-+/g, '-')
        .replace(/^-+/, '')
        .replace(/-+$/, '');
    }

    async function handleFormSubmit(e) {
      e.preventDefault();
      var saveBtn = document.getElementById('saveExamBtn');
      var examId = document.getElementById('formExamId').value;
      var isEdit = !!examId;

      var title = document.getElementById('formTitle').value.trim();
      var slug = document.getElementById('formSlug').value.trim();

      if (!title) {
        alert('Please enter the Exam Title.');
        document.getElementById('formTitle').focus();
        return;
      }
      if (!slug) {
        slug = slugify(title);
        document.getElementById('formSlug').value = slug;
      }

      saveBtn.disabled = true;
      saveBtn.innerText = '⏳ Saving...';

      var locTypeSelect = document.getElementById('formLocationTypeId').value;
      var finalLocationType = locTypeSelect;
      if (locTypeSelect === 'Other') {
        var customVal = document.getElementById('formCustomLocationType').value.trim();
        if (!customVal) {
          alert('Please enter a Custom Location Type Name.');
          document.getElementById('formCustomLocationType').focus();
          saveBtn.disabled = false;
          saveBtn.innerText = isEdit ? '💾 Update Exam' : '💾 Create Exam';
          return;
        }
        finalLocationType = customVal;
      }

      var payload = {
        title: title,
        slug: slug,
        subtitle: document.getElementById('formSubtitle').value.trim(),
        marks_right: !isNaN(parseFloat(document.getElementById('formMarksRight').value)) ? parseFloat(document.getElementById('formMarksRight').value) : 1.0,
        marks_wrong: !isNaN(parseFloat(document.getElementById('formMarksWrong').value)) ? Math.abs(parseFloat(document.getElementById('formMarksWrong').value)) : 0.25,
        location_type_id: finalLocationType,
        location_id: document.getElementById('formLocationId').value.trim() || 'All India',
        external_exam_id: document.getElementById('formExternalExamId').value.trim(),
        answerkey_login_link: document.getElementById('formAnswerLink').value.trim(),
        youtube_link: document.getElementById('formYoutubeLink').value.trim(),
        description: getDescriptionContent().trim(),
        is_visible: (document.getElementById('formSetOnTop').checked || document.getElementById('formIsVisible').checked) ? 1 : 0,
        set_on_top: document.getElementById('formSetOnTop').checked ? 1 : 0,
        is_latest: (document.getElementById('formSetOnTop').checked || document.getElementById('formIsLatest').checked) ? 1 : 0
      };

      try {
        var res;
        if (isEdit) {
          payload.id = parseInt(examId, 10);
          res = await fetch('/api/exams/' + examId, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
        } else {
          res = await fetch('/api/exams', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
        }

        var data = await res.json();
        saveBtn.disabled = false;
        saveBtn.innerHTML = isEdit ? '💾 Update Exam' : '💾 Create Exam';

        if (!data.success) {
          alert('Database Error: ' + (data.error || 'Failed to save exam to D1 database'));
          return;
        }

        closeModal('examModal');
        showToast(isEdit ? 'Exam updated in D1 Database!' : 'Exam added to D1 Database!');
        await refreshData();
      } catch (err) {
        saveBtn.disabled = false;
        saveBtn.innerHTML = isEdit ? '💾 Update Exam' : '💾 Create Exam';
        alert('Network / API Error: ' + err.message);
      }
    }

    async function toggleFlag(id, field) {
      var exam = null;
      for (var i = 0; i < allExams.length; i++) {
        if (allExams[i].id == id) { exam = allExams[i]; break; }
      }
      if (!exam) return;

      var currentVal = exam[field] == 1 ? 1 : 0;
      var newVal = currentVal ? 0 : 1;

      // Backup previous state for rollback on error
      var prevSnapshots = allExams.map(function(e) {
        return { id: e.id, set_on_top: e.set_on_top, is_visible: e.is_visible, is_latest: e.is_latest };
      });

      // EXCLUSIVE TOP RULE:
      // If setting this exam on top, all other exams must be unset from top (set_on_top = 0),
      // and this exam MUST automatically become Visible (is_visible = 1) and Latest (is_latest = 1).
      if (field === 'set_on_top' && newVal === 1) {
        for (var j = 0; j < allExams.length; j++) {
          if (allExams[j].id == id) {
            allExams[j].set_on_top = 1;
            allExams[j].is_visible = 1;
            allExams[j].is_latest = 1;
          } else {
            allExams[j].set_on_top = 0;
          }
        }
      } else {
        exam[field] = newVal;
      }

      renderAllViews();
      updateStats();

      try {
        var res = await fetch('/api/exams/toggle', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ id: id, field: field, value: newVal })
        });
        var data = await res.json();
        if (!data.success) {
          // Rollback on server failure
          for (var k = 0; k < allExams.length; k++) {
            for (var p = 0; p < prevSnapshots.length; p++) {
              if (allExams[k].id == prevSnapshots[p].id) {
                allExams[k].set_on_top = prevSnapshots[p].set_on_top;
                allExams[k].is_visible = prevSnapshots[p].is_visible;
                allExams[k].is_latest = prevSnapshots[p].is_latest;
                break;
              }
            }
          }
          renderAllViews();
          updateStats();
          alert('Toggle failed in D1: ' + (data.error || 'Unknown error'));
        } else {
          if (field === 'set_on_top' && newVal === 1) {
            showToast('Exam #' + id + ' set as the sole Top exam (also Visible & Latest)!');
          } else {
            showToast('Updated ' + field + ' for Exam #' + id + ' in D1');
          }
          await refreshData();
        }
      } catch (err) {
        // Rollback on network error
        for (var k = 0; k < allExams.length; k++) {
          for (var p = 0; p < prevSnapshots.length; p++) {
            if (allExams[k].id == prevSnapshots[p].id) {
              allExams[k].set_on_top = prevSnapshots[p].set_on_top;
              allExams[k].is_visible = prevSnapshots[p].is_visible;
              allExams[k].is_latest = prevSnapshots[p].is_latest;
              break;
            }
          }
        }
        renderAllViews();
        updateStats();
        alert('Network error on toggle: ' + err.message);
      }
    }
