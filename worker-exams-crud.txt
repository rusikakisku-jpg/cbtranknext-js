// =============================================================================
// Helper: Discover Cloudflare D1 Database Binding in 'env'
// =============================================================================
function getD1Database(env) {
  if (!env) return null;
  if (env.DB && typeof env.DB.prepare === 'function') return env.DB;
  if (env.DATABASE && typeof env.DATABASE.prepare === 'function') return env.DATABASE;
  if (env.d1 && typeof env.d1.prepare === 'function') return env.d1;
  // Auto-scan all env properties to support custom variable names (e.g. cbt_db, prod_db)
  for (const key of Object.keys(env)) {
    const val = env[key];
    if (val && typeof val === 'object' && typeof val.prepare === 'function') {
      return val;
    }
  }
  return null;
}

// =============================================================================
// Helper: D1 Query Executor (Handles Parameterized Bindings & Mutations)
// =============================================================================
async function executeD1(env, sql, params = []) {
  const db = getD1Database(env);
  if (!db) {
    throw new Error("Cloudflare D1 database binding 'DB' is missing. Please bind your D1 database under Worker Settings -> Bindings -> D1 Database Bindings with variable name DB.");
  }

  const cleanParams = params.map(p => {
    if (p === undefined || p === null) return null;
    if (typeof p === 'boolean') return p ? 1 : 0;
    return p;
  });

  try {
    const stmt = db.prepare(sql);
    const bound = cleanParams.length > 0 ? stmt.bind(...cleanParams) : stmt;
    const isSelectOrPragma = /^\s*(SELECT|PRAGMA)/i.test(sql);

    if (isSelectOrPragma) {
      const res = await bound.all();
      return (res && res.results) ? res.results : (res || []);
    } else {
      const res = await bound.run();
      return res;
    }
  } catch (err) {
    console.error("D1 Execution Error:", err.message, "\nSQL:", sql, "\nParams:", params);
    throw err;
  }
}

// =============================================================================
// Helper: Inspect Existing Columns of 'exams' Table via PRAGMA
// =============================================================================
async function getTableColumns(env) {
  try {
    const info = await executeD1(env, "PRAGMA table_info(exams)");
    if (Array.isArray(info) && info.length > 0) {
      return info.map(c => c.name);
    }
  } catch (e) {
    // Table may not exist yet
  }
  return [];
}

// =============================================================================
// Helper: Auto-create & Self-Healing Migration for 'exams' Table
// =============================================================================
async function ensureExamsTable(env) {
  const db = getD1Database(env);
  if (!db) return [];

  const createSql = `
    CREATE TABLE IF NOT EXISTS exams (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      subtitle TEXT DEFAULT '',
      slug TEXT NOT NULL UNIQUE,
      is_latest INTEGER DEFAULT 1,
      is_visible INTEGER DEFAULT 1,
      set_on_top INTEGER DEFAULT 0,
      location_type_id TEXT DEFAULT 'States/UT',
      location_id TEXT DEFAULT '["All India"]',
      marks_right REAL DEFAULT 1.0,
      marks_wrong REAL DEFAULT 0.25,
      answerkey_login_link TEXT DEFAULT '',
      description TEXT DEFAULT '',
      external_exam_id TEXT DEFAULT '',
      youtube_link TEXT DEFAULT '',
      created_at TEXT DEFAULT CURRENT_TIMESTAMP,
      updated_at TEXT DEFAULT CURRENT_TIMESTAMP
    );
  `;
  await db.prepare(createSql).run();

  // Self-healing: verify all required columns exist, add if missing
  const cols = await getTableColumns(env);
  const requiredCols = [
    { name: 'subtitle', type: "TEXT DEFAULT ''" },
    { name: 'is_latest', type: "INTEGER DEFAULT 1" },
    { name: 'is_visible', type: "INTEGER DEFAULT 1" },
    { name: 'set_on_top', type: "INTEGER DEFAULT 0" },
    { name: 'location_type_id', type: "TEXT DEFAULT 'States/UT'" },
    { name: 'location_id', type: "TEXT DEFAULT '[\"All India\"]'" },
    { name: 'marks_right', type: "REAL DEFAULT 1.0" },
    { name: 'marks_wrong', type: "REAL DEFAULT 0.25" },
    { name: 'answerkey_login_link', type: "TEXT DEFAULT ''" },
    { name: 'description', type: "TEXT DEFAULT ''" },
    { name: 'external_exam_id', type: "TEXT DEFAULT ''" },
    { name: 'youtube_link', type: "TEXT DEFAULT ''" },
    { name: 'created_at', type: "TEXT DEFAULT CURRENT_TIMESTAMP" },
    { name: 'updated_at', type: "TEXT DEFAULT CURRENT_TIMESTAMP" }
  ];

  for (const col of requiredCols) {
    if (!cols.includes(col.name)) {
      try {
        await db.prepare(`ALTER TABLE exams ADD COLUMN ${col.name} ${col.type}`).run();
      } catch (e) {
        // Handled
      }
    }
  }

  return await getTableColumns(env);
}

// =============================================================================
// Helper: Seed Default 27 CBT Exams into D1 Table
// =============================================================================
async function seedDefaultExams(env) {
  const db = getD1Database(env);
  if (!db) return 0;
  await ensureExamsTable(env);

  let insertedCount = 0;
  for (const ex of DEFAULT_EXAMS) {
    try {
      const locStr = Array.isArray(ex.location_id) ? JSON.stringify(ex.location_id) : JSON.stringify([ex.location_id || 'All India']);
      await db.prepare(`
        INSERT OR IGNORE INTO exams (id, title, subtitle, slug, is_latest, is_visible, set_on_top, location_type_id, location_id, marks_right, marks_wrong, answerkey_login_link, description, external_exam_id, youtube_link)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).bind(
        ex.id, ex.title, ex.subtitle || '', ex.slug,
        ex.is_latest ? 1 : 0, ex.is_visible ? 1 : 0, ex.set_on_top ? 1 : 0,
        ex.location_type_id || 'States/UT', locStr,
        ex.marks_right || 1.0, ex.marks_wrong || 0.25,
        ex.answerkey_login_link || '', ex.description || '', ex.external_exam_id || '', ex.youtube_link || ''
      ).run();
      insertedCount++;
    } catch (err) {
      console.warn("Seed row notice:", ex.slug, err.message);
    }
  }
  return insertedCount;
}

// Helper: Location Value Normalizer
// =============================================================================
function normalizeLocationId(loc) {
  if (!loc) return '["All India"]';
  if (Array.isArray(loc)) return JSON.stringify(loc);
  if (typeof loc === 'string') {
    const trimmed = loc.trim();
    if (trimmed.startsWith('[') && trimmed.endsWith(']')) {
      try {
        JSON.parse(trimmed);
        return trimmed;
      } catch (e) {}
    }
    const parts = trimmed.split(',').map(s => s.trim()).filter(Boolean);
    if (parts.length > 1) {
      return JSON.stringify(parts);
    }
    return JSON.stringify([trimmed]);
  }
  return JSON.stringify([String(loc)]);
}

// =============================================================================
// Helper: CORS & JSON Responses
// =============================================================================
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type, Authorization, x-api-key',
  'Access-Control-Max-Age': '86400'
};

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...corsHeaders,
      'Content-Type': 'application/json; charset=utf-8'
    }
  });
}

function errorResponse(message, status = 400, extra = {}) {
  return jsonResponse({ success: false, error: message, ...extra }, status);
}

// =============================================================================
// Main Worker Router & Handler
// =============================================================================
export default {
  async fetch(request, env, ctx) {
    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders, status: 204 });
    }

    const url = new URL(request.url);
    let pathname = url.pathname.replace(/\/+/g, '/');
    if (pathname.length > 1 && pathname.endsWith('/')) {
      pathname = pathname.slice(0, -1);
    }

    const dbInstance = getD1Database(env);
    const isD1Bound = !!dbInstance;

    // -------------------------------------------------------------------------
    // ROUTE 0: Seed / Reset Default 27 CBT Exams (/api/seed-exams)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/seed-exams' || pathname === '/api/exams/seed') && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) {
        return jsonResponse({
          success: true,
          seeded: DEFAULT_EXAMS.length,
          message: "Running in preview mode (D1 not bound). Showing 27 built-in default exams.",
          isD1Bound: false
        });
      }
      try {
        const count = await seedDefaultExams(env);
        return jsonResponse({
          success: true,
          seeded: count,
          message: `Successfully seeded/verified ${count} default CBT exams into D1 database.`,
          isD1Bound: true
        });
      } catch (err) {
        return errorResponse("Seeding failed: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 1: Schema Diagnostics & Setup
    // -------------------------------------------------------------------------
    if (pathname === '/api/schema' || pathname === '/api/diagnostics') {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500, { isD1Bound: false });
      try {
        const columns = await getTableColumns(env);
        const countRes = columns.length > 0 ? await executeD1(env, "SELECT COUNT(*) as total FROM exams") : [{ total: 0 }];
        return jsonResponse({
          success: true,
          isD1Bound: true,
          tableExists: columns.length > 0,
          totalExams: countRes[0]?.total ?? 0,
          columns: columns
        });
      } catch (err) {
        return errorResponse(err.message, 500);
      }
    }

    if (pathname === '/api/setup-table' && (request.method === 'POST' || request.method === 'GET')) {
      if (!isD1Bound) return errorResponse("D1 database 'DB' is not bound.", 500);
      try {
        await ensureExamsTable(env);
        const columns = await getTableColumns(env);
        return jsonResponse({
          success: true,
          message: "Table 'exams' verified / created successfully.",
          columns: columns
        });
      } catch (err) {
        return errorResponse("Failed to setup table: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 2: Quick Status Toggle (is_visible, is_latest, set_on_top)
    // -------------------------------------------------------------------------
    if (pathname === '/api/exams/toggle' && request.method === 'POST') {
      if (!isD1Bound) return errorResponse("D1 Database not bound", 500);
      try {
        const body = await request.json();
        const id = parseInt(body.id, 10);
        const field = String(body.field || '').trim();
        const allowedFields = ['is_visible', 'is_latest', 'set_on_top'];

        if (isNaN(id) || !allowedFields.includes(field)) {
          return errorResponse("Invalid id or toggle field. Allowed: " + allowedFields.join(', '));
        }

        const currentExam = await executeD1(env, `SELECT ${field} FROM exams WHERE id = ?`, [id]);
        if (!currentExam || currentExam.length === 0) {
          return errorResponse("Exam not found with id: " + id, 404);
        }

        const currentValue = currentExam[0][field] ? 1 : 0;
        const newValue = (body.value !== undefined) ? (body.value ? 1 : 0) : (currentValue ? 0 : 1);

        const availableCols = await getTableColumns(env);
        let updateSql = `UPDATE exams SET ${field} = ?`;
        if (availableCols.includes('updated_at')) {
          updateSql += ", updated_at = datetime('now')";
        }
        updateSql += " WHERE id = ?";

        await executeD1(env, updateSql, [newValue, id]);

        return jsonResponse({
          success: true,
          message: `Field '${field}' updated to ${newValue}`,
          id: id,
          field: field,
          newValue: newValue
        });
      } catch (err) {
        return errorResponse("Toggle failed: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 3: Add New Exam (POST /api/exams or POST /api/exams/add)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/exams' || pathname === '/api/exams/add') && request.method === 'POST') {
      if (!isD1Bound) return errorResponse("D1 Database not bound", 500);
      try {
        const body = await request.json();
        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();

        if (!title) return errorResponse("Field 'title' is required.");
        if (!slug) {
          slug = title
            .toLowerCase()
            .replace(/[^a-z0-9]+/g, '-')
            .replace(/^-+|-+$/g, '');
        }

        const existing = await executeD1(env, "SELECT id FROM exams WHERE slug = ? LIMIT 1", [slug]);
        if (existing && existing.length > 0) {
          return errorResponse(`An exam with slug '${slug}' already exists (ID: #${existing[0].id}). Please choose a unique slug.`);
        }

        const subtitle = String(body.subtitle || '').trim();
        const isLatest = (body.is_latest !== undefined && body.is_latest !== null) ? (body.is_latest ? 1 : 0) : 1;
        const isVisible = (body.is_visible !== undefined && body.is_visible !== null) ? (body.is_visible ? 1 : 0) : 1;
        const setOnTop = body.set_on_top ? 1 : 0;
        const locTypeId = String(body.location_type_id || 'States/UT').trim();
        const locId = normalizeLocationId(body.location_id);
        const marksRight = (body.marks_right !== undefined && body.marks_right !== '') ? parseFloat(body.marks_right) : 1.0;
        const marksWrong = (body.marks_wrong !== undefined && body.marks_wrong !== '') ? parseFloat(body.marks_wrong) : 0.25;
        const answerLink = String(body.answerkey_login_link || '').trim();
        const description = String(body.description || '').trim();
        const externalExamId = String(body.external_exam_id || '').trim();
        const youtubeLink = String(body.youtube_link || '').trim();

        const availableCols = await getTableColumns(env);
        const dataMap = {
          title, subtitle, slug, is_latest: isLatest, is_visible: isVisible, set_on_top: setOnTop,
          location_type_id: locTypeId, location_id: locId, marks_right: marksRight, marks_wrong: marksWrong,
          answerkey_login_link: answerLink, description, external_exam_id: externalExamId, youtube_link: youtubeLink
        };

        let colsToInsert = [];
        let placeholders = [];
        let valuesToInsert = [];

        Object.keys(dataMap).forEach(col => {
          if (availableCols.length === 0 || availableCols.includes(col)) {
            colsToInsert.push(col);
            placeholders.push('?');
            valuesToInsert.push(dataMap[col]);
          }
        });

        if (availableCols.includes('created_at')) {
          colsToInsert.push('created_at');
          placeholders.push("datetime('now')");
        }
        if (availableCols.includes('updated_at')) {
          colsToInsert.push('updated_at');
          placeholders.push("datetime('now')");
        }

        const insertSql = `INSERT INTO exams (${colsToInsert.join(', ')}) VALUES (${placeholders.join(', ')})`;
        const result = await executeD1(env, insertSql, valuesToInsert);
        const newId = result?.meta?.last_row_id || result?.lastRowId || null;

        return jsonResponse({
          success: true,
          message: "Exam added successfully!",
          id: newId,
          slug: slug
        }, 201);
      } catch (err) {
        return errorResponse("Failed to add exam: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 4: Update Exam (PUT /api/exams/:id or POST /api/exams/update)
    // -------------------------------------------------------------------------
    const updateMatch = pathname.match(/^\/api\/exams\/(\d+)$/);
    if ((updateMatch && (request.method === 'PUT' || request.method === 'POST')) || (pathname === '/api/exams/update' && request.method === 'POST')) {
      if (!isD1Bound) return errorResponse("D1 Database not bound", 500);
      try {
        const body = await request.json();
        const examId = updateMatch ? parseInt(updateMatch[1], 10) : parseInt(body.id, 10);
        if (isNaN(examId)) return errorResponse("Invalid or missing Exam ID for update.");

        const existing = await executeD1(env, "SELECT id, slug FROM exams WHERE id = ?", [examId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No exam found with ID: #${examId}`, 404);
        }

        const title = String(body.title || '').trim();
        let slug = String(body.slug || '').trim().toLowerCase();
        if (!title) return errorResponse("Title cannot be empty.");
        if (!slug) {
          slug = title.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
        }

        const slugCheck = await executeD1(env, "SELECT id FROM exams WHERE slug = ? AND id != ? LIMIT 1", [slug, examId]);
        if (slugCheck && slugCheck.length > 0) {
          return errorResponse(`Slug '${slug}' is already in use by another exam (ID: #${slugCheck[0].id}).`);
        }

        const subtitle = String(body.subtitle || '').trim();
        const isLatest = (body.is_latest !== undefined && body.is_latest !== null) ? (body.is_latest ? 1 : 0) : 1;
        const isVisible = (body.is_visible !== undefined && body.is_visible !== null) ? (body.is_visible ? 1 : 0) : 1;
        const setOnTop = body.set_on_top ? 1 : 0;
        const locTypeId = String(body.location_type_id || 'States/UT').trim();
        const locId = normalizeLocationId(body.location_id);
        const marksRight = (body.marks_right !== undefined && body.marks_right !== '') ? parseFloat(body.marks_right) : 1.0;
        const marksWrong = (body.marks_wrong !== undefined && body.marks_wrong !== '') ? parseFloat(body.marks_wrong) : 0.25;
        const answerLink = String(body.answerkey_login_link || '').trim();
        const description = String(body.description || '').trim();
        const externalExamId = String(body.external_exam_id || '').trim();
        const youtubeLink = String(body.youtube_link || '').trim();

        const availableCols = await getTableColumns(env);
        const dataMap = {
          title, subtitle, slug, is_latest: isLatest, is_visible: isVisible, set_on_top: setOnTop,
          location_type_id: locTypeId, location_id: locId, marks_right: marksRight, marks_wrong: marksWrong,
          answerkey_login_link: answerLink, description, external_exam_id: externalExamId, youtube_link: youtubeLink
        };

        let setClauses = [];
        let setValues = [];

        Object.keys(dataMap).forEach(col => {
          if (availableCols.length === 0 || availableCols.includes(col)) {
            setClauses.push(`${col} = ?`);
            setValues.push(dataMap[col]);
          }
        });

        if (availableCols.includes('updated_at')) {
          setClauses.push("updated_at = datetime('now')");
        }

        setValues.push(examId);
        const updateSql = `UPDATE exams SET ${setClauses.join(', ')} WHERE id = ?`;
        await executeD1(env, updateSql, setValues);

        return jsonResponse({
          success: true,
          message: "Exam updated successfully!",
          id: examId
        });
      } catch (err) {
        return errorResponse("Failed to update exam: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 5: Delete Exam (DELETE /api/exams/:id or POST /api/exams/delete)
    // -------------------------------------------------------------------------
    const deleteMatch = pathname.match(/^\/api\/exams\/(\d+)$/);
    if ((deleteMatch && request.method === 'DELETE') || (pathname === '/api/exams/delete' && (request.method === 'POST' || request.method === 'DELETE'))) {
      if (!isD1Bound) return errorResponse("D1 Database not bound", 500);
      try {
        let examId;
        if (deleteMatch) {
          examId = parseInt(deleteMatch[1], 10);
        } else {
          const body = await request.json().catch(() => ({}));
          examId = parseInt(body.id || url.searchParams.get('id'), 10);
        }

        if (isNaN(examId)) return errorResponse("Invalid or missing Exam ID for deletion.");

        const existing = await executeD1(env, "SELECT id, title FROM exams WHERE id = ?", [examId]);
        if (!existing || existing.length === 0) {
          return errorResponse(`No exam found with ID: #${examId}`, 404);
        }

        await executeD1(env, "DELETE FROM exams WHERE id = ?", [examId]);

        return jsonResponse({
          success: true,
          message: `Exam #${examId} ('${existing[0].title}') deleted successfully.`,
          id: examId
        });
      } catch (err) {
        return errorResponse("Failed to delete exam: " + err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 6: Get Single Exam Details (GET /api/exams/:id or :slug)
    // -------------------------------------------------------------------------
    const getSingleMatch = pathname.match(/^\/api\/exams\/([a-zA-Z0-9_-]+)$/);
    if (getSingleMatch && request.method === 'GET') {
      const identifier = getSingleMatch[1];
      if (!isD1Bound) {
        const found = DEFAULT_EXAMS.find(e => e.slug === identifier || String(e.id) === identifier);
        if (found) return jsonResponse({ success: true, data: found, isD1Bound: false });
        return errorResponse("Exam not found", 404);
      }
      try {
        const identifier = getSingleMatch[1];
        let sql = "SELECT * FROM exams WHERE slug = ? LIMIT 1";
        let params = [identifier];
        if (/^\d+$/.test(identifier)) {
          sql = "SELECT * FROM exams WHERE id = ? OR slug = ? LIMIT 1";
          params = [parseInt(identifier, 10), identifier];
        }
        const rows = await executeD1(env, sql, params);
        if (rows && rows.length > 0) {
          const item = rows[0];
          if (typeof item.location_id === 'string') {
            try { item.location_id = JSON.parse(item.location_id); } catch (e) {}
          }
          return jsonResponse({ success: true, data: item });
        }
        return errorResponse("Exam not found", 404);
      } catch (err) {
        return errorResponse(err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 7: List All Exams (GET /api/exams or /exams)
    // -------------------------------------------------------------------------
    if ((pathname === '/api/exams' || pathname === '/exams' || pathname === '/get_exams') && request.method === 'GET') {
      const search = url.searchParams.get('search') ? url.searchParams.get('search').trim() : '';
      const visibleOnly = url.searchParams.get('visible_only') === '1' || url.searchParams.get('active') === '1';
      const rawArray = url.searchParams.get('raw') === '1' || url.searchParams.get('format') === 'array';

      if (!isD1Bound) {
        let rows = JSON.parse(JSON.stringify(DEFAULT_EXAMS));
        if (visibleOnly) rows = rows.filter(r => r.is_visible);
        if (search) {
          const sLower = search.toLowerCase();
          rows = rows.filter(r => (r.title && r.title.toLowerCase().includes(sLower)) || (r.slug && r.slug.toLowerCase().includes(sLower)));
        }
        if (rawArray) {
          return new Response(JSON.stringify(rows), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' },
            status: 200
          });
        }
        return jsonResponse({
          success: true,
          count: rows.length,
          data: rows,
          isD1Bound: false
        });
      }
      try {
        const search = url.searchParams.get('search') ? url.searchParams.get('search').trim() : '';
        const visibleOnly = url.searchParams.get('visible_only') === '1' || url.searchParams.get('active') === '1';
        const rawArray = url.searchParams.get('raw') === '1' || url.searchParams.get('format') === 'array';

        let sql = "SELECT * FROM exams";
        let wheres = [];
        let params = [];

        if (visibleOnly) {
          wheres.push("is_visible = 1");
        }
        if (search) {
          wheres.push("(title LIKE ? OR slug LIKE ? OR external_exam_id LIKE ?)");
          const term = `%${search}%`;
          params.push(term, term, term);
        }

        if (wheres.length > 0) {
          sql += " WHERE " + wheres.join(" AND ");
        }
        sql += " ORDER BY set_on_top DESC, id DESC";

        let rows = [];
        try {
          rows = await executeD1(env, sql, params);
        } catch (err) {
          console.warn("API query fallback:", err.message);
        }

        if (!rows || rows.length === 0) {
          rows = JSON.parse(JSON.stringify(DEFAULT_EXAMS));
          if (visibleOnly) rows = rows.filter(r => r.is_visible);
          if (search) {
            const sLower = search.toLowerCase();
            rows = rows.filter(r => (r.title && r.title.toLowerCase().includes(sLower)) || (r.slug && r.slug.toLowerCase().includes(sLower)));
          }
        } else if (Array.isArray(rows)) {
          rows.forEach(r => {
            if (typeof r.location_id === 'string') {
              try { r.location_id = JSON.parse(r.location_id); } catch (e) {}
            }
          });
        }

        if (rawArray) {
          return new Response(JSON.stringify(rows), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' },
            status: 200
          });
        }

        return jsonResponse({
          success: true,
          count: rows.length,
          data: rows
        });
      } catch (err) {
        return errorResponse(err.message, 500);
      }
    }

    // -------------------------------------------------------------------------
    // ROUTE 8: Web Dashboard Portal (GET / or browser requests)
    // -------------------------------------------------------------------------
    if (request.method === 'GET') {
      let initialExams = [];
      let tableCols = [];
      let dbError = null;

      if (isD1Bound) {
        try {
          // 1. Auto-create table & migrate missing columns
          tableCols = await ensureExamsTable(env);
          const hasSetOnTop = tableCols.includes('set_on_top');
          const orderSql = hasSetOnTop ? "ORDER BY set_on_top DESC, id DESC" : "ORDER BY id DESC";

          // 2. Fetch existing exams
          initialExams = await executeD1(env, "SELECT * FROM exams " + orderSql);

          // 3. Auto-seed 27 default CBT exams if database table is empty!
          if (!initialExams || initialExams.length === 0) {
            await seedDefaultExams(env);
            initialExams = await executeD1(env, "SELECT * FROM exams " + orderSql);
          }

          if (Array.isArray(initialExams)) {
            initialExams.forEach(r => {
              if (typeof r.location_id === 'string') {
                try { r.location_id = JSON.parse(r.location_id); } catch (e) {}
              }
            });
          }
        } catch (e) {
          dbError = e.message;
          console.error("D1 Load Error:", e);
        }
      }

      // 4. Fallback: If D1 is not bound or query returned 0 rows, use 27 built-in default exams!
      if (!initialExams || initialExams.length === 0) {
        initialExams = JSON.parse(JSON.stringify(DEFAULT_EXAMS));
      }

      return renderExamsManagerPortal(isD1Bound, tableCols.length > 0, initialExams, dbError);
    }

    return errorResponse("Method or endpoint not found", 404);
  }
};

    let allExams = ${initialJson};
    let currentFilter = 'all';
    let searchQuery = '';
    let isSlugManual = false;

    // Initialize UI on load
    document.addEventListener('DOMContentLoaded', function() {

    function showToast(msg, type) {
      type = type || 'success';
      var c = document.getElementById('toastContainer');
      if (!c) return;
      var t = document.createElement('div');
      t.className = 'toast ' + (type === 'error' ? 'toast-error' : 'toast-success');
      t.innerHTML = (type === 'error' ? '❌ ' : '✅ ') + escapeHtml(msg);
      c.appendChild(t);
      setTimeout(function() {
        t.style.opacity = '0';
        t.style.transform = 'translateX(100%)';
        setTimeout(function() { t.remove(); }, 300);
      }, 3500);
    }

    function onSearchChange(val) {
      searchQuery = (val || '').trim().toLowerCase();
      renderAllViews();
    }

    function setFilter(filterName) {
      currentFilter = filterName;
      var tabs = document.querySelectorAll('.filter-tab');
      for (var i = 0; i < tabs.length; i++) {
        if (tabs[i].getAttribute('data-filter') === filterName) {
          tabs[i].classList.add('active');
        } else {
          tabs[i].classList.remove('active');
        }
      }
      renderAllViews();
    }

    function updateStats() {
      var total = allExams.length;
      var visible = 0;
      var top = 0;
      var latest = 0;
      for (var i = 0; i < allExams.length; i++) {
        if (allExams[i].is_visible == 1) visible++;
        if (allExams[i].set_on_top == 1) top++;
        if (allExams[i].is_latest == 1) latest++;
      }
      var hidden = total - visible;

      document.getElementById('statTotal').innerText = total;
      document.getElementById('statVisible').innerText = visible;
      document.getElementById('statTop').innerText = top;
      document.getElementById('statLatest').innerText = latest;

      document.getElementById('cntAll').innerText = total;
      document.getElementById('cntVisible').innerText = visible;
      document.getElementById('cntHidden').innerText = hidden;
      document.getElementById('cntTop').innerText = top;
      document.getElementById('cntLatest').innerText = latest;
    }

    function getFilteredExams() {
      return allExams.filter(function(item) {
        if (currentFilter === 'visible' && item.is_visible != 1) return false;
        if (currentFilter === 'hidden' && item.is_visible == 1) return false;
        if (currentFilter === 'top' && item.set_on_top != 1) return false;
        if (currentFilter === 'latest' && item.is_latest != 1) return false;

        if (searchQuery) {
          var matchTitle = (item.title || '').toLowerCase().indexOf(searchQuery) !== -1;
          var matchSlug = (item.slug || '').toLowerCase().indexOf(searchQuery) !== -1;
          var matchExt = String(item.external_exam_id || '').toLowerCase().indexOf(searchQuery) !== -1;
          var matchId = String(item.id || '') === searchQuery;
          if (!matchTitle && !matchSlug && !matchExt && !matchId) return false;
        }

        return true;
      });
    }

    function renderAllViews() {
      var filtered = getFilteredExams();
      renderCards(filtered);
      renderTable(filtered);
    }

    // Render Mobile Cards View

          } else {
            locText = String(exam.location_id);
          }
        }

        var isVis = isTruthy(exam.is_visible, true);
        var isTop = isTruthy(exam.set_on_top, false);
        var isLat = isTruthy(exam.is_latest, true);

        return '<div class="exam-mobile-card">' +
          '<div class="card-top-row">' +
            '<span class="badge-id">#' + exam.id + '</span>' +
            '<div class="card-status-bar">' +

              '</span>' +
            '</div>' +
          '</div>' +

          '<div class="card-title">' + escapeHtml(exam.title) + '</div>' +
          (exam.subtitle ? '<div class="card-subtitle">' + escapeHtml(exam.subtitle) + '</div>' : '') +

              '</span>' +
            '</div>' +
            '<div class="card-detail-item">' +
              '<span class="card-detail-label">Location Type</span>' +

    // Safe Truthy check for SQLite boolean/integer/string values
    function isTruthy(val, defaultVal) {
      if (val === undefined || val === null) return !!defaultVal;
      if (typeof val === 'boolean') return val;
      if (typeof val === 'number') return val === 1;
      if (typeof val === 'string') {
        var s = val.trim().toLowerCase();
        return s === '1' || s === 'true';
      }
      return Boolean(val);
    }

    // Format location array/string cleanly for text input
    function formatLocationInput(loc) {
      if (!loc) return 'All India';
      if (Array.isArray(loc)) return loc.join(', ');
      if (typeof loc === 'string') {
        var trimmed = loc.trim();
        if (trimmed.startsWith('[') && trimmed.endsWith(']')) {
          try {
            var parsed = JSON.parse(trimmed);
            if (Array.isArray(parsed)) return parsed.join(', ');
          } catch (e) {}
        }
        return trimmed;
      }
      return String(loc);
    }

    function slugify(text) {
      return (text || '')
        .toString()
        .toLowerCase()
        .trim()
        .replace(/\s+/g, '-')
        .replace(/[^a-z0-9\-]+/g, '')
        .replace(/\-\-+/g, '-')
        .replace(/^-+/, '')
        .replace(/-+$/, '');
    }

    async function handleFormSubmit(e) {
      e.preventDefault();
      var saveBtn = document.getElementById('saveExamBtn');
      var examId = document.getElementById('formExamId').value;
      var isEdit = !!examId;

      var title = document.getElementById('formTitle').value.trim();
      var slug = document.getElementById('formSlug').value.trim();

      if (!title) {
        alert('Please enter the Exam Title.');
        document.getElementById('formTitle').focus();
        return;
      }
      if (!slug) {
        slug = slugify(title);
        document.getElementById('formSlug').value = slug;
      }

      saveBtn.disabled = true;
      saveBtn.innerText = '⏳ Saving...';

      var payload = {
        title: title,
        slug: slug,
        subtitle: document.getElementById('formSubtitle').value.trim(),
        marks_right: parseFloat(document.getElementById('formMarksRight').value) || 1.0,
        marks_wrong: parseFloat(document.getElementById('formMarksWrong').value) || 0.25,
        location_type_id: document.getElementById('formLocationTypeId').value,
        location_id: document.getElementById('formLocationId').value.trim(),
        external_exam_id: document.getElementById('formExternalExamId').value.trim(),
        answerkey_login_link: document.getElementById('formAnswerLink').value.trim(),
        youtube_link: document.getElementById('formYoutubeLink').value.trim(),
        description: document.getElementById('formDescription').value.trim(),
        is_visible: document.getElementById('formIsVisible').checked ? 1 : 0,
        set_on_top: document.getElementById('formSetOnTop').checked ? 1 : 0,
        is_latest: document.getElementById('formIsLatest').checked ? 1 : 0
      };

      try {
        var res;
        if (isEdit) {
          payload.id = parseInt(examId, 10);
          res = await fetch('/api/exams/' + examId, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
        } else {
          res = await fetch('/api/exams', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
        }

        var data = await res.json();
        saveBtn.disabled = false;
        saveBtn.innerHTML = isEdit ? '💾 Update Exam' : '💾 Create Exam';

        if (!data.success) {
          alert('Error: ' + (data.error || 'Failed to save exam'));
          return;
        }

        closeModal('examModal');
        showToast(isEdit ? 'Exam updated successfully!' : 'Exam added successfully!');
        await refreshData();
      } catch (err) {
        saveBtn.disabled = false;
        saveBtn.innerHTML = isEdit ? '💾 Update Exam' : '💾 Create Exam';
        alert('Network Error: ' + err.message);
      }
    }

    async function toggleFlag(id, field) {
      var exam = null;
      for (var i = 0; i < allExams.length; i++) {
        if (allExams[i].id == id) { exam = allExams[i]; break; }
      }
      if (!exam) return;

      var currentVal = exam[field] == 1 ? 1 : 0;
      var newVal = currentVal ? 0 : 1;
      exam[field] = newVal;
      renderAllViews();
      updateStats();

      try {
        var res = await fetch('/api/exams/toggle', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ id: id, field: field, value: newVal })
        });
        var data = await res.json();
        if (!data.success) {
          exam[field] = currentVal;
          renderAllViews();
          updateStats();
          showToast('Toggle failed: ' + data.error, 'error');
        } else {
          showToast('Updated ' + field + ' for Exam #' + id);
        }
      } catch (err) {
        exam[field] = currentVal;
        renderAllViews();
        updateStats();
        showToast('Network error on toggle', 'error');
      }
    }
